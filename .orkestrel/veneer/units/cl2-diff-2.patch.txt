+// condition that switches at it therefore cannot name two different widths, and the proofs guard
+// the emission rather than a hand-copied pair. The values live here rather than in `_tokens.scss`

+@function breakpoints() {

+// Returns one boundary's width, refusing a name the ramp does not carry so a typo fails the build
+// rather than emitting a condition nothing matches.
+@function breakpoint($name) {
+	$widths: breakpoints();
+	@if not map.has-key($widths, $name) {
+		@error 'Unknown breakpoint #{$name}. The ramp carries #{map.keys($widths)}.';
+	}
+	@return map.get($widths, $name);

+// Emits its content for every viewport at and above the named boundary.

+// there rather than a condition that is always true.

+/**
+ * Collects the media conditions that gate one selector's declarations.
+ *
+ * @param rules - The cascade rules to read, such as the ones a {@link collectNestedRules} walk
+ *   returns or the installed `readRules` export reports for the whole page.
+ * @param selector - The gated rule's selector text, compared the way
+ *   {@link collectScopeProperties} compares one.
+ * @returns The `conditionText` of every media rule holding a rule with that selector, in the order
+ *   the rules were given. A selector no media rule gates reports nothing, which is the same reading
+ *   a selector no rule declares gives.
+ *
+ * @remarks
+ * A mixin that wraps its content in `@media` is proved by the condition it emitted, and the CSSOM
+ * reports that condition on the grouping rule rather than on the style rule the caller named. This
+ * walks from the media rule down, so a caller asks about the selector it wrote and reads back the
+ * gate around it.
+ *
+ * A browser normalizes a condition as it serializes it, so compare the width the condition names
+ * through the `parseMediaWidth` export in `tests/setupStyles.ts` rather than the text.
+ */
+export function collectMediaConditions(
+	rules: readonly CSSRule[],
+	selector: string,
+): readonly string[] {
+	const wanted = normalizeSelectorText(selector)
+	return rules.flatMap((rule) =>
+		rule instanceof CSSMediaRule &&
+		collectNestedRules(rule.cssRules).some(
+			(nested) =>
+				nested instanceof CSSStyleRule && normalizeSelectorText(nested.selectorText) === wanted,
+		)
+			? [rule.conditionText]
+			: [],
+	)
+}

+/**
+ * Parses the viewport width one media condition switches at.
+ *
+ * @param condition - The condition text, such as the `conditionText` a `CSSMediaRule` carries.
+ * @returns The width in pixels, and `undefined` where the condition names no width in pixels.
+ *
+ * @remarks
+ * A browser is free to serialize `(width >= 576px)` back as `(min-width: 576px)`, and the two
+ * spellings name one boundary, so a proof comparing an emitted condition with a published token
+ * reads the number rather than the text. The first alternative reads the range form Veneer
+ * authors and the second the legacy pair, each carrying the width as the one length in the
+ * condition. A width written in any other unit reads as `undefined` rather than as a pixel count,
+ * because the comparison this serves is against a token the document resolved to pixels.
+ */
+export function parseMediaWidth(condition: string): number | undefined {
+	const matched =
+		/\bwidth\s*[<>]=?\s*(-?\d+(?:\.\d+)?)px|\b(?:min|max)-width\s*:\s*(-?\d+(?:\.\d+)?)px/u.exec(
+			condition,
+		)
+	if (matched === null) return undefined
+	const width = matched[1] ?? matched[2]
+	if (width === undefined) return undefined
+	return Number.parseFloat(width)
+}
