+/**

+ *
+ * @param host - The host to read, whose attribute mutations and received clicks are recorded.
+ * @param action - The action to run against the rendered surface. A refusal it raises as an `Error`
+ *   is recorded as the reading's refusal rather than propagated; anything else it raises propagates.
+ * @returns The reading after the action: the host's ordered classes, its attributes, the attribute
+ *   names mutated in order, each received click's cancellation, the focused accessible name, and the
+ *   refusal.
+ *
+ * @remarks
+ * The recording is the official recorder's, verb for verb, so the two readings are comparable: one
+ * document-level `click` listener counting the clicks whose target is the host itself, and one
+ * `MutationObserver` over the host's attributes. The queue is drained with `takeRecords` before the
+ * observer is disconnected, so a mutation the browser had not yet delivered is still counted once.
+ *
+ * Focus is read from `:focus` rather than from `activeElement`, because a document where nothing
+ * holds focus reports its own body as active and would name the whole page.

+ * @example
+ * ```ts
+ * const reading = await recordState(host, () => clickAccessible('button', 'Toggle'))
+ * ```

+export async function recordState(
+	host: HTMLElement,
+	action: () => Promise<void>,
+): Promise<ButtonReading> {
+	const mutations: string[] = []
+	const clicks: boolean[] = []
+	const controller = new AbortController()
+	const observer = new MutationObserver((records) => {
+		for (const record of records) {
+			if (record.attributeName !== null) mutations.push(record.attributeName)
+		}
+	})
+	host.ownerDocument.addEventListener(
+		'click',
+		(event) => {
+			if (event.target === host) clicks.push(event.defaultPrevented)
+		},
+		{ signal: controller.signal },
+	)
+	observer.observe(host, { attributes: true })
+	let refusal: string | undefined
+	try {
+		try {
+			await action()
+		} catch (error) {
+			if (!(error instanceof Error)) throw error
+			refusal = error.message
+		}
+		for (const record of observer.takeRecords()) {
+			if (record.attributeName !== null) mutations.push(record.attributeName)
+		}
+	} finally {
+		observer.disconnect()
+		controller.abort()
+	}
+	const focused = host.ownerDocument.querySelector(':focus')
+	return {
+		classes: [...host.classList],
+		attributes: Object.fromEntries(
+			[...host.attributes].map((attribute) => [attribute.name, attribute.value]),
+		),
+		mutations,
+		clicks,
+		focus: focused === null ? undefined : readName(focused),

+	 * @param host - The element receiving the section.

+		this.#host = host
+		this.#section = document.createElement('section')
+		this.#mount()

+		return this.#host
+	}
+

+		for (const engine of this.#engines) engine.destroy()
+		this.#engines.length = 0
+		this.#section.remove()

+		this.#section.append(paragraph, grid)
+		this.#host.append(this.#section)
+	}

+	it('owns every plain host and leaves every delegated host to the delegation', () => {
+		const host = mount(build('div'))
+		const section = new ButtonSection(host)
+		try {
+			const rendered = BUTTON_SPECIMENS.map((specimen) => resolveButton(host, specimen.name))
+			const delegated = rendered.filter((element) => element.matches(BUTTON_SELECTOR))
+			const plain = rendered.filter((element) => !element.matches(BUTTON_SELECTOR))
+			// The rendered partition is set beside the table's own declaration, so the selector and the
+			// specimen that carries it have to agree before either side of the sweep means anything.
+			expect(delegated.map((element) => readName(element))).toStrictEqual(
+				BUTTON_SPECIMENS.filter(
+					(specimen) => specimen.attributes['data-bs-toggle'] !== undefined,
+				).map((specimen) => specimen.name),
+			)
+			expect(plain).not.toStrictEqual([])
+			// Reclaiming a host after destruction cannot prove the section held it beforehand, so the
+			// whole partition is read while the section is live: a second engine over a host the
+			// section owns is refused by name, and a delegated host accepts one because the section
+			// bound none to it. Each probe engine is destroyed at once, and destruction restores the
+			// class and the attribute the engine captured, so the markup is where it started.
+			const refused: string[] = []
+			for (const element of plain) {
+				try {
+					new Button(element).destroy()
+					refused.push(`${readName(element)}: accepted`)
+				} catch (error) {
+					refused.push(`${readName(element)}: ${isAppError(error) ? error.code : String(error)}`)
+				}
+			}
+			expect(refused).toStrictEqual(
+				plain.map((element) => `${readName(element)}: BUTTON_HOST_OWNED`),
+			)
+			const accepted: string[] = []
+			for (const element of delegated) {
+				const rival = new Button(element)
+				try {
+					accepted.push(`${readName(element)}: ${String(rival.host === element)}`)
+				} finally {
+					rival.destroy()
+				}
+			}
+			expect(accepted).toStrictEqual(delegated.map((element) => `${readName(element)}: true`))
+			// The refusal is caught rather than thrown at the matcher, so the code and the sentence are
+			// both read off the one error the section's ownership produced.
+			let raised: unknown
+			try {
+				raised = new Button(resolveButton(host, 'Primary'))
+			} catch (error) {
+				raised = error
+			}
+			expect(isAppError(raised) && raised.code).toBe('BUTTON_HOST_OWNED')
+			expect(raised instanceof Error && raised.message).toBe('Button host already has a live owner')
+		} finally {
+			section.destroy()
+			host.remove()
+		}
+	})
