import { DropdownSection } from '@app/browser'
import { build, mount, readStyle } from '@orkestrel/test/browser'
import { expect, it } from 'vitest'
import { commands } from 'vitest/browser'
import release from '../../node_modules/bootstrap/dist/css/bootstrap.css?inline'

it('reads the dropdown header weight in each cascade', async () => {
	await import('../../src/styles/index.scss')
	const host = mount(build('div'))
	const section = new DropdownSection(host)
	const header = host.querySelector('.dropdown-header')
	const veneer = header === null ? 'none' : `${header.tagName} ${readStyle(header, 'font-weight')}`
	section.destroy()
	host.remove()
	const frame = build('iframe')
	document.body.append(frame)
	const inner = frame.contentDocument
	if (inner === null) throw new Error('no frame document')
	inner.open()
	inner.write(`<!doctype html><style>${release}</style><ul class="dropdown-menu show"><li><h6 class="dropdown-header">Shipment</h6></li></ul>`)
	inner.close()
	const target = inner.querySelector('.dropdown-header')
	const bootstrap = target === null ? 'none' : getComputedStyle(target).fontWeight
	frame.remove()
	await commands.writeFile('tmp/probe/bcf-weight.txt', JSON.stringify({ veneer, bootstrap }))
	expect(veneer).not.toBe('none')
})
