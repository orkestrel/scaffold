import { NavbarSection } from '@app/browser'
import { build, mount, readStyle } from '@orkestrel/test/browser'
import { expect, it } from 'vitest'
import { commands } from 'vitest/browser'

it('reads the opened bar menu cascade', async () => {
	await import('../../src/styles/index.scss')
	await import('../../app/browser/styles/index.scss')
	const host = mount(build('div'))
	const section = new NavbarSection(host)
	const menu = host.querySelector('[data-specimen="Navbar opened"] .dropdown-menu.show')
	if (menu === null) throw new Error('no menu')
	const toggle = menu.previousElementSibling
	if (toggle === null) throw new Error('no toggle')
	const reading = {

		width: window.innerWidth,
		attribute: menu.getAttribute('data-bs-popper'),
		position: readStyle(menu, 'position'),
		top: readStyle(menu, 'top'),
		left: readStyle(menu, 'left'),
		marginTop: readStyle(menu, 'margin-top'),
		gap: menu.getBoundingClientRect().top - toggle.getBoundingClientRect().bottom,
		specimenHeight: menu.closest('[data-specimen]')?.getBoundingClientRect().height,
	}
	section.destroy()
	host.remove()
	await commands.writeFile(`tmp/probe/bcf3-cascade-${String(import.meta.env.VITE_BCF3 ?? 'x')}.json`, JSON.stringify(reading))
	expect(reading.position).not.toBe('')
})
