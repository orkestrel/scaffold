import veneer from '../../../../dist/src/styles/index.css?raw'
import release from '../../../../node_modules/bootstrap/dist/css/bootstrap.css?raw'
import { expect, it } from 'vitest'

const MARKUP = ['bg-layered', 'bg-plain', 'bg-primary']
	.map(
		(utility) =>
			`<div class="offcanvas-lg offcanvas-end ${utility}" data-utility="${utility}"><div class="offcanvas-body ${utility}">Body</div></div>`,
	)
	.join('')
const PROBES =
	'@layer utilities { .bg-layered { background-color: rgb(1, 2, 3) !important; } } .bg-plain { background-color: rgb(4, 5, 6) !important; }'
const RELEASE_PROBES =
	'.bg-layered { background-color: rgb(1, 2, 3) !important; } .bg-plain { background-color: rgb(4, 5, 6) !important; }'

it('reads an important background utility on the inline and the fixed lg panel', async () => {
	const readings: Record<string, unknown> = {}
	for (const [sheet, css, probes] of [
		['veneer', veneer, PROBES],
		['release', release, RELEASE_PROBES],
	] as const)
		for (const width of [991, 992]) {
			const frame = document.createElement('iframe')
			frame.style.width = `${String(width)}px`
			frame.style.height = '400px'
			frame.style.border = '0'
			document.body.append(frame)
			const loaded = new Promise((resolve) => frame.addEventListener('load', resolve, { once: true }))
			frame.srcdoc = `<!doctype html><html><head><style>${css}</style><style>${probes}</style></head><body>${MARKUP}</body></html>`
			await loaded
			const doc = frame.contentDocument
			const view = frame.contentWindow
			if (doc === null || view === null) throw new Error('no frame document')
			readings[`${sheet} ${String(width)} innerWidth`] = view.innerWidth
			for (const panel of doc.querySelectorAll<HTMLElement>('[data-utility]')) {
				const body = panel.querySelector('.offcanvas-body')
				if (body === null) throw new Error('no body')
				readings[`${sheet} ${String(width)} ${panel.dataset.utility ?? ''}`] = {
					position: view.getComputedStyle(panel).position,
					panel: view.getComputedStyle(panel).backgroundColor,
					body: view.getComputedStyle(body).backgroundColor,
				}
			}
			frame.remove()
		}
	console.log(JSON.stringify(readings, undefined, 1))
	expect(Object.keys(readings).length).toBeGreaterThan(0)
})
