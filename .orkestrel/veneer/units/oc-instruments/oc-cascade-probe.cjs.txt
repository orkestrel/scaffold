const postcss = require('postcss')
const fs = require('fs')
const css = fs.readFileSync('dist/src/styles/index.css', 'utf8')
const inv = require('./tests/fixtures/oracle/inventory.json').components.offcanvas.selectors
const norm = (c) => c === undefined ? '' : c.replace(/^@media\s*/, '').split(/\s+and\s+/).map((f) => {
  let m = /^\(min-width:\s*([\d.]+)px\)$/.exec(f); if (m) return `(width>=${m[1]}px)`
  m = /^\(max-width:\s*([\d.]+)px\)$/.exec(f); if (m) return `(width<${Math.ceil(Number(m[1]))}px)`
  return f.replace(/\s+/g, '').replace('width>=', 'width>=').replace('width<', 'width<')
}).join(' and ')
const sel = (s) => s.replace(/\s+/g, ' ').replace(/\s*>\s*/g, ' > ').trim()
const built = new Map()
postcss.parse(css).walkRules((rule) => {
  const conds = []; for (let n = rule.parent; n && n.type === 'atrule'; n = n.parent) if (n.name === 'media') conds.unshift(n.params)
  const c = conds.map((x) => norm(x)).join(' and ').replace(/\s+/g, '')
  for (const s of rule.selectors) if (/\.offcanvas/.test(s)) {
    const k = `${sel(s)} @ ${c}`
    const d = built.get(k) ?? []
    rule.walkDecls((x) => d.push(`${x.prop}: ${x.value}${x.important ? ' !important' : ''}`))
    built.set(k, d)
  }
})
const recorded = new Set(inv.map((r) => `${sel(r.selector)} @ ${norm(r.condition).replace(/\s+/g, '')}`))
const missing = [...recorded].filter((k) => !built.has(k))
const extra = [...built.keys()].filter((k) => !recorded.has(k))
console.log('recorded selector-condition sites:', recorded.size, 'built sites naming offcanvas classes:', built.size)
console.log('missing from cascade:', JSON.stringify(missing))
console.log('built beyond the record:', JSON.stringify(extra))
for (const [k, d] of built) console.log(k, '|', d.join('; '))
