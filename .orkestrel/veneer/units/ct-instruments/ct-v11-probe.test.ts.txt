import { requireValue } from '@orkestrel/test'
import {
	blendColor,
	holdAccessible,
	hoverAccessible,
	measureContrast,
	measureLuminance,
	parseCSSColor,
	readStyle,
	releasePointer,
	stageMedia,
} from '@orkestrel/test/browser'
import { expect, it } from 'vitest'
import { scene } from '../../setupBrowser.js'

it('measures the dark state mixer', async () => {
	const out: string[] = []
	for (const mixer of ['', '--vn-state-mixer: var(--vn-palette-black-base);']) {
		for (const role of ['', 'btn btn-primary', 'btn btn-secondary', 'btn btn-info', 'btn btn-warning', 'btn btn-dark', 'btn btn-light', 'btn']) {
			const container = scene.mount(
				`<div data-bs-theme="dark" style="background-color: var(--bs-body-bg)"><div style="${mixer}">${role ? `<button class="${role}">` : '<button>'}Specimen</button></div></div>`,
			)
			const button = requireValue(container.querySelector('button'), 'b')
			const page = requireValue(parseCSSColor(readStyle(requireValue(container.firstElementChild, 'p'), 'background-color')), 'page')
			await stageMedia({ motion: false })
			const read = (): readonly number[] => blendColor(requireValue(parseCSSColor(readStyle(button, 'background-color')), 'c'), page)
			const rest = read()
			await hoverAccessible('Specimen')
			const hover = read()
			await holdAccessible('Specimen')
			const active = read()
			await releasePointer()
			out.push(`${mixer ? 'black' : 'white'} ${role || 'bare'} rest-L ${measureLuminance(rest).toFixed(4)} hover-L ${measureLuminance(hover).toFixed(4)} active-L ${measureLuminance(active).toFixed(4)} hover-vs-rest ${measureContrast(hover, rest).toFixed(3)} active-vs-rest ${measureContrast(active, rest).toFixed(3)}`)
			scene.clear()
		}
	}
	expect(out).toEqual([])
})
