import { writeFileSync } from 'node:fs'
import { expect, it } from 'vitest'
import {
	collectShippedComponents,
	collectValueGaps,
	compileExpandedCascade,
	readCascadeBlocks,
	readCompatibility,
	readOracleInventory,
} from '../../tests/setupServer.js'

it('measures every site the property fallback reaches', () => {
	const inventory = readOracleInventory()
	const shipped = collectShippedComponents(readCompatibility())
	const blocks = readCascadeBlocks(compileExpandedCascade())
	Reflect.set(globalThis, 'ctFallback', [])
	collectValueGaps(blocks, inventory, shipped)
	const reached: unknown = Reflect.get(globalThis, 'ctFallback')
	const theme = new Set(
		(inventory.components.theme?.rules ?? []).map((rule) => rule.selector.replaceAll(' ', '')),
	)
	writeFileSync(
		'tmp/probe/ct-x12.json',
		JSON.stringify({ shipped: shipped.includes('theme'), theme: [...theme], reached }, undefined, '\t'),
	)
	expect(Array.isArray(reached)).toBe(true)
}, 120_000)
