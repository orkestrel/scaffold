+import type { SectionInterface } from '../types.js'
+import { CONTENT_COPY, CONTENT_SPECIMENS } from '../constants.js'
+
+/**
+ * Renders the declared text specimens into a named region.
+ *
+ * @example
+ * ```ts
+ * const section = new ContentSection(document.body)
+ * section.destroy()
+ * ```
+ */
+export class ContentSection implements SectionInterface {
+	readonly #host: HTMLElement
+	readonly #section: HTMLElement
+
+	/**
+	 * Mounts the region and its specimens in table order.
+	 *
+	 * @param host - The element receiving the section.
+	 */
+	constructor(host: HTMLElement) {
+		this.#host = host
+		this.#section = document.createElement('section')
+		this.#section.setAttribute('aria-label', CONTENT_COPY.region)
+		const paragraph = document.createElement('p')
+		paragraph.textContent = CONTENT_COPY.paragraph
+		this.#section.append(paragraph)
+		for (const specimen of CONTENT_SPECIMENS) {
+			const container = document.createElement('div')
+			container.setAttribute('data-specimen', specimen.name)
+			container.innerHTML = specimen.markup
+			this.#section.append(container)
+		}
+		this.#host.append(this.#section)
+	}
+
+	get host(): HTMLElement {
+		return this.#host
+	}
+
+	destroy(): void {
+		this.#section.remove()
+	}
+}

+import { requireValue } from '@orkestrel/test'
+import { readStyle, releaseMedia, stageMedia } from '@orkestrel/test/browser'
+import { afterEach, describe, expect, it } from 'vitest'
+import { specimens } from '../../setupBrowser.js'
+
+afterEach(async () => {
+	specimens.clear()
+	await releaseMedia()
+})
+
+describe('document reset', () => {
+	it('sizes elements and their pseudo-elements with the border box', () => {
+		const host = specimens.mount('<div class="box">Box</div>')
+		specimens.load('.box::before, .box::after { content: "box" }')
+		const element = requireValue(host.firstElementChild, 'No box')
+		expect(readStyle(element, 'box-sizing')).toBe('border-box')
+		expect(getComputedStyle(element, '::before').boxSizing).toBe('border-box')
+		expect(getComputedStyle(element, '::after').boxSizing).toBe('border-box')
+	})
+	it('hides against normal classes, unlayered paint, and later-layer important paint', () => {
+		const host = specimens.mount(
+			'<div hidden class="ordinary">Ordinary</div><div hidden class="unlayered">Unlayered</div><div hidden class="important">Important</div>',
+		)
+		specimens.load(
+			'@layer components { .ordinary { display: block } } .unlayered { display: flex } @layer utilities { .important { display: grid !important } }',
+		)
+		expect([...host.children].map((element) => readStyle(element, 'display'))).toEqual([
+			'none',
+			'none',
+			'none',
+		])
+		for (const element of host.children) element.removeAttribute('hidden')
+		expect([...host.children].map((element) => readStyle(element, 'display'))).toEqual([
+			'block',
+			'flex',
+			'grid',
+		])
+	})
+	it('scrolls smoothly only when the media preference permits motion', async () => {
+		await stageMedia({ motion: true })
+		expect(readStyle(document.documentElement, 'scroll-behavior')).toBe('smooth')
+		await stageMedia({ motion: false })
+		expect(readStyle(document.documentElement, 'scroll-behavior')).toBe('auto')
+	})
+})
