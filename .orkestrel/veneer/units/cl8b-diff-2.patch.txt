+import { TOKEN_NAMES } from '@src/core'

+import { afterEach, describe, expect, it } from 'vitest'
+import { specimens, visitBreakpoint } from '../../../setupBrowser.js'
+import { GAP_STEP_CASES, GRID_BREAKPOINT_CASES } from '../../../setupStyles.js'

+afterEach(() => {
+	specimens.clear()

+describe('gap utilities', () => {
+	it.each(GRID_BREAKPOINT_CASES)(
+		'resolves every step around the $name boundary',
+		async ({ name, boundary, readings }) => {
+			const infix = boundary === 0 ? '' : `-${name}`
+			const host = specimens.mount(
+				['g', 'gx', 'gy', 'row-gap']
+					.flatMap((key) =>
+						GAP_STEP_CASES.map(
+							({ step }) => `<div class="row ${key}${infix}-${String(step)}"><div></div></div>`,
+						),
+					)
+					.join(''),
+			)
+			for (const viewport of readings) {
+				await visitBreakpoint(viewport, () => {
+					expect(window.innerWidth).toBe(viewport)
+					for (const key of ['g', 'gx', 'gy', 'row-gap']) {
+						for (const { step, pixels, token, value } of GAP_STEP_CASES) {
+							const row = requireValue(
+								host.querySelector(`.${key}${infix}-${String(step)}`),
+								'No step row',
+							)
+							const child = requireValue(row.firstElementChild, 'No step child')
+							const horizontal = viewport >= boundary && (key === 'g' || key === 'gx') ? pixels : 24
+							const vertical = viewport >= boundary && (key === 'g' || key === 'gy') ? pixels : 0
+							expect(readToken(row, token).replace(/^\./u, '0.')).toBe(value)
+							expect(readPixels(row, 'margin-left')).toBe(horizontal === 0 ? 0 : -horizontal / 2)
+							expect(readPixels(row, 'margin-right')).toBe(horizontal === 0 ? 0 : -horizontal / 2)
+							expect(readPixels(child, 'padding-left')).toBe(horizontal / 2)
+							expect(readPixels(child, 'padding-right')).toBe(horizontal / 2)
+							expect(readPixels(row, 'margin-top')).toBe(vertical === 0 ? 0 : -vertical)
+							expect(readPixels(child, 'margin-top')).toBe(vertical)
+							expect(readStyle(row, 'row-gap')).toBe(
+								viewport >= boundary && key === 'row-gap' ? `${String(pixels)}px` : 'normal',
+							)
+						}
+					}
+				})
+			}
+		},
+	)
+
+	it('keeps the steps independent of density and follows a retuned step on every axis', () => {
+		const host = specimens.mount(
+			`<div class="row"><div></div></div><div class="row g-4 row-gap-4" style="padding-top: var(${TOKEN_NAMES.space[3]})"><div></div></div>`,
+		)
+		const baseline = requireValue(host.querySelector('.row'), 'No default row')
+		const row = requireValue(host.querySelector('.g-4'), 'No retuned row')
+		const child = requireValue(row.firstElementChild, 'No retuned child')
+		expect(readPixels(row, 'padding-top')).toBe(6)
+		document.documentElement.style.setProperty(TOKEN_NAMES.factor.density, '2')
+		try {

+			expect(readPixels(row, 'margin-left')).toBe(readPixels(baseline, 'margin-left'))
+			expect(readPixels(child, 'padding-left')).toBe(12)
+			expect(readPixels(row, 'margin-top')).toBe(-24)

+			expect(readPixels(row, 'row-gap')).toBe(32)
