+import type { ContentSpecimen, SectionCopy, SectionInterface } from '../types.js'
+
+/**
+ * Renders a declared specimen table into a named region and owns that region until destruction.
+ *
+ * @remarks
+ * Construction mounts the region: a constructed section is a mounted section, so there is no second
+ * lifecycle verb and no unmounted state to tear down. The copy and the table are the whole of what
+ * separates one specimen section from another, so each named section supplies its own pair and
+ * inherits every behavior from here.
+ *
+ * @example
+ * ```ts
+ * const section = new SpecimenSection(document.body, CONTENT_COPY, CONTENT_SPECIMENS)
+ * section.destroy()
+ * ```
+ */
+export class SpecimenSection implements SectionInterface {
+	readonly #host: HTMLElement
+	readonly #section: HTMLElement
+
+	/**
+	 * Mounts the region and its specimens in table order.
+	 *
+	 * @param host - The element receiving the section.
+	 * @param copy - The region's accessible name and lead paragraph.
+	 * @param specimens - The specimens the region renders, in render order.
+	 */
+	constructor(host: HTMLElement, copy: SectionCopy, specimens: readonly ContentSpecimen[]) {
+		this.#host = host
+		this.#section = document.createElement('section')
+		this.#section.setAttribute('aria-label', copy.region)
+		const paragraph = document.createElement('p')
+		paragraph.textContent = copy.paragraph
+		this.#section.append(paragraph)
+		for (const specimen of specimens) {
+			const container = document.createElement('div')
+			container.setAttribute('data-specimen', specimen.name)
+			container.innerHTML = specimen.markup
+			this.#section.append(container)
+		}
+		this.#host.append(this.#section)
+	}
+
+	get host(): HTMLElement {
+		return this.#host
+	}
+
+	destroy(): void {
+		this.#section.remove()
+	}
+}

+			// this region carries names no shipped table holds and the reading is of the base itself.
+			expect([...region.children].map((element) => element.localName)).toEqual(['p', 'div', 'div'])
+		} finally {
+			section.destroy()
+			host.remove()
+		}
+	})

+		const host = mount(build('div'))

+		try {
+			const region = requireValue(host.querySelector('section'), 'No region')

+		} finally {
+			section.destroy()
+			twin.destroy()
