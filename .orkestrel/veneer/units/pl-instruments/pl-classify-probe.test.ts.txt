import { writeFileSync } from 'node:fs'
import { afterAll, beforeAll, it } from 'vitest'
import { readBuiltCascade, SheetReader } from '../../tests/setupServer.js'
import { compileProfile, STAGE_TIMEOUT, stage, TAILWIND_PATHS } from '../../tests/setupService.js'
import { collectTypeSelectors, ELEMENT_TAGS, NEUTRAL_MARKUP } from '../../tests/setupStyles.js'

const EMULATION = `@layer theme.pl {
select { background-color: rgb(255, 255, 255); contain: size; contain-intrinsic-height: 21px; }
table { border-color: currentcolor; }
}`

const profile = await compileProfile(TAILWIND_PATHS.preflight)
const reset = new SheetReader(profile).layers.filter((layer) => layer.name === 'base')
const overlap = collectTypeSelectors(reset.flatMap((layer) => layer.selectors)).filter((tag) =>
	ELEMENT_TAGS.some(([, tags]) => tags.includes(tag)),
)

beforeAll(async () => {
	await stage.open()
}, STAGE_TIMEOUT)
afterAll(async () => {
	await stage.destroy()
})

it('classifies', async () => {
	const lines: string[] = []
	const baseCss = reset
		.flatMap((layer) => layer.declarations)
		.map((d) => `${d.selector} { ${d.property}: ${d.value}${d.important ? ' !important' : ''} }`)
		.join('\n')
	const rules = await stage.expand(baseCss)
	const all = await stage.expand(profile)
	const properties = [...new Set(all.flatMap((r) => r.properties.filter((p) => !p.startsWith('--'))))]
	lines.push(`properties has height: ${properties.includes('height')}`)
	for (const emulate of [false, true]) {
		lines.push(`=== emulate ${emulate}`)
		for (const tag of overlap) {
			const fixture = NEUTRAL_MARKUP.find(([name]) => name === tag)
			if (fixture === undefined) continue
			await stage.mount(fixture[1])
			if (emulate) await stage.load(EMULATION)
			const [before] = await stage.read(tag, properties)
			await stage.load(profile)
			const [after] = await stage.read(tag, properties)
			const declared: string[] = []
			for (const rule of rules) {
				const hits = await stage.read(`${tag}:is(${rule.selector})`, [])
				if (hits.length > 0) declared.push(...rule.properties)
			}
			for (const [property, value] of before ?? new Map<string, string>()) {
				const resolved = after?.get(property)
				if (resolved === value) continue
				lines.push(
					`${tag} | ${property} | ${value} -> ${resolved} | ${declared.includes(property) ? 'declared' : 'CONSEQUENCE'}`,
				)
			}
			await stage.clear()
		}
	}
	writeFileSync('tmp/probe/pl-classify.out.txt', `${lines.join('\n')}\n`)
}, 120_000)
