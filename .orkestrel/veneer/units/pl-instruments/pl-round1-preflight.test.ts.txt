import type { LonghandRule } from '../../setupServer.js'
import { requireValue } from '@orkestrel/test'
import { afterAll, afterEach, beforeAll, describe, expect, it } from 'vitest'
import { readBuiltCascade, readVeneerGuide, SheetReader } from '../../setupServer.js'
import { compileProfile, STAGE_TIMEOUT, stage, TAILWIND_PATHS } from '../../setupService.js'
import {
	collectTypeSelectors,
	computeContentExtent,
	ELEMENT_TAGS,
	NEUTRAL_MARKUP,
	PREFLIGHT_BUILDS,
	PREFLIGHT_DIMENSIONS,
	readPreflightDepartures,
} from '../../setupStyles.js'

// The tags the overlap has to reach before a reading taken over it means anything. Each one is a
// tag Tailwind's reset and Veneer's elements layer both treat, so a derivation that found nothing
// cannot pass the case that uses it.
const FLOOR_TAGS = ['a', 'button', 'h1', 'hr', 'img', 'input', 'table', 'textarea']

// The profile compiles once for the file, from its own path, through the installed plugin.
const preflightProfile = await compileProfile(TAILWIND_PATHS.preflight)
const reset = new SheetReader(preflightProfile).layers.filter((layer) => layer.name === 'base')
const elements = new SheetReader(readBuiltCascade()).layers.filter(
	(layer) => layer.name === 'elements',
)

// The tags the reset treats are the type selectors of its own style rules. Its universal rule and
// its pseudo-element rules name none, which is why the reading is the leading type selector rather
// than the selector text.
const overlap = collectTypeSelectors(reset.flatMap((layer) => layer.selectors)).filter((tag) =>
	ELEMENT_TAGS.some(([, tags]) => tags.includes(tag)),
)

// The utility rules the profile generates for the shared names off the exclusion line, which the
// profiles proof holds equal to those names. They sit outside the reset, so the property population
// leaves them out.
const generated = new SheetReader(preflightProfile).layers.flatMap((layer) =>
	layer.name === 'utilities' ? layer.selectors : [],
)

// The property population the comparison walks, and the longhands each elements-layer rule
// declares, as Chromium expands them. Both are the engine's answer, so they are read after the
// stage opens.
let properties: readonly string[] = []
let treated: readonly LonghandRule[] = []

beforeAll(async () => {
	await stage.open()
	properties = [
		...new Set(
			(await stage.expand(preflightProfile)).flatMap((rule) =>
				generated.includes(rule.selector)
					? []
					: rule.properties.filter((property) => !property.startsWith('--')),
			),
		),
	]
	// One rule per declaration keeps each declaration beside the selector it is written under,
	// which is the whole reading a type selector needs.
	treated = await stage.expand(
		elements
			.flatMap((layer) => layer.declarations)
			.map(
				(declaration) =>
					`${declaration.selector} { ${declaration.property}: ${declaration.value}${declaration.important ? ' !important' : ''} }`,
			)
			.join('\n'),
	)
}, STAGE_TIMEOUT)

afterEach(async () => {
	await stage.clear()
})

afterAll(async () => {
	await stage.destroy()
})

describe('the preflight pairing', () => {
	it('derives the overlapping tags and the properties the reset declares', async () => {
		for (const tag of FLOOR_TAGS) expect(overlap).toContain(tag)
		// The property set is what the comparison walks, so the reset's universal rule contributes
		// its properties to every element's reading rather than to the elements it names. Chromium
		// expands the whole profile, so `border: 0 solid` is read as every side's width, style, and
		// color rather than as the shorthand the reset writes. The profile less its generated
		// utilities is the reset's population, because every other property it declares outside a
		// custom property sits in the `base` layer.
		expect(
			new SheetReader(preflightProfile).declarations.filter(
				(declaration) =>
					!declaration.property.startsWith('--') &&
					declaration.layer !== 'base' &&
					!generated.includes(declaration.selector),
			),
		).toEqual([])
		expect(properties).toContain('tab-size')
		expect(properties).toContain('border-top-style')
		expect(properties).not.toContain('border')
		// A moved dimension is compared through its sizing and its edges, so each one is read.
		for (const [dimension, edges] of PREFLIGHT_DIMENSIONS)
			for (const property of ['box-sizing', dimension, ...edges])
				expect(properties).toContain(property)
		// The fixture table is held to the derived overlap rather than maintained beside it, so a
		// tag the reset gains is a tag this proof has to render before it can read anything.
		expect(overlap.filter((tag) => !NEUTRAL_MARKUP.some(([name]) => name === tag))).toEqual([])
	})

	it('stages the Chromium 153 defaults beneath the cascade and the profile', async () => {
		// A staged sheet that never reached the page would leave every build reading the host's own
		// defaults, and the pairing would pass under each of them for that reason alone. So the
		// staged sheet is read through what it changes, measured against the element's own reading
		// rather than against a value written beside it.
		const [, defaults] = requireValue(
			PREFLIGHT_BUILDS.find(([build]) => build === 'Chromium 153'),
			'The builds table stages no Chromium 153 defaults',
		)
		const dimension = requireValue(
			PREFLIGHT_DIMENSIONS.find(([name]) => name === 'height'),
			'The dimensions table reads no height',
		)
		const markup = ['table', 'select'].map(
			(tag) =>
				requireValue(
					NEUTRAL_MARKUP.find(([name]) => name === tag),
					`The fixture table renders no ${tag}`,
				)[1],
		)
		await stage.mount(markup.join(''))
		await stage.load(defaults)
		const [bordered] = await stage.read('table', ['color', 'border-top-color'])
		const table = requireValue(bordered, 'The fixture rendered no table')
		const [before] = await stage.read('select', [
			'background-color',
			'box-sizing',
			`contain-intrinsic-${dimension[0]}`,
			dimension[0],
			...dimension[1],
		])
		const standalone = requireValue(before, 'The fixture rendered no select')
		expect(table.get('border-top-color')).toBe(table.get('color'))
		expect(computeContentExtent(standalone, dimension)).toBe(
			Number.parseFloat(String(standalone.get(`contain-intrinsic-${dimension[0]}`))),
		)
		// The profile's reset beats the staged defaults, the way it beats the user-agent origin.
		await stage.load(preflightProfile)
		const [after] = await stage.read('select', ['background-color'])
		expect(after?.get('background-color')).not.toBe(standalone.get('background-color'))
	})

	it.each(PREFLIGHT_BUILDS)(
		'keeps every property the elements layer declares, and holds every move the profile makes to its recorded preflight value, under %s',
		async (_build, defaults) => {
			for (const tag of FLOOR_TAGS) expect(overlap).toContain(tag)
			const measured: string[] = []
			const kept: string[] = []
			const sized: string[] = []
			const resolved: string[] = []
			for (const tag of overlap) {
				const fixture = requireValue(
					NEUTRAL_MARKUP.find(([name]) => name === tag),
					`The fixture table renders no ${tag}`,
				)
				await stage.mount(fixture[1])
				await stage.load(defaults)
				// The fixture renders the tag once, and the stage page carries no element of these tags
				// outside its container, so the first match is the fixture's own; `html` reads the root.
				const [before] = await stage.read(tag, properties)
				const standalone = requireValue(before, `The fixture rendered no ${tag}`)
				await stage.load(preflightProfile)
				const [after] = await stage.read(tag, properties)
				const paired = requireValue(after, `The fixture rendered no ${tag}`)
				// Veneer's `elements` layer sits after `base`, so every property Veneer's own reboot
				// declares for the tag keeps the value the shipped cascade resolves for it. A rule
				// naming the tag in any of its states counts, because the layer order decides the
				// question for all of them at once.
				const declared = treated.flatMap((rule) =>
					collectTypeSelectors([rule.selector]).includes(tag) ? rule.properties : [],
				)
				for (const property of properties) {
					if (!declared.includes(property)) continue
					kept.push(`${tag} | ${property}`)
					expect(`${tag} | ${property} | ${String(paired.get(property))}`).toBe(
						`${tag} | ${property} | ${String(standalone.get(property))}`,
					)
				}
				for (const [property, value] of standalone) {
					const settled = paired.get(property)
					resolved.push(`${tag} | ${property} | ${String(settled)}`)
					if (settled === value) continue
					// A moved dimension follows the padding and border the reset zeroes around a content
					// box the build's own control renders, so it is compared by that content box, which
					// the profile leaves where the standalone cascade put it.
					const dimension = PREFLIGHT_DIMENSIONS.find(([name]) => name === property)
					if (dimension === undefined) {
						measured.push(`${tag} | ${property} | ${String(settled)}`)
						continue
					}
					sized.push(`${tag} | ${property}`)
					expect(`${tag} | ${property} | ${String(computeContentExtent(paired, dimension))}`).toBe(
						`${tag} | ${property} | ${String(computeContentExtent(standalone, dimension))}`,
					)
				}
				await stage.clear()
			}
			// A run whose comparison read nothing at all would report no move, no kept property, and
			// no moved dimension, and pass every reading taken over each list.
			expect(kept).toContain('hr | border-top-style')
			expect(measured).toContain('html | tab-size | 4')
			expect(sized).toContain('select | height')
			// Each move is compared by tag, property, and the value Tailwind's preflight sets, never
			// by the standalone value: a user-agent default (a `select` or `table` treatment, for
			// example) varies across builds while the value the reset declares does not. So
			// every move is a row of the guide's table, and every row is the value the profile
			// resolves for its pair, whether or not this build's default already held it. A row
			// naming a pair Veneer declares, or a dimension, records something the profile does not
			// set, so neither may stand in the table.
			const rows = readPreflightDepartures(readVeneerGuide())
			const recorded = rows.map((row) => `${row.tag} | ${row.property} | ${row.preflight}`)
			expect(measured.filter((move) => !recorded.includes(move))).toEqual([])
			expect(recorded.filter((row) => !resolved.includes(row))).toEqual([])
			expect(
				rows
					.filter(
						(row) =>
							kept.includes(`${row.tag} | ${row.property}`) ||
							PREFLIGHT_DIMENSIONS.some(([dimension]) => dimension === row.property),
					)
					.map((row) => `${row.tag} | ${row.property}`),
			).toEqual([])
		},
	)
})
