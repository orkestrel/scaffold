import { expect, it } from 'vitest'
import {
	attributeSelector,
	collectShippedComponents,
	indexRecordingKeys,
	readCompatibility,
	readOracleInventory,
} from '../../tests/setupServer.js'
import { normalizeComplexSelector } from '../../tests/setupStyles.js'

it('reads the key the ladder answers for each recorded transition selector', () => {
	const inventory = readOracleInventory()
	const shipped = collectShippedComponents(readCompatibility())
	const recording = indexRecordingKeys(inventory)
	const vocabulary = inventory.components.transition
	const answers = [
		...new Set((vocabulary?.rules ?? []).map((rule) => normalizeComplexSelector(rule.selector))),
	].map((selector) => [selector, recording.get(selector), attributeSelector(selector, 'components', recording, shipped)])
	console.log(JSON.stringify({ shipped: shipped.includes('transition'), answers }))
	expect(answers.length).toBeGreaterThan(0)
})
