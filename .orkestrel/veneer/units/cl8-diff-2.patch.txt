+export async function compileBreakpointRamp(): Promise<{
+	readonly zero: readonly string[]
+	readonly members: ReadonlyMap<string, number>

+	const { compileString } = await import('sass')
+	const ramp = compileString(

+	})

+/**
+ * Collects the grid selector and enclosing media-condition multiset from a CSS cascade.
+ * @param cascade - The CSS text to inspect, including withheld grid families when present.
+ * @returns Sorted serialized selector/condition pairs, retaining duplicate rules.
+ * @remarks Normalizes selector spacing and equivalent minimum-width notation, never values.

+export function collectGridVocabulary(cascade: string): readonly string[] {
+	const vocabulary: string[] = []
+	parse(cascade).walkRules((rule) => {
+		for (const selector of rule.selectors) {
+			if (!/^\.(?:row|col|offset)(?:-|[^\w-]|$)/u.test(selector)) continue
+			const conditions: string[] = []
+			for (let parent: AnyNode | undefined = rule.parent; parent; parent = parent.parent) {
+				if (parent.type === 'atrule' && parent.name === 'media')
+					conditions.unshift(

+					)
+			}
+			vocabulary.push(JSON.stringify([normalizeComplexSelector(selector), conditions]))
+		}
+	})
