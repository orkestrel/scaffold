+/** Describes written declarations belonging to one brace block in a style partial. */
+export interface StyleBlock {
+	readonly path: string
+	readonly line: number
+	readonly declarations: readonly string[]
+}
+
+/** Describes a declaration block shared by distinct style partials. */
+export interface StyleOverlap {
+	readonly left: StyleBlock
+	readonly right: StyleBlock
+	readonly declarations: readonly string[]
+}
+
+/** Reports the discovered partials, compared file pairs, and shared declaration blocks. */
+export interface StyleSweepResult {
+	readonly files: readonly string[]
+	readonly pairs: number
+	readonly shared: readonly StyleOverlap[]
+}
+
+/**
+ * Scans every SCSS partial under a directory for shared written declaration blocks.
+ * @param root - The styles directory. Default: the workspace's `src/styles` directory.
+ * @returns Relative partial paths, the compared pair count, and every block intersection carrying
+ * at least two distinct identical declarations.
+ * @throws When the directory or a discovered partial cannot be read.
+ * @remarks
+ * Reads authored text without expanding includes. Braces inside strings, comments, interpolation,
+ * and parenthesized values do not open declaration blocks. Whitespace outside strings is folded;
+ * declaration order and repeated copies of one declaration do not affect an intersection.
+ * @example
+ * ```ts
+ * scanStyleBlocks().shared // [] after shared blocks move into mixins
+ * ```
+ */
+export function scanStyleBlocks(
+	root: string = resolve(WORKSPACE_ROOT, 'src/styles'),
+): StyleSweepResult {
+	const files = readdirSync(root, { recursive: true, withFileTypes: true })
+		.filter((entry) => entry.isFile() && /^_.*\.scss$/u.test(entry.name))
+		.map((entry) => relative(root, resolve(entry.parentPath, entry.name)).replaceAll('\\', '/'))
+		.sort()
+	const members = files.map((path) => {
+		const blocks: StyleBlock[] = []
+		const stack: Array<{ readonly line: number; readonly declarations: Set<string> }> = []
+		let statement = ''
+		let line = 1
+		let parentheses = 0
+		let interpolation = 0
+		// Token boundaries protect quoted punctuation and comments before brace grouping.
+		for (const match of readFileSync(resolve(root, path), 'utf8').matchAll(
+			/\/\*[\s\S]*?\*\/|\/\/[^\r\n]*|"(?:\\[\s\S]|[^"\\])*"|'(?:\\[\s\S]|[^'\\])*'|#\{|\\[\s\S]|\r\n|\n|[^]/gu,
+		)) {
+			const token = match[0]
+			const start = line
+			line += token.split(/\r\n|\n/u).length - 1
+			if (token.startsWith('/*') || token.startsWith('//')) {
+				statement += ' '
+				continue
+			}
+			if (token === '#{') interpolation += 1
+			else if (interpolation > 0 && token === '{') interpolation += 1
+			else if (interpolation > 0 && token === '}') {
+				interpolation -= 1
+				statement += token
+				continue
+			}
+			if (interpolation > 0 || /^\s+$/u.test(token)) {
+				statement += /^\s+$/u.test(token) ? (statement.endsWith(' ') ? '' : ' ') : token
+				continue
+			}
+			if (token === '(') parentheses += 1
+			if (token === ')') parentheses = Math.max(0, parentheses - 1)
+			if (parentheses === 0 && token === '{') {
+				stack.push({ line: start, declarations: new Set() })
+				statement = ''
+				continue
+			}
+			if (parentheses === 0 && (token === ';' || token === '}')) {
+				const declaration =
+					/^\s*(--(?:[\w-]|#\{[^}]*\})+|(?:[a-z-]|#\{[^}]*\})+)\s*:\s*(\S[\s\S]*)$/u.exec(statement)
+				const property = declaration?.[1]
+				const value = declaration?.[2]?.trim()
+				if (property !== undefined && value !== undefined)
+					stack.at(-1)?.declarations.add(`${property}: ${value}`)
+				statement = ''
+				if (token === '}') {
+					const block = stack.pop()
+					if (block !== undefined && block.declarations.size > 0)
+						blocks.push({ path, line: block.line, declarations: [...block.declarations] })
+				}
+				continue
+			}
+			statement += token
+		}
+		return blocks
+	})
+	const shared: StyleOverlap[] = []
+	let pairs = 0
+	for (const [index, left] of members.entries()) {
+		for (const right of members.slice(index + 1)) {
+			pairs += 1
+			for (const first of left) {
+				for (const second of right) {
+					const declarations = first.declarations.filter((entry) =>
+						second.declarations.includes(entry),
+					)
+					if (declarations.length >= 2) shared.push({ left: first, right: second, declarations })
+				}
+			}
+		}
+	}
+	return { files, pairs, shared }
+}
