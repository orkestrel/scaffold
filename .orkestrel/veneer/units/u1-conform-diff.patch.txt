+import type { ColorModeInterface, ColorModeOptions, ColorModeState } from './types.js'
+import { COLOR_MODE_ATTRIBUTE, COLOR_MODE_KEY } from './constants.js'
+import { isColorModeState } from './validators.js'
+
+/**
+ * Applies and optionally restores a root's color mode without registering listeners.
+ *
+ * @example
+ * ```ts
+ * const controller = new ColorMode({ root: document.documentElement })
+ * controller.apply('dark')
+ * controller.destroy()
+ * ```
+ */
+export class ColorMode implements ColorModeInterface {
+	readonly #root: HTMLElement
+	readonly #storage: Storage | undefined
+	#written = false
+
+	constructor(options?: ColorModeOptions) {
+		this.#root = options?.root ?? document.documentElement
+		this.#storage = options?.storage
+		const stored = this.#storage?.getItem(COLOR_MODE_KEY)
+		if (isColorModeState(stored)) this.apply(stored)
+	}
+
+	get root(): HTMLElement {
+		return this.#root
+	}
+
+	get mode(): ColorModeState {
+		return this.#root.getAttribute(COLOR_MODE_ATTRIBUTE) === 'dark' ? 'dark' : 'light'
+	}
+
+	apply(mode: ColorModeState): void {
+		if (mode === 'dark') this.#root.setAttribute(COLOR_MODE_ATTRIBUTE, mode)
+		else this.#root.removeAttribute(COLOR_MODE_ATTRIBUTE)
+		this.#written = mode === 'dark'
+		this.#storage?.setItem(COLOR_MODE_KEY, mode)
+	}
+
+	toggle(): ColorModeState {
+		const mode = this.mode === 'dark' ? 'light' : 'dark'
+		this.apply(mode)
+		return mode
+	}
+
+	destroy(): void {
+		if (!this.#written) return
+		this.#root.removeAttribute(COLOR_MODE_ATTRIBUTE)
+		this.#written = false
+	}
+}

+import { ColorMode, COLOR_MODE_ATTRIBUTE, COLOR_MODE_KEY } from '@src/browser'
+import { build } from '@orkestrel/test/browser'
+import { afterEach, describe, expect, it } from 'vitest'
+
+afterEach(() => {
+	sessionStorage.clear()
+	document.documentElement.removeAttribute('data-bs-theme')
+})
+
+describe('ColorMode', () => {
+	it('reads the root on every access without changing an existing attribute', () => {
+		const root = build('div')
+		root.setAttribute('data-bs-theme', 'dark')
+		const controller = new ColorMode({ root })
+		expect(controller.root).toBe(root)
+		expect(controller.mode).toBe('dark')
+		root.setAttribute('data-bs-theme', 'auto')
+		expect(controller.mode).toBe('light')
+		controller.destroy()
+		expect(root.getAttribute('data-bs-theme')).toBe('auto')
+	})
+
+	it('toggles, persists, and removes the light attribute', () => {
+		const root = build('div')
+		const controller = new ColorMode({ root, storage: sessionStorage })
+		expect(COLOR_MODE_ATTRIBUTE).toBe('data-bs-theme')
+		expect(COLOR_MODE_KEY).toBe('color-mode')
+		expect(controller.toggle()).toBe('dark')
+		expect(root.getAttribute('data-bs-theme')).toBe('dark')
+		expect(sessionStorage.getItem('color-mode')).toBe('dark')
+		expect(controller.toggle()).toBe('light')
+		expect(root.hasAttribute('data-bs-theme')).toBe(false)
+		expect(sessionStorage.getItem('color-mode')).toBe('light')
+		controller.destroy()
+	})
+
+	it('uses the document root when no options are supplied', () => {
+		const controller = new ColorMode()
+		try {
+			expect(controller.root).toBe(document.documentElement)
+			expect(controller.toggle()).toBe('dark')
+		} finally {
+			controller.destroy()
+		}
+		expect(document.documentElement.hasAttribute('data-bs-theme')).toBe(false)
+	})
+
+	it('threads the supplied root and real storage into the controller', () => {
+		const root = build('section')
+		sessionStorage.setItem('color-mode', 'dark')
+		const controller = new ColorMode({ root, storage: sessionStorage })
+		try {
+			expect(controller.root).toBe(root)
+			expect(controller.mode).toBe('dark')
+			controller.apply('light')
+			expect(sessionStorage.getItem('color-mode')).toBe('light')
+		} finally {
+			controller.destroy()
+		}
+	})
+
+	it('restores a stored dark mode and preserves storage during destruction', () => {
+		sessionStorage.setItem('color-mode', 'dark')
+		const controller = new ColorMode({ storage: sessionStorage })
+		expect(controller.root).toBe(document.documentElement)
+		expect(controller.mode).toBe('dark')
+		controller.destroy()
+		expect(document.documentElement.hasAttribute('data-bs-theme')).toBe(false)
+		expect(sessionStorage.getItem('color-mode')).toBe('dark')
+		controller.destroy()
+	})
+
+	it('restores stored light by removing a preexisting dark attribute', () => {
+		const root = build('div')
+		root.setAttribute('data-bs-theme', 'dark')
+		sessionStorage.setItem('color-mode', 'light')
+		const controller = new ColorMode({ root, storage: sessionStorage })
+		expect(root.hasAttribute('data-bs-theme')).toBe(false)
+		expect(controller.mode).toBe('light')
+		controller.destroy()
+	})
+
+	it('leaves the root untouched for an invalid stored mode', () => {
+		const root = build('div')
+		root.setAttribute('data-bs-theme', 'dark')
+		sessionStorage.setItem('color-mode', 'auto')
+		const controller = new ColorMode({ root, storage: sessionStorage })
+		expect(controller.mode).toBe('dark')
+		controller.destroy()
+		expect(root.getAttribute('data-bs-theme')).toBe('dark')
+		expect(sessionStorage.getItem('color-mode')).toBe('auto')
+	})
+
+	it('leaves an external write in place after its own removal', () => {
+		const root = build('div')
+		const controller = new ColorMode({ root })
+		controller.apply('light')
+		root.setAttribute('data-bs-theme', 'dark')
+		controller.destroy()
+		expect(root.getAttribute('data-bs-theme')).toBe('dark')
+	})
+
+	it('removes only its attribute after writing and preserves unrelated root content', () => {
+		const root = build('div')
+		root.setAttribute('aria-label', 'Existing root')
+		const controller = new ColorMode({ root })
+		controller.apply('dark')
+		controller.destroy()
+		expect(root.hasAttribute('data-bs-theme')).toBe(false)
+		expect(root.getAttribute('aria-label')).toBe('Existing root')
+		root.setAttribute('data-bs-theme', 'dark')
+		controller.destroy()
+		expect(root.getAttribute('data-bs-theme')).toBe('dark')
+	})
+})
