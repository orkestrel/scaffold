import { requireValue } from '@orkestrel/test'
import { readFileSync } from 'node:fs'
import { describe, expect, it } from 'vitest'
import { TOKEN_PREFIX } from '../../setup.js'
import {
	collectInlineSources,
	collectSharedNames,
	readBuiltCascade,
	readVeneerGuide,
	SheetReader,
} from '../../setupServer.js'
import { CANDIDATE_FLOOR, compileProfile, TAILWIND_PATHS } from '../../setupService.js'
import { collectFencedBlocks } from '../../setupStyles.js'

// The order Veneer's own `_tokens.scss` partial declares, which every profile repeats unchanged.
const ORDER = ['theme', 'reset', 'base', 'elements', 'components', 'utilities']

// The theme variables the instrument's control utilities read: `px-8` resolves its padding from the
// spacing scale and `font-bold` its weight from the weight scale. Tailwind keeps a theme variable
// only for a utility it generated, so these are present exactly when the theme import was composed
// and a utility that reads them was emitted.
const CONTROL_VARIABLES = ['--spacing', '--font-weight-bold']

// Each profile compiles once for the file, from its own path, through the installed plugin.
// Readiness has written the candidate list every one of them scans before this module runs.
const tailwindSource = readFileSync(TAILWIND_PATHS.tailwind, 'utf8')
const preflightSource = readFileSync(TAILWIND_PATHS.preflight, 'utf8')
const instrumentSource = readFileSync(TAILWIND_PATHS.instrument, 'utf8')
const tailwindProfile = await compileProfile(TAILWIND_PATHS.tailwind)
const preflightProfile = await compileProfile(TAILWIND_PATHS.preflight)
const instrumentProfile = await compileProfile(TAILWIND_PATHS.instrument)
const cascade = readBuiltCascade()

describe('stylesheet profiles', () => {
	it('declares the one order line in every profile, and leaves the document order unmoved', () => {
		// A page links the shipped cascade first, so its own placement is the document's. A profile
		// arriving after it re-declares names that are already placed, which is what makes the
		// pairing safe: a layer's position is settled by its first declaration across the page's
		// sheets, so the cascade followed by the profile is the order the page resolves against.
		// The reading is therefore the first placement of each layer across the linked cascade
		// followed by the loaded profile. That sheet sequence is the one `stage.open` and
		// `stage.load` fix, so this proof assumes it rather than observing it on a page.
		expect(new SheetReader(cascade).order).toEqual(ORDER)
		for (const compiled of [tailwindProfile, preflightProfile]) {
			expect(new SheetReader(compiled).statement).toEqual(ORDER)
			expect(new SheetReader(`${cascade}\n${compiled}`).order).toEqual(ORDER)
		}
		// The instrument's control registers a custom property, so Tailwind prepends a generated
		// layer statement of its own and the order line follows it. The following case reads that
		// placement; the reading here is that the order line is declared unchanged behind it.
		expect(new SheetReader(instrumentProfile).order.slice(-ORDER.length)).toEqual(ORDER)
		expect(
			new SheetReader(`${cascade}\n${instrumentProfile}`).order.slice(0, ORDER.length),
		).toEqual(ORDER)
	})

	it('fills Tailwind reset only under the preflight profile', () => {
		// This is the difference the two profile names stand for. The `tailwind` profile composes
		// the theme and utilities imports and reaches no reset: it fills only the layers the
		// utilities it generates need, the `utilities` block for their rules and the `theme` block
		// for the variables they read. The `preflight` profile takes the bare import, which fills
		// the `base` layer with Tailwind's reset and declares Tailwind's own font tokens beside it.
		// Both scan the one candidate list under the one exclusion line, so both generate the same
		// utilities, and the reset and its tokens are all that tells them apart.
		const tailwind = new SheetReader(tailwindProfile)
		expect(tailwind.layers.map((layer) => layer.name)).toEqual(['theme', 'utilities'])
		// The generated gap steps read the spacing scale and no other theme variable, so the
		// profile keeps that one variable, and a token the reset or a static theme would declare
		// breaks the equality.
		expect(tailwind.variables).toEqual(['--spacing'])
		const preflight = new SheetReader(preflightProfile)
		expect(preflight.layers.map((layer) => layer.name)).toEqual(['theme', 'base', 'utilities'])
		expect(preflight.variables).toContain('--font-sans')
		expect(
			preflight.layers.flatMap((layer) => (layer.name === 'base' ? layer.selectors : [])),
		).not.toEqual([])
	})

	it('composes the theme and utilities imports, proved by a candidate the cascade never ships', () => {
		// The `tailwind` profile emits only the shared names off the exclusion line, so what it
		// emits follows which names a release declares important rather than which imports the
		// profile composes. The instrument carries an `@source inline(…)` control naming utilities
		// the built cascade declares no class for, so its emission is positive whatever the line
		// holds: the `theme` block holds the variables those utilities read, and the `utilities`
		// block holds their rules. Removing either import takes one of those readings away.
		const [control] = collectInlineSources(instrumentSource).filter((source) => !source.excluded)
		const names = requireValue(control, 'The instrument declares no candidate control').names
		const instrument = new SheetReader(instrumentProfile)
		// Each reading names the layer block it is asking about rather than taking the whole sheet,
		// because the claim is about where the compiler put the declaration. A theme variable read
		// off the whole stylesheet is still found when the block holding it is no longer `theme`.
		const variables = instrument.layers.flatMap((layer) =>
			layer.name === 'theme' ? layer.declarations.map((declaration) => declaration.property) : [],
		)
		for (const variable of CONTROL_VARIABLES) expect(variables).toContain(variable)
		const utilities = instrument.layers.flatMap((layer) =>
			layer.name === 'utilities' ? layer.selectors : [],
		)
		for (const name of names) expect(utilities).toContain(`.${name}`)
		// Both readings are only as good as their ability to fail on a declaration that moved, and
		// the compiler puts each one where it belongs, so the control is planted: the same compiled
		// bytes with the `theme` block relabelled. The scoped reading reports nothing there, while a
		// reading taken across the whole stylesheet still finds every variable.
		const moved = new SheetReader(instrumentProfile.replace('@layer theme {', '@layer reset {'))
		expect(moved.layers.filter((layer) => layer.name === 'theme')).toEqual([])
		for (const variable of CONTROL_VARIABLES) expect(moved.variables).toContain(variable)
	})

	it('places the generated properties layer before the order line, leaving the named layers in order', () => {
		// A utility that registers a custom property makes Tailwind generate a `properties` layer,
		// and Tailwind declares that layer in a statement of its own ahead of the order line the
		// entry wrote. The placement is what a consumer meets, so the reading is of the compiled
		// stylesheet's own order: the generated layer comes first and Veneer's named layers follow
		// in their declared order.
		const instrument = new SheetReader(instrumentProfile)
		expect(instrument.statement).toEqual(['properties'])
		expect(instrument.layers.map((layer) => layer.name)).toContain('properties')
		expect(instrument.order).toEqual(['properties', ...ORDER])
		// On a page the cascade placed the named layers before the profile arrived, so the
		// generated layer takes the one position left rather than moving any of them. The reading
		// assumes the same fixed sheet sequence as the document-order case.
		expect(new SheetReader(`${cascade}\n${instrumentProfile}`).order).toEqual([
			...ORDER,
			'properties',
		])
	})

	it('withholds every shared class name the exclusion line names, and emits exactly the shared names off it', () => {
		const [exclusion] = collectInlineSources(tailwindSource).filter((source) => source.excluded)
		const excluded = requireValue(
			exclusion,
			'The tailwind profile declares no exclusion line',
		).names
		for (const name of CANDIDATE_FLOOR) expect(excluded).toContain(name)
		// The instrument is the `tailwind` profile without its exclusion line, so what it emits and
		// the profile does not is exactly what the line withheld. Reading the instrument first
		// proves the candidates list reached the compiler at all.
		const instrument = new SheetReader(instrumentProfile)
		for (const name of CANDIDATE_FLOOR) expect(instrument.selectors).toContain(`.${name}`)
		// The instrument's own control names are read from the directive that adds them rather than
		// repeated here, so the list has one home on this side too. Everything the compiler
		// generated that the cascade does not also declare is the control.
		const control = collectInlineSources(instrumentSource).flatMap((source) =>
			source.excluded ? [] : source.names,
		)
		const shared = collectSharedNames(instrument.names, new SheetReader(cascade).names)
		expect(instrument.names.filter((name) => !shared.includes(name)).sort()).toEqual(
			[...control].sort(),
		)
		// A shared name leaves the line only where the cascade declares it important on every
		// longhand Tailwind's rule for it declares, which the consumer pairing reads as an equality.
		// What this case reads is the other side of that line: each executed profile emits
		// exactly the shared names off it and nothing else, so a name that stays on the line is
		// withheld and a name that leaves it is generated. The gap steps ship off the line, so the
		// reading covers a name rather than an empty set.
		const off = shared.filter((name) => !excluded.includes(name))
		expect(off).toContain('gap-3')
		for (const compiled of [tailwindProfile, preflightProfile]) {
			const emitted = new SheetReader(compiled).layers.flatMap((layer) =>
				layer.name === 'utilities' ? layer.selectors : [],
			)
			expect([...emitted].sort()).toEqual(off.map((name) => `.${name}`).sort())
		}
		expect([...new SheetReader(tailwindProfile).names].sort()).toEqual([...off].sort())
	})

	it('holds every written copy of the exclusion line equal to the profile that declares it', () => {
		// The guide's recipes and the preflight fixture each repeat the list, so the claim that the
		// profile is its one home is only true if something holds the copies to it. This is that
		// something: a release that extends the line and leaves a copy behind reddens here rather
		// than shipping two lists that disagree.
		const [exclusion] = collectInlineSources(tailwindSource).filter((source) => source.excluded)
		const declared = requireValue(
			exclusion,
			'The tailwind profile declares no exclusion line',
		).names
		// Only the guide's CSS fences are read as CSS, through the shared fence reader. Its prose
		// names the directive too, and the reader refuses a form it cannot parse rather than
		// skipping it, so handing it the whole document reports the prose as a malformed directive
		// — which is the reader working.
		const fences = collectFencedBlocks(readVeneerGuide(), 'css')
		expect(fences).not.toEqual([])
		const consumer = readFileSync(TAILWIND_PATHS.consumer, 'utf8')
		const copies = [
			...collectInlineSources(fences.join('\n')),
			...collectInlineSources(preflightSource),
			...collectInlineSources(consumer),
		]
		expect(copies).not.toEqual([])
		for (const copy of copies) {
			expect(copy.excluded).toBe(true)
			expect(copy.names).toEqual(declared)
		}
	})

	it('declares the Tailwind parts each profile is named for, and repeats them in the instrument', () => {
		// The preceding case proves the instrument's imports load-bearing, and the instrument is a
		// copy. This holds the copy to the profile and pins what the profile imports, so deleting a
		// part from either file reports here. Without it the `tailwind` profile could lose a part
		// and stay green, because every name its candidate list offers is excluded and a profile
		// with no Tailwind parts at all emits the same nothing.
		const declared = tailwindSource.split('\n').filter((line) => line.startsWith('@import '))
		expect(declared).toEqual([
			"@import 'tailwindcss/theme.css' layer(theme);",
			"@import 'tailwindcss/utilities.css' layer(utilities) source(none);",
		])
		expect(instrumentSource.split('\n').filter((line) => line.startsWith('@import '))).toEqual(
			declared,
		)
		// The `preflight` profile is the one that differs, and the bare import is the difference.
		expect(preflightSource.split('\n').filter((line) => line.startsWith('@import '))).toEqual([
			"@import 'tailwindcss' source(none);",
		])
	})

	it("fills the preflight theme block with no token in Veneer's namespace", () => {
		// The browser reader that picks Veneer's sheet off a page selects a `theme` block declaring
		// a token in Veneer's namespace. The preflight profile fills a `theme` block too, so what
		// keeps that reader from answering with a Tailwind sheet is that the block carries none of
		// Veneer's tokens, while the cascade's own block does.
		const preflight = new SheetReader(preflightProfile)
		const theme = preflight.layers.flatMap((layer) =>
			layer.name === 'theme' ? layer.declarations : [],
		)
		expect(theme).not.toEqual([])
		expect(theme.filter((declaration) => declaration.property.startsWith(TOKEN_PREFIX))).toEqual([])
		expect(
			new SheetReader(cascade).declarations.some(
				(declaration) =>
					declaration.layer === 'theme' && declaration.property.startsWith(TOKEN_PREFIX),
			),
		).toBe(true)
	})
})
