import { readFileSync } from 'node:fs'
import { afterAll, beforeAll, describe, expect, it } from 'vitest'
import { collectImportantNames, collectRuleLonghands, collectSharedNames, readBuiltCascade, SheetReader } from '../../setupServer.js'
import { compileProfile, STAGE_TIMEOUT, stage, TAILWIND_PATHS } from '../../setupService.js'

const instrumentProfile = await compileProfile(TAILWIND_PATHS.instrument)
const cascade = new SheetReader(readBuiltCascade())
const shared = collectSharedNames(new SheetReader(instrumentProfile).names, cascade.names)
beforeAll(async () => { await stage.open() }, STAGE_TIMEOUT)
afterAll(async () => { await stage.destroy() })
describe('ut probe', () => {
	it('records the text shared names', async () => {
		const longhands = collectRuleLonghands(await stage.expand(instrumentProfile), shared)
		const important = collectImportantNames(await stage.expand(readBuiltCascade()), longhands)
		const mine = shared.filter((name) => /^(text|link)-/u.test(name))
		const veneer = collectRuleLonghands(await stage.expand(readBuiltCascade()), mine)
		const rows = mine.map((name) => `${name} | tailwind: ${(longhands.get(name) ?? []).join(', ')} | veneer: ${(veneer.get(name) ?? []).join(', ')} | ${important.includes(name) ? 'off the line' : 'on the line'}`)
		console.log(`UTPROBE\n${rows.join('\n')}\nSHARED ${shared.join(' ')}`)
		expect(mine.length).toBeGreaterThan(0)
		readFileSync
	})
})
