import { measureContent } from '@orkestrel/test/browser'
import { inject, it } from 'vitest'
import { commands } from 'vitest/browser'
import { CAPTURE_KEYS } from '../../tests/setup.js'
import { applyTheme, mountShowcase, readSubject } from '../../tests/setupBrowser.js'

it('measures every main child', async () => {
	const variant = inject('variant')
	const mounted = await mountShowcase()
	await applyTheme(variant)
	const main = mounted.host.querySelector('main')
	const header = mounted.host.querySelector('header')
	if (main === null || header === null) throw new Error('no main')
	const rows: string[] = []
	rows.push(`variant ${variant} viewport ${String(window.innerWidth)}x${String(window.innerHeight)}`)
	rows.push(`content ${String(measureContent())}`)
	rows.push(`header ${String(header.getBoundingClientRect().height)} top ${String(header.getBoundingClientRect().top)}`)
	rows.push(`main top ${String(main.getBoundingClientRect().top)} height ${String(main.getBoundingClientRect().height)}`)
	const holders = new Map<Element, string[]>()
	for (const key of CAPTURE_KEYS) {
		const subject = readSubject(mounted.host, key.subject)
		const child = [...main.children].find((candidate) => candidate.contains(subject))
		if (child === undefined) continue
		holders.set(child, [...(holders.get(child) ?? []), key.scenario])
	}
	for (const child of main.children) {
		const box = child.getBoundingClientRect()
		rows.push(
			`${child.getAttribute('aria-label') ?? child.localName}\t${box.height.toFixed(1)}\t${(holders.get(child) ?? []).join(',')}`,
		)
	}
	await commands.writeFile(`tmp/probe/pf-heights-${variant}.txt`, rows.join('\n'))
	mounted.cleanup()
})
