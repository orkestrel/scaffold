+import type { ButtonInterface, ButtonOptions } from './types.js'
+import { AppError } from '@src/core'
+import { BUTTON_ACTIVE, BUTTON_TOGGLE } from './constants.js'
+import { bindEventMap, emitEvent } from './helpers.js'
+import { isButtonHost } from './validators.js'
+
+/**
+ * Toggles a host's pressed state and restores its original state on destruction.
+ *
+ * @remarks
+ * The host must be an HTMLElement in the current realm and have no live button owner.
+ * Programmatic toggles also apply to disabled hosts. Initial hooks use the host's DOM events.
+ *
+ * @example
+ * ```ts
+ * const button = new Button(document.createElement('button'))
+ * button.toggle()
+ * button.destroy()
+ * ```
+ */
+export class Button implements ButtonInterface {
+	static readonly #hosts = new WeakSet<HTMLElement>()
+	readonly #host: HTMLElement
+	readonly #active: boolean
+	readonly #attribute: string | null
+	readonly #controller = new AbortController()
+
+	/**
+	 * Captures the host's managed state and binds initial hooks.
+	 *
+	 * @param host - The HTML element to own.
+	 * @param options - The initial event subscriptions.
+	 * @throws Thrown when the host is not an HTMLElement or already has a live owner.
+	 */
+	constructor(host: HTMLElement, options?: ButtonOptions) {
+		if (!isButtonHost(host)) {
+			let tag: string | undefined
+			try {
+				const value: unknown = host
+				if (value instanceof Element) tag = value.tagName
+			} catch {
+				tag = undefined
+			}
+			throw new AppError('Button requires an HTMLElement host', 'BUTTON_HOST_INVALID', { tag })
+		}
+		if (Button.#hosts.has(host)) {
+			throw new AppError('Button host already has a live owner', 'BUTTON_HOST_OWNED', {
+				tag: host.tagName,
+			})
+		}
+		this.#host = host
+		this.#active = host.classList.contains(BUTTON_ACTIVE)
+		this.#attribute = host.getAttribute('aria-pressed')
+		bindEventMap(host, options?.on, this.#controller.signal)
+		Button.#hosts.add(host)
+	}
+
+	get host(): HTMLElement {
+		return this.#host
+	}
+
+	get pressed(): boolean {
+		return this.#host.classList.contains(BUTTON_ACTIVE)
+	}
+
+	toggle(): boolean {
+		if (this.#controller.signal.aborted) return this.pressed
+		const pressed = this.#host.classList.toggle(BUTTON_ACTIVE)
+		this.#host.setAttribute('aria-pressed', String(pressed))
+		emitEvent(this.#host, BUTTON_TOGGLE, { pressed })
+		return pressed
+	}
+
+	destroy(): void {
+		if (this.#controller.signal.aborted) return
+		this.#controller.abort()
+		this.#host.classList.toggle(BUTTON_ACTIVE, this.#active)
+		if (this.#attribute === null) this.#host.removeAttribute('aria-pressed')
+		else this.#host.setAttribute('aria-pressed', this.#attribute)
+		Button.#hosts.delete(this.#host)
+	}
+}

+import type { DelegateInterface, DelegateOptions } from './types.js'
+import { isAppError } from '@src/core'
+import { Button } from './Button.js'
+import { BUTTON_SELECTOR } from './constants.js'
+import { isButtonHost } from './validators.js'
+
+/**
+ * Activates data-attribute button hosts through a root's delegated click listener.
+ *
+ * @remarks
+ * Construction installs the listener. Importing the browser barrel installs nothing.
+ * Hosts with another live button owner are left to that owner.
+ * Destruction releases the listener and restores every host this delegate acquired.
+ *
+ * @example
+ * ```ts
+ * const delegate = new Delegate({ root: document })
+ * delegate.destroy()
+ * ```
+ */
+export class Delegate implements DelegateInterface {
+	readonly #root: ParentNode
+	readonly #controller = new AbortController()
+	#buttons = new WeakMap<HTMLElement, Button>()
+	readonly #owned = new Set<Button>()
+
+	/**
+	 * Binds native clicks on the supplied root.
+	 *
+	 * @param options - The delegation root. Default: `document`.
+	 */
+	constructor(options?: DelegateOptions) {
+		this.#root = options?.root ?? document
+		this.#root.addEventListener('click', (event) => this.#activate(event), {
+			signal: this.#controller.signal,
+		})
+	}
+
+	get root(): ParentNode {
+		return this.#root
+	}
+
+	destroy(): void {
+		if (this.#controller.signal.aborted) return
+		this.#controller.abort()
+		for (const button of this.#owned) button.destroy()
+		this.#owned.clear()
+		this.#buttons = new WeakMap()
+	}
+
+	#activate(event: Event): void {
+		for (const button of this.#owned) {
+			if (this.#root.contains(button.host)) continue
+			button.destroy()
+			this.#buttons.delete(button.host)
+			this.#owned.delete(button)
+		}
+		if (!(event.target instanceof Element)) return
+		const host = event.target.closest(BUTTON_SELECTOR)
+		if (!isButtonHost(host) || !this.#root.contains(host)) return
+		event.preventDefault()
+		if (
+			('disabled' in host && host.disabled === true) ||
+			host.classList.contains('disabled') ||
+			host.getAttribute('aria-disabled') === 'true'
+		) {
+			return
+		}
+		let button = this.#buttons.get(host)
+		if (!button) {
+			try {
+				button = new Button(host)
+			} catch (error) {
+				if (isAppError(error) && error.code === 'BUTTON_HOST_OWNED') return
+				throw error
+			}
+			this.#buttons.set(host, button)
+			this.#owned.add(button)
+		}
+		button.toggle()
+	}
+}

+/**
+ * Dispatches a bubbling, non-cancelable DOM event carrying the supplied detail.
+ *
+ * @param host - The element receiving the event.
+ * @param type - The event's wire name.
+ * @param detail - The completed operation's payload.
+ * @returns Nothing; listeners run synchronously during dispatch.
+ * @example
+ * ```ts
+ * emitEvent(host, BUTTON_TOGGLE, { pressed: true })
+ * ```
+ */
+export function emitEvent<T>(host: HTMLElement, type: string, detail: T): void {
+	host.dispatchEvent(new CustomEvent(type, { bubbles: true, cancelable: false, detail }))
+}
+
+/**
+ * Binds button hooks to their DOM wire events until the signal aborts.
+ *
+ * @param host - The element receiving events.
+ * @param hooks - The initial subscriptions, when supplied.
+ * @param signal - The lifetime shared by the subscriptions.
+ * @returns Nothing; an aborted signal leaves no subscriptions.
+ * @example
+ * ```ts
+ * bindEventMap(host, { toggle: (event) => console.log(event.detail.pressed) }, controller.signal)
+ * ```
+ */
+export function bindEventMap(
+	host: HTMLElement,
+	hooks: ButtonHooks | undefined,
+	signal: AbortSignal,
+): void {
+	const toggle = hooks?.toggle
+	if (toggle) {
+		host.addEventListener(
+			BUTTON_TOGGLE,
+			(event) => {
+				if (isButtonEvent(event)) toggle(event)
+			},
+			{ signal },
+		)
+	}
+}

+/**
+ * Describes a programmer error with a machine-readable code and optional context.
+ *
+ * @example
+ * ```ts
+ * const error = new AppError('Host is already owned', 'BUTTON_HOST_OWNED', { tag: 'BUTTON' })
+ * ```
+ */
+export class AppError extends Error {
+	readonly code: string
+	readonly context?: Readonly<Record<string, unknown>>
+
+	/**
+	 * Retains the diagnostic and forwards native error options.
+	 *
+	 * @param message - The diagnostic message.
+	 * @param code - The machine-readable error code.
+	 * @param context - The optional diagnostic context.
+	 * @param options - The native error options carrying an optional cause.
+	 */
+	constructor(
+		message: string,
+		code: string,
+		context?: Readonly<Record<string, unknown>>,
+		options?: ErrorOptions,
+	) {
+		super(message, options)
+		this.name = 'AppError'
+		this.code = code
+		if (context !== undefined) this.context = context
+	}
+}
+
+/**
+ * Checks whether a value belongs to the application's error class.
+ *
+ * @param value - The value to inspect.
+ * @returns True if the value is an instance of the application's error class; false otherwise.
+ * @example
+ * ```ts
+ * isAppError(new AppError('Invalid host', 'BUTTON_HOST_INVALID')) // true
+ * isAppError(new Error('Invalid host')) // false
+ * ```
+ */
+export function isAppError(value: unknown): value is AppError {
+	try {
+		return value instanceof AppError
+	} catch {
+		return false
+	}
+}

+		const error = new AppError('Host is already owned', 'BUTTON_HOST_OWNED', context)
+		expect(error.message).toBe('Host is already owned')
+		expect(error.code).toBe('BUTTON_HOST_OWNED')
+		expect(error.context).toBe(context)
+		expect(error.name).toBe('AppError')
+		expect(error).toBeInstanceOf(AppError)
+		expect(error).toBeInstanceOf(Error)
+		expect(isAppError(error)).toBe(true)
+	})
+
+	it('permits absent context and forwards a cause through native error options', () => {
+		const cause = new Error('Underlying refusal')
+		const error = new AppError('Invalid host', 'BUTTON_HOST_INVALID', undefined, { cause })
+		expect(error.context).toBeUndefined()
+		expect(error.cause).toBe(cause)
+		expect(Object.getOwnPropertyDescriptor(error, 'cause')?.enumerable).toBe(false)
+		expect(Object.hasOwn(new AppError('Invalid host', 'BUTTON_HOST_INVALID'), 'cause')).toBe(false)
+	})
+
+	it('refuses native errors, coded objects, absent values, and hostile prototypes', () => {
