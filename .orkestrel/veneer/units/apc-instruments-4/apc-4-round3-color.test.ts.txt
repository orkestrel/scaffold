import { TOKEN_NAMES } from '@src/core'
import { requireValue } from '@orkestrel/test'
import {
	matchesColor,
	parseCSSColor,
	readContrast,
	readStyle,
	readToken,
} from '@orkestrel/test/browser'
import { afterEach, describe, expect, it } from 'vitest'
import { CONTRAST_BAR } from '../../../setup.js'
import { scene } from '../../../setupBrowser.js'
import {
	TEXT_COLOR_CASES,
	TEXT_EMPHASIS_CASES,
	TEXT_MODES,
	TEXT_NEUTRAL_ROLES,
	TEXT_OPACITY_CASES,
	TEXT_TIER_CASES,
	TEXT_TIER_SHARE,
} from '../../../setupStyles.js'

// A color no class writes. A scope sets it for its children to inherit, and an override writes it,
// so a class that resolves to it has painted no color of its own.
const INHERITED = 'rgb(1, 2, 3)'

afterEach(() => {
	scene.clear()
})

describe('color utilities', () => {
	it.each(TEXT_MODES)(
		'resolves each text color to the value the release records, and each role outside the neutral roles to its on-canvas tier, in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="color: ${INHERITED}">${TEXT_COLOR_CASES.map(
					({ key, value }) =>
						`<span class="text-${key}">Class</span><span data-recorded="${key}" style="--bs-text-opacity: 1; color: ${value}">Recorded</span>`,
				).join('')}${TEXT_TIER_CASES.map(
					({ key, value }) => `<span data-tier="${key}" style="color: ${value}">Tier</span>`,
				).join('')}</div>`,
			)
			const tiers = new Set(TEXT_TIER_CASES.map(({ key }) => key))
			for (const { key } of TEXT_COLOR_CASES.filter((entry) => !tiers.has(entry.key))) {
				const element = requireValue(host.querySelector(`.text-${key}`), `No text-${key} element`)
				const recorded = requireValue(
					host.querySelector(`[data-recorded="${key}"]`),
					`No recorded ${key} element`,
				)
				expect(readToken(element, '--bs-text-opacity')).toBe('1')
				expect(readStyle(element, 'color')).toBe(readStyle(recorded, 'color'))
				// Only the reset key inherits; every other key paints a color of its own.
				expect(readStyle(element, 'color') === INHERITED).toBe(key === 'reset')
			}
			// A role outside the neutral roles paints the ruling's mix of its fill with the body text,
			// written here as its own expression rather than through the alias the class reads, and
			// leaves the channel reading the release records.
			for (const { key } of TEXT_TIER_CASES) {
				const element = requireValue(host.querySelector(`.text-${key}`), `No text-${key} element`)
				const tier = requireValue(host.querySelector(`[data-tier="${key}"]`), `No ${key} tier`)
				const recorded = requireValue(
					host.querySelector(`[data-recorded="${key}"]`),
					`No recorded ${key} element`,
				)
				expect(readToken(element, '--bs-text-opacity')).toBe('1')
				expect
					.soft(matchesColor(readStyle(element, 'color'), readStyle(tier, 'color')), `${key} tier`)
					.toBe(true)
				expect
					.soft(
						matchesColor(readStyle(element, 'color'), readStyle(recorded, 'color')),
						`${key} channel`,
					)
					.toBe(false)
			}
		},
	)

	it.each(TEXT_MODES)(
		'keeps the neutral roles on their own channels rather than their emphasis tiers in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}">${TEXT_NEUTRAL_ROLES.map(
					(role) => `<span class="text-${role}">Neutral</span>`,
				).join('')}</div>`,
			)
			// Each neutral tier is a gray step that differs from the role's channels in at least one
			// mode (the light role's tier in light mode, the dark role's in dark), so a neutral role
			// moved onto its tier reports here.
			for (const role of TEXT_NEUTRAL_ROLES) {
				const element = requireValue(host.querySelector(`.text-${role}`), `No ${role} text`)
				expect
					.soft(
						matchesColor(
							readStyle(element, 'color'),
							`rgb(${readToken(element, `--bs-${role}-rgb`)})`,
						),
						`${role} channels`,
					)
					.toBe(true)
			}
		},
	)

	it.each(TEXT_MODES)(
		'reads each role color outside the neutral roles at or above the contrast bar against the page in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">${TEXT_TIER_CASES.map(
					({ key }) => `<span class="text-${key}">Role text</span>`,
				).join('')}</div>`,
			)
			for (const { key } of TEXT_TIER_CASES) {
				const element = requireValue(host.querySelector(`.text-${key}`), `No ${key} text`)
				expect
					.soft(readContrast(element), `${key} ${mode} contrast`)
					.toBeGreaterThanOrEqual(CONTRAST_BAR)
			}
		},
	)

	it.each(TEXT_MODES)(
		'paints each role color outside the neutral roles and its emphasis class in one color at the default opacity in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">${TEXT_TIER_CASES.map(
					({ key }) =>
						`<span class="text-${key}">Role</span><span class="text-${key}-emphasis">Emphasis</span>`,
				).join('')}</div>`,
			)
			for (const { key } of TEXT_TIER_CASES) {
				const role = requireValue(host.querySelector(`.text-${key}`), `No ${key} text`)
				const emphasis = requireValue(
					host.querySelector(`.text-${key}-emphasis`),
					`No ${key} emphasis text`,
				)
				expect
					.soft(matchesColor(readStyle(role, 'color'), readStyle(emphasis, 'color')), key)
					.toBe(true)
			}
		},
	)

	it('resolves each emphasis tier through the alias its own mode declares', () => {
		const host = scene.mount(
			TEXT_MODES.map(
				(mode) =>
					`<div data-bs-theme="${mode}">${TEXT_EMPHASIS_CASES.map(
						({ key, value }) =>
							`<span class="text-${key}">Class</span><span data-recorded="${key}" style="color: ${value}">Recorded</span>`,
					).join('')}</div>`,
			).join(''),
		)
		const readings = TEXT_MODES.map((mode) =>
			TEXT_EMPHASIS_CASES.map(({ key }) => {
				const scope = requireValue(
					host.querySelector(`[data-bs-theme="${mode}"]`),
					`No ${mode} scope`,
				)
				const element = requireValue(scope.querySelector(`.text-${key}`), `No ${key} element`)
				const recorded = requireValue(
					scope.querySelector(`[data-recorded="${key}"]`),
					`No recorded ${key} element`,
				)
				expect(readStyle(element, 'color')).toBe(readStyle(recorded, 'color'))
				return readStyle(element, 'color')
			}),
		)
		const [light, dark] = readings
		// The dark island resolves its own tier for every role, never the light tier it would carry
		// if the class wrote the value rather than reading the mode's alias.
		expect(
			requireValue(dark, 'No dark readings').filter(
				(color, index) => color === requireValue(light, 'No light readings')[index],
			),
		).toEqual([])
	})

	it.each(TEXT_MODES)(
		'paints each opacity step as the on-canvas tier of every role outside the neutral roles, at the alpha the step names, in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">${TEXT_TIER_CASES.map(
					({ key }) =>
						`<span class="text-${key}-emphasis">Tier</span>` +
						TEXT_OPACITY_CASES.map(
							({ step }) => `<span class="text-${key} text-opacity-${String(step)}">Faded</span>`,
						).join(''),
				).join('')}</div>`,
			)
			for (const { key } of TEXT_TIER_CASES) {
				const [red, green, blue] = requireValue(
					parseCSSColor(
						readStyle(
							requireValue(host.querySelector(`.text-${key}-emphasis`), `No ${key} tier`),
							'color',
						),
					),
					`No ${key} tier color`,
				)
				for (const { step, value } of TEXT_OPACITY_CASES) {
					const element = requireValue(
						host.querySelector(`.text-${key}.text-opacity-${String(step)}`),
						`No ${key} text-opacity-${String(step)} element`,
					)
					// The color entry sets the variable to the `1` value on the same element, and the
					// opacity entry follows it in the map, so the step is what the color reads.
					expect(Number(readToken(element, '--bs-text-opacity'))).toBe(value)
					const color = requireValue(parseCSSColor(readStyle(element, 'color')), 'No color')
					expect.soft(color[3], `${key} ${String(step)} alpha`).toBe(value)
					expect
						.soft(matchesColor(color, [red, green, blue, value]), `${key} ${String(step)} channels`)
						.toBe(true)
				}
			}
		},
	)

	it.each(TEXT_MODES)(
		'leaves a role color on its tier when the role channels or the density factor are retuned, in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">` +
					'<span class="text-primary">Role</span><span class="text-primary-emphasis">Emphasis</span>' +
					'<span class="text-bg-primary">Channel consumer</span></div>',
			)
			const scope = requireValue(host.querySelector<HTMLElement>('[data-bs-theme]'), 'No scope')
			const role = requireValue(host.querySelector('.text-primary'), 'No role element')
			const emphasis = requireValue(host.querySelector('.text-primary-emphasis'), 'No emphasis')
			const consumer = requireValue(host.querySelector('.text-bg-primary'), 'No channel consumer')
			const resting = [readStyle(role, 'color'), readStyle(emphasis, 'color')]
			// The density factor is declared at the document root alone, so it is retuned there.
			document.documentElement.style.setProperty(TOKEN_NAMES.factor.density, '2')
			try {
				expect([readStyle(role, 'color'), readStyle(emphasis, 'color')]).toEqual(resting)
			} finally {
				document.documentElement.style.removeProperty(TOKEN_NAMES.factor.density)
			}
			// The channel moves the fills that read it and leaves the role's text on its tier.
			scope.style.setProperty('--vn-color-primary-rgb', '20, 80, 140')
			expect(readStyle(consumer, 'background-color')).toBe('rgb(20, 80, 140)')
			expect(readStyle(role, 'color')).toBe(resting[0])
		},
	)

	it.each(TEXT_MODES)(
		'moves a role color to the tier of a fill or body text retuned at the scope declaring the %s theme',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">` +
					'<span class="text-primary">Role</span><span class="text-primary-emphasis">Emphasis</span>' +
					`<span id="fill-twin" style="color: color-mix(in oklab, rgb(200, 0, 0) ${TEXT_TIER_SHARE}, var(--vn-text-body-base))">Fill twin</span>` +
					`<span id="text-twin" style="color: color-mix(in oklab, var(--vn-color-primary-base) ${TEXT_TIER_SHARE}, rgb(0, 128, 0))">Text twin</span></div>`,
			)
			const scope = requireValue(host.querySelector<HTMLElement>('[data-bs-theme]'), 'No scope')
			const role = requireValue(host.querySelector('.text-primary'), 'No role element')
			const emphasis = requireValue(host.querySelector('.text-primary-emphasis'), 'No emphasis')
			const resting = readStyle(role, 'color')
			// The fill moves the tier, and the text follows it to the mix of the retuned fill.
			scope.style.setProperty('--vn-color-primary-base', 'rgb(200, 0, 0)')
			const fill = readStyle(role, 'color')
			expect(fill).not.toBe(resting)
			expect(
				matchesColor(
					fill,
					readStyle(requireValue(host.querySelector('#fill-twin'), 'No twin'), 'color'),
				),
			).toBe(true)
			expect(matchesColor(readStyle(emphasis, 'color'), fill)).toBe(true)
			scope.style.removeProperty('--vn-color-primary-base')
			// The body text is the mix's partner, so retuning it moves the text as well.
			scope.style.setProperty('--vn-text-body-base', 'rgb(0, 128, 0)')
			const partner = readStyle(role, 'color')
			expect(partner).not.toBe(resting)
			expect(
				matchesColor(
					partner,
					readStyle(requireValue(host.querySelector('#text-twin'), 'No twin'), 'color'),
				),
			).toBe(true)
		},
	)

	it.each(TEXT_MODES)(
		'moves a role color to an emphasis token retuned at the scope declaring the %s theme, and to an alias retuned on its own element',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">` +
					'<span class="text-primary">Role</span><span id="local" class="text-primary">Local</span></div>',
			)
			const scope = requireValue(host.querySelector<HTMLElement>('[data-bs-theme]'), 'No scope')
			const role = requireValue(host.querySelector('.text-primary'), 'No role element')
			const local = requireValue(host.querySelector<HTMLElement>('#local'), 'No local element')
			const resting = readStyle(role, 'color')
			expect(matchesColor(resting, 'rgb(0, 90, 60)')).toBe(false)
			// The emphasis token retuned where the theme declares it moves the text through the alias.
			scope.style.setProperty('--vn-color-primary-emphasis', 'rgb(0, 90, 60)')
			expect
				.soft(matchesColor(readStyle(role, 'color'), 'rgb(0, 90, 60)'), 'token at the theme scope')
				.toBe(true)
			scope.style.removeProperty('--vn-color-primary-emphasis')
			// The alias retuned on one element moves that element's text alone.
			local.style.setProperty('--bs-primary-text-emphasis', 'rgb(0, 90, 60)')
			expect
				.soft(matchesColor(readStyle(local, 'color'), 'rgb(0, 90, 60)'), 'alias on the element')
				.toBe(true)
			expect(readStyle(role, 'color')).toBe(resting)
		},
	)

	it.each(TEXT_MODES)(
		'leaves a role color and moves the colored link when the emphasis token is retuned below the scope declaring the %s theme',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">` +
					'<span id="local" class="text-primary">Local</span><a id="local-link" href="#content" class="link-primary">Local link</a></div>',
			)
			const local = requireValue(host.querySelector<HTMLElement>('#local'), 'No local element')
			const link = requireValue(host.querySelector<HTMLElement>('#local-link'), 'No local link')
			const resting = readStyle(local, 'color')
			expect(matchesColor(readStyle(link, 'color'), 'rgb(0, 90, 60)')).toBe(false)
			// Below the theme scope the alias has already substituted the token, so the token retuned
			// on an element leaves its text color and moves the colored link, which reads the token.
			local.style.setProperty('--vn-color-primary-emphasis', 'rgb(0, 90, 60)')
			link.style.setProperty('--vn-color-primary-emphasis', 'rgb(0, 90, 60)')
			expect
				.soft(readStyle(local, 'color'), 'token below the theme scope leaves the text')
				.toBe(resting)
			expect
				.soft(
					matchesColor(readStyle(link, 'color'), 'rgb(0, 90, 60)'),
					'token below the theme scope moves the link',
				)
				.toBe(true)
		},
	)

	it.each(TEXT_MODES)(
		'keeps each emphasis class opaque under an opacity step and fades the role class beside it, in %s mode',
		(mode) => {
			const host = scene.mount(
				`<div data-bs-theme="${mode}" style="background-color: var(--bs-body-bg)">${TEXT_TIER_CASES.map(
					({ key }) =>
						`<span class="text-${key}-emphasis text-opacity-50">Emphasis</span><span class="text-${key} text-opacity-50">Role</span>`,
				).join('')}</div>`,
			)
			// The emphasis class sets the color alone, as the release writes it, so the step's variable
			// reaches no alpha there; the role class reads it.
			for (const { key } of TEXT_TIER_CASES) {
				const emphasis = requireValue(
					host.querySelector(`.text-${key}-emphasis`),
					`No ${key} emphasis text`,
				)
				const role = requireValue(host.querySelector(`.text-${key}`), `No ${key} text`)
				expect
					.soft(
						requireValue(parseCSSColor(readStyle(emphasis, 'color')), 'No color')[3],
						`${key} emphasis alpha`,
					)
					.toBe(1)
				expect
					.soft(
						requireValue(parseCSSColor(readStyle(role, 'color')), 'No color')[3],
						`${key} role alpha`,
					)
					.toBe(0.5)
			}
		},
	)

	it('keeps its priority over a later unlayered consumer rule', () => {
		const host = scene.mount(
			'<span class="probe text-success">Utility</span><span class="probe text-bg-success">Pair</span><span class="probe">Consumer</span>',
		)
		scene.load(`.probe { color: ${INHERITED}; background-color: ${INHERITED} }`)
		const utility = requireValue(host.querySelector('.text-success'), 'No utility element')
		const pair = requireValue(host.querySelector('.text-bg-success'), 'No pair element')
		const consumer = requireValue(host.lastElementChild, 'No consumer element')
		expect(readStyle(consumer, 'color')).toBe(INHERITED)
		expect(readStyle(utility, 'color')).not.toBe(INHERITED)
		expect(readStyle(pair, 'background-color')).not.toBe(INHERITED)
	})

	it('yields to an important override inside the utilities layer and to no unlayered one', () => {
		const host = scene.mount(
			'<span class="text-primary">Utility</span><span class="text-bg-dark">Pair</span>',
		)
		const element = requireValue(host.querySelector('.text-primary'), 'No utility element')
		const pair = requireValue(host.querySelector('.text-bg-dark'), 'No pair element')
		const resting = [readStyle(element, 'color'), readStyle(pair, 'background-color')]
		// An unlayered `!important` declaration sorts after every layered one, so the shipped rules
		// keep their values; the same declarations re-opened inside the utilities layer arrive after
		// the shipped rules in that layer and win.
		scene.load(
			`.text-primary { color: ${INHERITED} !important } .text-bg-dark { background-color: ${INHERITED} !important }`,
		)
		expect([readStyle(element, 'color'), readStyle(pair, 'background-color')]).toEqual(resting)
		scene.load(
			`@layer utilities { .text-primary { color: ${INHERITED} !important } .text-bg-dark { background-color: ${INHERITED} !important } }`,
		)
		expect([readStyle(element, 'color'), readStyle(pair, 'background-color')]).toEqual([
			INHERITED,
			INHERITED,
		])
	})
})
