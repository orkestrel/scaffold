import { requireValue } from '@orkestrel/test'
import { build, hoverAccessible, readName, releasePane, releasePointer, stagePane } from '@orkestrel/test/browser'
import { expect, it } from 'vitest'
import { page } from 'vitest/browser'
import { mountShowcase, readSpecimen } from '../../tests/setupBrowser.js'

it('reports which lifted control the pointer reaches', async () => {
	await page.viewport(1280, 800)
	const lines: string[] = []
	await stagePane(1280, 800)
	const frame = window.frameElement?.getBoundingClientRect()
	lines.push(`parent ${String(window.parent.innerWidth)}x${String(window.parent.innerHeight)} frame ${JSON.stringify(frame)} top ${String(window.top?.innerWidth)}`)
	await releasePane()
	for (const [subject, selector, padded] of [
		['Captioned carousel', '.carousel-control-prev', false],
		['Captioned carousel', '.carousel-control-next', false],
		['Fading carousel', '.carousel-control-next', false],
		['Fading carousel', '.carousel-control-next', true],
		['Fading carousel', '.carousel-control-prev', true],
	] as const) {
		const mounted = await mountShowcase()
		const specimen = readSpecimen(mounted.host, subject)
		const control = requireValue(specimen.querySelector<HTMLElement>(selector), 'No control')
		const marker = document.createComment('probe')
		specimen.before(marker)
		const lifted = build('div', padded ? { classes: 'p-2' } : {})
		lifted.append(specimen)
		document.body.prepend(lifted)
		try {
			await stagePane(window.innerWidth, window.innerHeight)
			const box = control.getBoundingClientRect()
			const outcome = await Promise.race([
				hoverAccessible('button', readName(control)).then(() => `hovered ${String(control.matches(':hover'))}`),
				new Promise<string>((resolve) => setTimeout(() => resolve('timed out after 8 s'), 8000)),
			])
			const hit = document.elementFromPoint(box.left + box.width / 2, box.top + box.height / 2)
			lines.push(`${subject} ${selector} padded ${String(padded)} box ${Math.round(box.left)},${Math.round(box.top)},${Math.round(box.width)}x${Math.round(box.height)} inner ${window.innerWidth}x${window.innerHeight} scrollY ${window.scrollY} hit ${hit === control || (hit !== null && control.contains(hit)) ? 'control' : String(hit?.outerHTML.slice(0, 80))} ${outcome}`)
		} finally {
			await releasePane()
			await releasePointer()
			marker.replaceWith(specimen)
			lifted.remove()
			mounted.cleanup()
		}
	}
	expect(lines).toEqual([])
}, 120_000)
