+/**
+ * Stages the tester's print medium and motion preference through the browser provider.
+ *
+ * @param options - The media axes to override.
+ * @returns A promise resolving after the staged queries reach the tester.
+ * @throws Thrown when no axis is supplied or a staged query does not reach the tester.
+ *
+ * @remarks
+ * If `print` is true, uses print; if false, uses screen. If `motion` is true, uses no preference;
+ * if false, uses reduced motion. Omitted axes stay unchanged. Register {@link releaseMedia} in
+ * teardown; it restores provider defaults rather than an earlier override.
+ *
+ * @example
+ * ```ts
+ * await stageMedia({ motion: false, print: true })
+ * await releaseMedia()
+ * ```
+ */
+export async function stageMedia(options: MediaOptions): Promise<void> {
+	if (options.print === undefined && options.motion === undefined) {
+		throw new Error('Media emulation was staged with nothing to emulate')
+	}
+	const queries: string[] = []
+	if (options.print === true) queries.push('print')
+	if (options.motion !== undefined) {
+		queries.push(`(prefers-reduced-motion: ${options.motion ? 'no-preference' : 'reduce'})`)
+	}
+	await sendProtocol('Emulation.setEmulatedMedia', {
+		media: (options.print ?? matchMedia('print').matches) ? 'print' : 'screen',
+		features: [
+			{
+				name: 'prefers-reduced-motion',
+				value:
+					(options.motion ?? matchMedia('(prefers-reduced-motion: no-preference)').matches)
+						? 'no-preference'
+						: 'reduce',
+			},
+		],
+	})
+	await waitForFrame()
+	for (const query of queries) {
+		if (!matchMedia(query).matches) {
+			await releaseMedia()
+			throw new Error(`Media emulation did not reach the tester: ${query}`)
+		}
+	}
