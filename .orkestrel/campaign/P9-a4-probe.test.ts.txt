// Orchestrator probe P9 — reproduces the A4 analyst's unexecuted vectors (claims 2, 3, 5) in the
// real src:browser project. Run by explicit path; deleted after the reading is retained.
import { describe, expect, it } from 'vitest'
import { createMessagePortTransport, createModelContext, createPageServer } from '@src/browser'
import { createTool, createToolManager } from '@orkestrel/tool'
import { waitForAbort, waitForDelay } from '@orkestrel/test'
import { installModelContext } from '../../fixtures/modelContext.js'

describe('P9 vectors', () => {
	it('claim 5a: publish reads the registry at call time or at queue time', async () => {
		const host = document.implementation.createHTMLDocument()
		const fixture = installModelContext(host)
		const bridge = createModelContext({ document: host })
		if (bridge === undefined) throw new Error('bridge absent')
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const pending = bridge.publish(tools)
		tools.clear()
		await pending
		const registered = await fixture.registry.getTools()
		console.log('P9 5a: registered after publish-then-clear =', registered.map((tool) => tool.name))
		bridge.destroy()
		expect(registered.map((tool) => tool.name)).toEqual(['echo'])
	})

	it('claim 5b: a failing tool rejects with the failure value', async () => {
		const host = document.implementation.createHTMLDocument()
		const fixture = installModelContext(host)
		const bridge = createModelContext({ document: host })
		if (bridge === undefined) throw new Error('bridge absent')
		const tools = createToolManager()
		const boom = new Error('kaboom')
		tools.add(
			createTool({
				name: 'fail',
				description: 'fails',
				execute: () => {
					throw boom
				},
			}),
		)
		await bridge.publish(tools)
		const [registered] = await fixture.registry.getTools()
		if (registered === undefined) throw new Error('not registered')
		let caught: unknown
		try {
			await fixture.registry.executeTool(registered, {})
		} catch (error) {
			caught = error
		}
		console.log('P9 5b: rejection is Error =', caught instanceof Error, 'message =', caught instanceof Error ? caught.message : String(caught), 'same as thrown =', caught === boom)
		bridge.destroy()
		expect(caught).toBe(boom)
	})

	it('claim 2: a call after destroy settles rather than hanging', async () => {
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const page = createPageServer({ tools })
		await page.client.connect()
		page.destroy()
		const outcome = await Promise.race([
			page.client.call('echo', {}).then(() => 'resolved', (error: unknown) => `rejected: ${error instanceof Error ? error.message : String(error)}`),
			waitForDelay(1500).then(() => 'still pending after 1500 ms'),
		])
		console.log('P9 claim 2: call after destroy =', outcome)
		expect(outcome).not.toBe('still pending after 1500 ms')
	})

	it('claim 3: the pair stays usable after a cancelled call', async () => {
		const tools = createToolManager()
		tools.add(createTool({ name: 'park', description: 'parks', execute: async (_args, context) => { await waitForAbort(context.signal); return 'aborted' } }))
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const page = createPageServer({ tools })
		await page.client.connect()
		const [parked] = (await page.client.tools()).filter((tool) => tool.name === 'park')
		const [echo] = (await page.client.tools()).filter((tool) => tool.name === 'echo')
		if (parked === undefined || echo === undefined) throw new Error('tools missing')
		const controller = new AbortController()
		const call = parked.execute({}, { signal: controller.signal })
		await waitForDelay(50)
		controller.abort()
		const first = await call.then(() => 'resolved', (error: unknown) => `rejected: ${error instanceof Error ? error.message : String(error)}`)
		const second = await echo.execute({}, { signal: new AbortController().signal })
		console.log('P9 claim 3: cancelled =', first, 'echo after =', second)
		page.destroy()
		expect(second).toBe('ok')
	})
})
