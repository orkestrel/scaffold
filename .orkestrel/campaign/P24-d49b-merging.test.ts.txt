// P24 — Orchestrator probe (AD4-9b claim 7, the Astra vector): does the collision key without `line` report one
// violation per name per file per owner across TypeScript declaration merging, and still one per file and one
// per owner where those differ? Run through the probe project; removed after the reading.
import { it } from 'vitest'
import {
	createPolicySurfaceFixture,
	createPolicySurfaceGuide,
	inspectPolicyWorkspace,
	POLICY_SURFACE_HOST,
} from '../../tests/setupPolicy.js'

const NL = String.fromCharCode(10)
const CLAIMED = 'surface name belongs to one package: waitForCondition'

function readCollisions(files: ReadonlyArray<readonly [string, string]>, owners: ReadonlyArray<string>) {
	const scratch = createPolicySurfaceFixture()
	try {
		for (const owner of owners) {
			scratch.write(`${POLICY_SURFACE_HOST}/guides/${owner}.md`, createPolicySurfaceGuide(['waitForCondition']))
		}
		for (const [path, text] of files) scratch.write(path, text)
		return inspectPolicyWorkspace(scratch.path)
			.filter((violation) => violation.message.startsWith(CLAIMED))
			.map((violation) => `${violation.path}:${violation.line} ${violation.message}`)
	} finally {
		scratch.destroy()
	}
}

it('P24 merging forms and controls', () => {
	const cases: ReadonlyArray<readonly [string, ReadonlyArray<readonly [string, string]>, ReadonlyArray<string>]> = [
		['interface merging, one file, one owner', [['tests/setupServer.ts', ['export interface waitForCondition { a: number }', 'export interface waitForCondition { b: number }'].join(NL)]], ['other']],
		['function + namespace merging, one file, one owner', [['tests/setupServer.ts', ['export function waitForCondition() {}', 'export namespace waitForCondition { export const value = 1 }'].join(NL)]], ['other']],
		['overload, one file, one owner', [['tests/setupServer.ts', ['export function waitForCondition(value: string): void', 'export function waitForCondition(value: number): void', 'export function waitForCondition() {}'].join(NL)]], ['other']],
		['control: distinct files, one owner', [['tests/setupServer.ts', 'export function waitForCondition() {}'], ['tests/setupBrowser.ts', 'export function waitForCondition() {}']], ['other']],
		['control: one file, distinct owners', [['tests/setupServer.ts', ['export function waitForCondition(value: string): void', 'export function waitForCondition() {}'].join(NL)]], ['other', 'third']],
	]
	for (const [label, files, owners] of cases) {
		console.log(`P24 ${label}: ${JSON.stringify(readCollisions(files, owners))}`)
	}
})
