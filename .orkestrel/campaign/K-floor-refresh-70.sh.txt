#!/usr/bin/env bash
# Orchestrator tracked command: the scaffold 0.0.70 floor refresh. The hosted guides are a published surface
# (`dist/host`), and they are read from each repository's pushed `main` because a published tarball ships no
# guides. mcp and ollama moved since 0.0.69; tool, agent, and browser already equal theirs.
# Then `npm run build` regenerates `host.json`, which REFUSES when the staged surface grows, so its exit code
# is the reading that says whether a refreshed mirror brought a new cross-guide collision.
# Log: scaffold tmp/units/floor-refresh-70.log.txt
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
cd "$S" || exit 9
{
	echo "# scaffold floor refresh for 0.0.70 — $(date -u +%FT%TZ) — head $(git rev-parse --short HEAD)"
	for g in mcp ollama; do
		p="/c/Users/mikes/WebstormProjects/$g"
		echo "== guides/$g.md from $g pushed main $(git -C "$p" rev-parse --short origin/main)"
		git -C "$p" show "origin/main:guides/$g.md" > "$S/guides/$g.md"
		echo "written, $(wc -l < "$S/guides/$g.md") lines"
	done
	echo "== git diff --stat guides"
	git diff --stat -- guides
	echo "== npm run build"
	npm run build > "$S/tmp/units/floor-refresh-70-build.out.txt" 2>&1
	echo "exit=$?"
	sed 's/\x1b\[[0-9;]*m//g' "$S/tmp/units/floor-refresh-70-build.out.txt" | grep -E 'staged|entries=|Surface|collision|error|Error' | head -20
	echo "== git diff --stat host.json"
	git diff --stat -- host.json
	echo "== the surface record's own movement"
	git diff -- host.json | grep -cE '^\+' || true
	echo "== status"
	git status --short | grep -v '^?? dist/'
	echo "# end $(date -u +%FT%TZ)"
} > "$S/tmp/units/floor-refresh-70.log.txt" 2>&1
cat "$S/tmp/units/floor-refresh-70.log.txt"
