// Orchestrator host probe for unit A5 (2026-09-15), successor to a5-probe-hop.mjs: what stop() waits on after
// (i) a completed round trip, (ii) a browser-side cancel mid-stream, (iii) the same cancel followed by destroy(),
// each on its own server, with the drain event's counts and the stop duration recorded.
import { createRelay, createRelayProvider } from 'file:///C:/Users/mikes/WebstormProjects/agent/dist/src/core/index.js'
import { createDispatcher } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/router/dist/src/core/index.js'
import { createServer } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/server/dist/src/server/index.js'

const say = (line) => console.log(line)
function createParser() {
	let pending = ''
	return {
		parse(chunk) { pending += chunk; const lines = pending.split('\n'); pending = lines.pop() ?? ''; return lines.filter((l) => l.trim().length > 0).map((l) => JSON.parse(l)) },
		clear() { pending = '' },
	}
}
function createUpstream(deltas, holdMs) {
	const record = { started: 0, abortedAt: undefined, returnedAt: undefined }
	async function* stream(_messages, signal) {
		record.started += 1
		signal.addEventListener('abort', () => { record.abortedAt = performance.now() }, { once: true })
		let content = ''
		try {
			for (const text of deltas) {
				if (signal.aborted) throw new Error('aborted before delta')
				content += text
				yield { channel: 'content', text }
				if (holdMs > 0) await new Promise((resolve) => setTimeout(resolve, holdMs))
			}
			return { content }
		} finally { record.returnedAt = performance.now() }
	}
	return { id: 'upstream', name: 'upstream', record, stream, async generate(m, s) { const g = stream(m, s); let r = await g.next(); while (!r.done) r = await g.next(); return r.value } }
}
const messages = [{ id: '1', role: 'user', content: 'Say hello.' }]

async function boot(upstream, options = {}) {
	const handler = createRelay({ provider: upstream, authorize: () => true })
	const dispatcher = createDispatcher({ routes: [{ method: 'POST', path: '/relay', handler }] })
	const server = createServer({ dispatcher, state: () => undefined, host: '127.0.0.1', ...options })
	server.emitter.on('drain', (pending, upgraded) => say(`  drain event: pending=${pending} upgraded=${upgraded}`))
	const port = await server.start()
	const browser = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser })
	return { server, browser }
}

// (i) completed round trip, then stop()
{
	const upstream = createUpstream(['a', 'b'], 0)
	const { server, browser } = await boot(upstream)
	say(`(i) result=${JSON.stringify(await browser.generate(messages, new AbortController().signal))}`)
	const t0 = performance.now()
	await server.stop()
	say(`(i) stop() after a round trip: ${(performance.now() - t0).toFixed(1)} ms`)
}

// (ii) cancel mid-stream, then stop()
{
	const upstream = createUpstream(['One,', ' two,', ' three'], 400)
	const { server, browser } = await boot(upstream)
	const abort = new AbortController()
	const stream = browser.stream(messages, abort.signal)
	await stream.next()
	const cancelledAt = performance.now()
	abort.abort()
	try { await stream.next() } catch (error) { say(`(ii) after abort: ${error.name} partial=${JSON.stringify(error.partial)}`) }
	const t0 = performance.now()
	await server.stop()
	const t1 = performance.now()
	say(`(ii) stop() after a cancel: ${(t1 - t0).toFixed(1)} ms; upstream aborted after ${upstream.record.abortedAt === undefined ? 'never' : (upstream.record.abortedAt - cancelledAt).toFixed(1) + ' ms'}; upstream generator returned after ${upstream.record.returnedAt === undefined ? 'never' : (upstream.record.returnedAt - cancelledAt).toFixed(1) + ' ms'}`)
}

// (iii) cancel mid-stream, wait for the upstream generator to return, then stop()
{
	const upstream = createUpstream(['One,', ' two,', ' three'], 400)
	const { server, browser } = await boot(upstream)
	const abort = new AbortController()
	const stream = browser.stream(messages, abort.signal)
	await stream.next()
	abort.abort()
	try { await stream.next() } catch {}
	for (let i = 0; i < 300 && upstream.record.returnedAt === undefined; i += 1) await new Promise((r) => setTimeout(r, 10))
	say(`(iii) upstream generator returned: ${upstream.record.returnedAt !== undefined}`)
	const t0 = performance.now()
	await server.stop()
	say(`(iii) stop() after the generator returned: ${(performance.now() - t0).toFixed(1)} ms`)
}

// (iv) cancel mid-stream, then destroy()
{

}

// (v) cancel mid-stream with drain: 0, then stop()
{
	const upstream = createUpstream(['One,', ' two,', ' three'], 400)

}
