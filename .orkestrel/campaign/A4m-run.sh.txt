#!/usr/bin/env bash
# A4m objective audit lane — GPT-6 Astra, read-only codex exec rooted at the mcp checkout. Cap 2400 s.
# Successor of A4k-run.sh: unit name, brief, journal, evidence set (the U4e-f pair, the A4l verdicts, the
# A4k analyst verdict, the P20 records, the U4e-f-only and whole patches, the after-u4e-f gates, the probe).
# Astra is the cross-engine auditor of a fix Opus wrote.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
J="$SCAF/tmp/codex/A4m-audit.jsonl"
L="$SCAF/tmp/codex/A4m-audit-last.md"
B="$MCP/tmp/codex/A4m-audit-brief.md"
cp "$SCAF/tmp/units/A4m-audit-brief.md" "$B"
cp "$SCAF/tmp/units/A4m-u4e-f-only.patch" "$MCP/tmp/codex/A4m-u4e-f-only.patch"
cp "$SCAF/tmp/units/A4m-whole.patch" "$MCP/tmp/codex/A4m-whole.patch"
cp "$SCAF/tmp/units/A4l-u4e-e-only.patch" "$MCP/tmp/codex/A4l-u4e-e-only.patch"
for f in U4e-f-mcp-shape-brief.md U4e-f-mcp-shape-report.md A4l-audit-reviewer.md A4l-audit-checker.md A4k-audit-analyst.md P20-a4k-probe.md P20-a4l-replay.md U4e-f-mcp-gates-orchestrator.log.txt U4e-f-mcp-gates-test-full.log.txt collide3-mcp-after-u4e-f.txt U4-chain-audit-verdict.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f"
done
echo "launch unit=A4m-audit model=gpt-6-astra sandbox=read-only cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/A4m-audit.launch.txt"
timeout 2400 codex exec --json -C "$MCP" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (every evidence file it names is staged beside it in tmp/codex/). You hold the OBJECTIVE lane (analyst) and the cross-engine seat: Claude Opus 5 wrote the unit under audit, and a GPT-6 Astra session wrote the round before it; audit from source and the Orchestrator's records, never from memory of any prior session. Perform the audit directly and spawn nothing. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/A4m-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/A4m-audit.launch.txt"
echo "unit=A4m-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
