<h1>B2 — OllamaProvider in the browser, direct and relayed</h1>
<pre id="out"></pre>
<script type="module">
import { createOllama } from '@orkestrel/ollama'
import { createRelayProvider, isProviderError, isProviderAbortError } from '@orkestrel/agent'
import { createNDJSONParser } from '@orkestrel/ndjson'

const out = document.getElementById('out')
const log = (line) => {
	out.textContent += line + '\n'
	console.log(line)
}
const messages = (text) => [{ id: 'm', role: 'user', content: text }]
const options = { num_predict: 24, temperature: 0 }

log('user agent: ' + navigator.userAgent)

try {
	const direct = createOllama({ model: 'MODEL_NAME', url: 'DAEMON_URL', options })
	const result = await direct.generate(messages('Reply with the single word: pong'), new AbortController().signal)
	log('case 1 direct generate: content=' + JSON.stringify(result.content) + ' usage=' + JSON.stringify(result.usage))
} catch (error) {
	log('case 1 FAILED: ' + String(error))
}

try {
	const direct = createOllama({ model: 'MODEL_NAME', url: 'DAEMON_URL', options })
	const controller = new AbortController()
	const stream = direct.stream(messages('Count slowly from one to twenty in words.'), controller.signal)
	let deltas = 0
	let text = ''
	for await (const delta of stream) {
		deltas += 1
		if (delta.channel === 'content') text += delta.text
		if (deltas === 2) controller.abort(new Error('browser cancel'))
	}
	log('case 2 FAILED: the stream ended without throwing after the cancel')
} catch (error) {
	log('case 2 direct stream cancel: ' + (isProviderAbortError(error) ? 'ProviderAbortError partial=' + JSON.stringify(error.partial) : 'OTHER ' + String(error)))
}

try {
	const relayed = createRelayProvider({ url: location.origin + '/relay', parser: createNDJSONParser, headers: () => ({ authorization: 'Bearer fixture-token' }) })
	const result = await relayed.generate(messages('Reply with the single word: pong'), new AbortController().signal)
	log('case 3 relayed generate: content=' + JSON.stringify(result.content) + ' usage=' + JSON.stringify(result.usage))
} catch (error) {
	log('case 3 FAILED: ' + String(error))
}

try {
	const relayed = createRelayProvider({ url: location.origin + '/relay', parser: createNDJSONParser, headers: () => ({ authorization: 'Bearer wrong-token' }) })
	await relayed.generate(messages('hi'), new AbortController().signal)
	log('case 4 FAILED: no throw on a refused token')
} catch (error) {
	log('case 4 relayed refusal: ' + (isProviderError(error) ? 'ProviderError ' + error.code + ' ' + error.status + ' "' + error.message + '"' : 'OTHER ' + String(error)))
}

log('done')
</script>
