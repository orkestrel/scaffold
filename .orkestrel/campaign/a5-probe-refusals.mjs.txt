// Orchestrator host probe for unit A5 (2026-09-15), successor to a5-probe-stop.mjs: which refusal over the socket
// makes stop() slow (a5-probe-hop.mjs measured 3001.6 ms after 405/404/401/413/cancel in one server; a5-probe-stop.mjs
// measured under 3 ms after a round trip or a cancel alone). One server per variant: (a) 405 and 404 from the
// dispatcher, (b) 401 from authorize, (c) 413 from the relay's limit, each followed by stop() with the drain counts.
import { createRelay, createRelayProvider } from 'file:///C:/Users/mikes/WebstormProjects/agent/dist/src/core/index.js'
import { createDispatcher } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/router/dist/src/core/index.js'
import { createServer } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/server/dist/src/server/index.js'

const say = (line) => console.log(line)
function createParser() {
	let pending = ''
	return {
		parse(chunk) { pending += chunk; const lines = pending.split('\n'); pending = lines.pop() ?? ''; return lines.filter((l) => l.trim().length > 0).map((l) => JSON.parse(l)) },
		clear() { pending = '' },
	}
}
function createUpstream() {
	const record = { started: 0 }
	async function* stream() { record.started += 1; yield { channel: 'content', text: 'a' }; return { content: 'a' } }
	return { id: 'upstream', name: 'upstream', record, stream, async generate(m, s) { const g = stream(m, s); let r = await g.next(); while (!r.done) r = await g.next(); return r.value } }
}
const messages = [{ id: '1', role: 'user', content: 'Say hello.' }]
async function boot(relay) {
	const dispatcher = createDispatcher({ routes: [{ method: 'POST', path: '/relay', handler: relay }] })
	const server = createServer({ dispatcher, state: () => undefined, host: '127.0.0.1' })
	server.emitter.on('drain', (pending, upgraded) => say(`  drain event: pending=${pending} upgraded=${upgraded}`))
	const port = await server.start()
	return { server, port }
}
async function timeStop(label, server) {
	const t0 = performance.now()
	await server.stop()
	say(`${label} stop(): ${(performance.now() - t0).toFixed(1)} ms`)
}

// (a) 405 and 404 from the dispatcher
{
	const upstream = createUpstream()
	const { server, port } = await boot(createRelay({ provider: upstream, authorize: () => true }))
	const get = await fetch(`http://127.0.0.1:${port}/relay`)
	say(`(a) GET /relay -> ${get.status} allow=${get.headers.get('allow')} body=${JSON.stringify(await get.text())}`)
	const other = await fetch(`http://127.0.0.1:${port}/other`, { method: 'POST', body: '{"messages":[]}' })
	say(`(a) POST /other -> ${other.status} body=${JSON.stringify(await other.text())}`)
	await timeStop('(a)', server)
}

// (b) 401 from authorize
{
	const upstream = createUpstream()
	const { server, port } = await boot(createRelay({ provider: upstream, authorize: (request) => request.headers.get('authorization') === 'Bearer fixture' }))
	const wrong = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser, headers: () => ({ authorization: 'Bearer wrong' }) })
	try { await wrong.generate(messages, new AbortController().signal) } catch (error) { say(`(b) 401: status=${error.status} started=${upstream.record.started}`) }
	await timeStop('(b)', server)
}

// (c) 413 from the relay's limit
{
	const upstream = createUpstream()
	const { server, port } = await boot(createRelay({ provider: upstream, authorize: () => true, limit: 8 }))
	const tiny = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser })
	try { await tiny.generate(messages, new AbortController().signal) } catch (error) { say(`(c) 413: status=${error.status} started=${upstream.record.started}`) }
	await timeStop('(c)', server)
}

// (d) 413 then a round trip on the same server, then stop()
{
	const upstream = createUpstream()
	const dispatcher = createDispatcher({ routes: [
		{ method: 'POST', path: '/tight', handler: createRelay({ provider: upstream, authorize: () => true, limit: 8 }) },
		{ method: 'POST', path: '/relay', handler: createRelay({ provider: upstream, authorize: () => true }) },
	] })
	const server = createServer({ dispatcher, state: () => undefined, host: '127.0.0.1' })
	server.emitter.on('drain', (pending, upgraded) => say(`  drain event: pending=${pending} upgraded=${upgraded}`))
	const port = await server.start()
	const tiny = createRelayProvider({ url: `http://127.0.0.1:${port}/tight`, parser: createParser })
	try { await tiny.generate(messages, new AbortController().signal) } catch (error) { say(`(d) 413: status=${error.status}`) }
	const browser = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser })
	say(`(d) round trip after the 413: ${JSON.stringify(await browser.generate(messages, new AbortController().signal))}`)
	await timeStop('(d)', server)
}
say('done')
