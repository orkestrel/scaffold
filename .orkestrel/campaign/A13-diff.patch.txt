+/**
+ * Derives the page's import map from the `@orkestrel` packages installed under a root.
+ *
+ * @param root - The workspace root holding `node_modules/@orkestrel`
+ * @returns Each installed package's bare specifier mapped to its declared ESM root entry,

+ * installed: a written map omits a closure member silently, and the browser then fails to
+ * resolve that member mid-run. A package whose manifest declares no
+ * `exports['.'].import.default` entry is left out, because it publishes no ESM root entry to

+export function buildImportMap(root: URL | string): Readonly<Record<string, string>> {
+	const path = readPath(root)
+	const base = join(path, 'node_modules', '@orkestrel')
+	const imports: Record<string, string> = {}
+	for (const name of readdirSync(base)) {
+		const entry = readModuleEntry(

+	}

+		imports[manifest.name] = `${DIST_PATH}/${own.replace(/^dist\//, '')}`

+/**
+ * Reads one package manifest's declared ESM root entry, relative to the package directory.
+ *
+ * @param manifest - The parsed `package.json` record, or `undefined` when it failed to parse
+ * @returns The entry path without its leading `./`, or `undefined` when the manifest declares

+export function readModuleEntry(manifest: unknown): string | undefined {

/** Holds one retained attempt's outcome and the traffic each recorder observed for it. */
interface Attempt {
	readonly outcome: PageOutcome
	readonly receipts: readonly PageReceipt[]
	readonly resources: readonly string[]
	/** Every path the page requested inside the operation window, in order. */
	readonly page: readonly string[]
	/** Every request the relay route received. */
	readonly relay: readonly RecordedRequest[]
	/** Every request the daemon transport issued. */
	readonly daemon: readonly RecordedRequest[]
}

describe('Agent tool loop in a real page (live) — a page tool through the relay', () => {
	it(
		'executes a page tool through an agent over a live Ollama relay and feeds its result into the next model turn',
		async () => {
			const attempts: Attempt[] = []
			const satisfied = await retryUntil(
				'have the model call the page tool and carry its receipt into the next turn',
				async () => {
					const daemon = createRecordingTransport()
					const session = await createPageSession({
						executable: BROWSER.executable,
						args: PAGE_BROWSER_ARGS,
						provider: createOllama({
							model: OLLAMA_CONFIG.model,
							url: OLLAMA_CONFIG.host,
							options: TOOL_LOOP_OPTIONS,
							fetch: daemon.fetch,
						}),
					})
					// The window opens after the driver parked its operations, so the module
					// fetches the page made while loading are outside it.
					const mark = session.requests.length
					try {
						const outcome = await readPage(
							session.page,
							`page.run(${JSON.stringify({ ...RUN, url: `${session.fixture.url}${INFERENCE_PATH}`, authorization: OBFUSCATED })})`,
							isPageOutcome,
						)
						const attempt: Attempt = {
							outcome,
							receipts: await readPage(session.page, 'page.receipts()', isPageReceipts),
							resources: await readPage(session.page, 'page.resources()', arrayOf(isString)),
							page: session.requests
								.slice(mark)
								.map((request) => `${request.method} ${new URL(request.url).pathname}`),
							relay: [...session.fixture.requests],
							daemon: [...daemon.requests],
						}
						attempts.push(attempt)
						return attempt
					} finally {
						await session.destroy()
					}
				},
				(attempt) =>
					attempt.outcome.tools.some((tool) => tool.call.name === PAGE_TOOL.name) &&
					attempt.receipts.length > 0 &&
					attempt.outcome.turns.length > 1,
				{ attempts: 3, budget: RETRY_BUDGET },
			)

			// ── The request accounting, over EVERY retained attempt ───────────────────
			// A satisfied attempt never excuses an attempt whose traffic was wrong, so the
			// invariants are asserted on each one: one page `POST /inference` per turn, one
			// relay request per turn, one daemon `/api/chat` per turn, and nothing else the
			// page asked for inside the window — which is the zero-request tool-execution
			// window stated as the whole window rather than as a timing slice.
			for (const attempt of attempts) {
				const turns = attempt.outcome.turns.length
				expect(attempt.page).toEqual(Array.from({ length: turns }, () => `POST ${INFERENCE_PATH}`))
				expect(attempt.relay.filter((request) => request.path === INFERENCE_PATH)).toHaveLength(
					turns,
				)
				expect(attempt.daemon.map((request) => request.path)).toEqual(
					Array.from({ length: turns }, () => OLLAMA_CHAT_PATH),
				)
				expect(
					attempt.relay.every(
						(request) =>
							request.path !== INFERENCE_PATH || request.headers.authorization === OBFUSCATED,
					),
				).toBe(true)
			}

			// ── The execution receipt ─────────────────────────────────────────────────
			const dispatched = requireValue(
				satisfied.outcome.tools.find((tool) => tool.call.name === PAGE_TOOL.name),
				'the satisfied attempt reported no call to the page tool',
			)
			const written = requireValue(
				satisfied.receipts.at(0),
				'the satisfied attempt wrote no receipt into the page',
			)
			// The model's recorded call carries the note it chose, and the page's DOM carries
			// that same note — the argument crossed the wire and reached the handler.
			expect(dispatched.call.arguments.note).toBe(written.note)
			// The agent reported the tool's own return value, and the DOM holds the same
			// receipt: the value was minted during execution, in the page.
			expect(dispatched.result).toMatchObject({ success: true, value: written.receipt })
			expect(written.receipt.startsWith('receipt-')).toBe(true)
			// The receipt was absent from the prompt, so a later request carrying it can only
			// be the tool result being fed back.
			expect(RUN.system.includes(written.receipt)).toBe(false)
			expect(RUN.prompt.includes(written.receipt)).toBe(false)
			// The receipt reached the relay's own wire and the daemon's own wire, on a request
			// after the one that requested the call.
			const relayed = satisfied.relay.filter(
				(request) => request.path === INFERENCE_PATH && request.text.includes(written.receipt),
			)
			expect(relayed.length).toBeGreaterThanOrEqual(1)
			const fed = satisfied.daemon.filter((request) =>
				wireMessages(request).some((message) => message.content.includes(written.receipt)),
			)
			expect(fed.length).toBeGreaterThanOrEqual(1)
			// The tool was advertised to the daemon on the first turn, by the name the shared
			// definition carries.
			expect(requireValue(satisfied.daemon.at(0)).text).toContain(PAGE_TOOL.name)
			// The run settled rather than exhausting or cancelling.
			const result = requireValue(
				satisfied.outcome.result,
				'the satisfied attempt reported a failure instead of a result',
			)
			expect(result.partial).toBe(false)
			expect(result.content.length).toBeGreaterThan(0)
			// The page's own Resource Timing drain corroborates the browser's request log.
			expect(
				satisfied.resources.filter((name) => name.endsWith(INFERENCE_PATH)).length,
			).toBeGreaterThanOrEqual(1)
		},
		RETRY_TIMEOUT,
	)
})

describe('Ollama in a real page (live) — the daemon driven directly', () => {
	it(
		'drives the daemon directly from the page behind its origin gate',
		async () => {
			const session = await createPageSession({
				executable: BROWSER.executable,
				args: PAGE_BROWSER_ARGS,
			})
			try {
				// The page dials the daemon from its own ephemeral loopback origin, so the daemon
				// decides a preflight first. A host that refuses it is a host where the guide's
				// direct-daemon claim is false, and this throws naming OLLAMA_ORIGINS.
				await requireDaemonOrigin(new URL(session.fixture.url).origin)
				const mark = session.requests.length
				const direct = await readPage(
					session.page,
					`page.direct(${JSON.stringify({ model: OLLAMA_CONFIG.model, url: OLLAMA_CONFIG.host, options: FAST_OPTIONS, timeout: RUN_TIMEOUT })})`,
					isPageGeneration,
				)
				expect(direct.name).toBe('ollama')
				expect(direct.content.length).toBeGreaterThan(0)
				// Asserted by filtering, never by a total: the browser sends its own preflight to
				// the daemon origin and that preflight is its own recorded request.
				const daemonOrigin = new URL(OLLAMA_CONFIG.host).origin
				const chats = session.requests
					.slice(mark)
					.filter(
						(request) =>
							request.method === 'POST' &&
							new URL(request.url).origin === daemonOrigin &&
							new URL(request.url).pathname === OLLAMA_CHAT_PATH,
					)
				expect(chats.length).toBeGreaterThanOrEqual(1)
				// Nothing reached the relay: this page talked to the daemon itself.
				expect(
					session.fixture.requests.filter((request) => request.path === INFERENCE_PATH),
				).toEqual([])
			} finally {
				await session.destroy()
			}
		},
		RUN_TIMEOUT,
	)
})
