#!/usr/bin/env bash
# AD4-9b objective audit lane — GPT-6 Astra, read-only codex exec rooted at the scaffold checkout. Cap 1800 s.
# Successor of A5-run.sh in shape: the brief and every evidence file already sit inside the scaffold tree
# (tmp/units/AD49b-audit-brief.md and .orkestrel/campaign/), so nothing is staged.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
J="$SCAF/tmp/codex/AD49b-audit.jsonl"
L="$SCAF/tmp/codex/AD49b-audit-last.md"
B="$SCAF/tmp/units/AD49b-audit-brief.md"
echo "launch unit=AD49b-audit model=gpt-6-astra sandbox=read-only cap=1800 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/AD49b-audit.launch.txt"
timeout 1800 codex exec --json -C "$SCAF" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (the evidence files it names live under .orkestrel/campaign/ in this tree). You hold the OBJECTIVE lane (analyst): a Claude Sonnet builder wrote the unit under audit. Perform the audit directly and spawn nothing; the sandbox cannot run Vitest, so name each unexecuted vector UNRESOLVED with its exact command. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/AD49b-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/AD49b-audit.launch.txt"
echo "unit=AD49b-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
