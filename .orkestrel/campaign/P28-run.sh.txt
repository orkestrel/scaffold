#!/usr/bin/env bash
# P28 — Orchestrator probe: does a stage that rejects still remove its scratch tree? Seeds the composition's
# module walk with a specifier the installed tree serves nothing for, so the stage's own guard throws after the
# consumer is installed and the browser is launched, then reads the temporary directory for a surviving
# `distribution-*` tree. Restores from a copy taken first and proves the restore with cmp.
# Run only with no gate and no lane live. Log: scratchpad/p28.log.txt
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
M=/c/Users/mikes/WebstormProjects/mcp
SP=/c/Users/mikes/AppData/Local/Temp/claude/C--Users-mikes-WebstormProjects-scaffold/bab586ba-1c34-4a7e-a280-b3a7e9f303d2/scratchpad
TEMP=/c/Users/mikes/AppData/Local/Temp
cd "$M" || exit 9
echo "# P28 — $(date -u +%FT%TZ) — mcp head $(git rev-parse --short HEAD)"
echo "== distribution-* trees before"
ls -d $TEMP/distribution-* 2>/dev/null | wc -l
cp tests/distribution.test.ts "$SP/distribution.test.ts.p28-backup"
node "$SP/p28-mutate.mjs"
npm run test:distribution -- --mode release > "$SP/p28-planted.out.txt" 2>&1
echo "exit=$?"
sed 's/\x1b\[[0-9;]*m//g' "$SP/p28-planted.out.txt" | grep -E 'Tests |Test Files|serves no module|planted-p28|FAIL|Error' | head -10
echo "== distribution-* trees after the planted stage failure"
ls -d $TEMP/distribution-* 2>/dev/null | wc -l
ls -d $TEMP/distribution-* 2>/dev/null | head -3
cp "$SP/distribution.test.ts.p28-backup" tests/distribution.test.ts
cmp tests/distribution.test.ts "$SP/distribution.test.ts.p28-backup" && echo "restored: cmp 0"
echo "== green after the restore"
npm run test:distribution -- --mode release > "$SP/p28-restored.out.txt" 2>&1
echo "exit=$?"
sed 's/\x1b\[[0-9;]*m//g' "$SP/p28-restored.out.txt" | grep -E 'Tests |Test Files'
echo "== distribution-* trees at the end"
ls -d $TEMP/distribution-* 2>/dev/null | wc -l
git -C "$M" status --short
echo "# P28 end $(date -u +%FT%TZ)"
