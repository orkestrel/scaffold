// P28 mutation: seed the composition's module walk with a specifier the installed tree serves nothing for, so
// the stage's own `outside` guard throws after the consumer is installed and the browser is launched. That is
// the exact failure path seam S6 claim 64 named, and the probe reads whether the scratch tree survives it.
// Usage: node p28-mutate.mjs
import { readFileSync, writeFileSync } from 'node:fs'

const file = 'C:/Users/mikes/WebstormProjects/mcp/tests/distribution.test.ts'
const before = 'const modules = walkModules(consumer, ['
const after = "const modules = walkModules(consumer, [\n\t\t\t'@orkestrel/planted-p28',"

const text = readFileSync(file, 'utf8')
const count = text.split(before).length - 1
if (count !== 1) {
	console.error(`expected one walk seed, found ${count}`)
	process.exit(2)
}
writeFileSync(file, text.replace(before, after))
console.log('mutated: the walk is seeded with @orkestrel/planted-p28, which the tree serves nothing for')
