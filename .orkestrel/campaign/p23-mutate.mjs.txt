// P23 mutation: restore `line` in the collision dedupe key of tests/setupPolicy.ts (line 1985) so the
// pin `reports an overloaded export's collision once` must read three entries. Usage: node p23-mutate.mjs
import { readFileSync, writeFileSync } from 'node:fs'
const file = 'C:/Users/mikes/WebstormProjects/scaffold/tests/setupPolicy.ts'
const before = '`${declaration.path}\\n${declaration.name}\\n${owner}`'
const after = '`${declaration.path}\\n${declaration.line}\\n${declaration.name}\\n${owner}`'
const text = readFileSync(file, 'utf8')
const count = text.split(before).length - 1
if (count !== 1) { console.error(`expected one key occurrence, found ${count}`); process.exit(2) }
writeFileSync(file, text.replace(before, after))
console.log('mutated: the key carries line again')
