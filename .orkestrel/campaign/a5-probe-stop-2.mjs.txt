// Orchestrator host probe for unit A5 (2026-09-15), successor to a5-probe-stop.mjs: the cancel-then-stop case
// varied on the two things the fast and the slow probes differed in — whether the upstream generator has a
// try/finally, and how long the caller waits between the browser-side abort and stop() — with the pending
// count at the drain event, the generator's return, and stop()'s duration recorded per variant.
import { createRelay, createRelayProvider } from 'file:///C:/Users/mikes/WebstormProjects/agent/dist/src/core/index.js'
import { createDispatcher } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/router/dist/src/core/index.js'
import { createServer } from 'file:///C:/Users/mikes/WebstormProjects/agent/node_modules/@orkestrel/server/dist/src/server/index.js'

const say = (line) => console.log(line)
function createParser() {
	let pending = ''
	return {
		parse(chunk) { pending += chunk; const lines = pending.split('\n'); pending = lines.pop() ?? ''; return lines.filter((l) => l.trim().length > 0).map((l) => JSON.parse(l)) },
		clear() { pending = '' },
	}
}
function createUpstream(withFinally, holdMs) {
	const record = { abortedAt: undefined, returnedAt: undefined, resumedAt: undefined }
	async function* plain(_messages, signal) {
		signal.addEventListener('abort', () => { record.abortedAt = performance.now() }, { once: true })
		yield { channel: 'content', text: 'One,' }
		await new Promise((resolve) => setTimeout(resolve, holdMs))
		record.resumedAt = performance.now()
		yield { channel: 'content', text: ' two,' }
		await new Promise((resolve) => setTimeout(resolve, holdMs))
		yield { channel: 'content', text: ' three' }
		return { content: 'One, two, three' }
	}
	async function* guarded(messages, signal) {
		try { return yield* plain(messages, signal) } finally { record.returnedAt = performance.now() }
	}
	const stream = withFinally ? guarded : plain
	return { id: 'upstream', name: 'upstream', record, stream, async generate(m, s) { const g = stream(m, s); let r = await g.next(); while (!r.done) r = await g.next(); return r.value } }
}
const messages = [{ id: '1', role: 'user', content: 'Say hello.' }]

async function variant(withFinally, delayMs, holdMs) {
	const upstream = createUpstream(withFinally, holdMs)
	const dispatcher = createDispatcher({ routes: [{ method: 'POST', path: '/relay', handler: createRelay({ provider: upstream, authorize: () => true }) }] })
	const server = createServer({ dispatcher, state: () => undefined, host: '127.0.0.1' })
	let drained = ''
	server.emitter.on('drain', (pending, upgraded) => { drained = `pending=${pending} upgraded=${upgraded}` })
	const port = await server.start()
	const browser = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser })
	const abort = new AbortController()
	const stream = browser.stream(messages, abort.signal)
	await stream.next()
	const cancelledAt = performance.now()
	abort.abort()
	try { await stream.next() } catch {}
	if (delayMs > 0) await new Promise((resolve) => setTimeout(resolve, delayMs))
	const t0 = performance.now()
	await server.stop()
	const took = performance.now() - t0
	const at = (value) => (value === undefined ? 'never' : `${(value - cancelledAt).toFixed(1)} ms`)
	say(`finally=${withFinally} delay=${delayMs} hold=${holdMs}: stop() ${took.toFixed(1)} ms; drain ${drained}; upstream aborted at ${at(upstream.record.abortedAt)}; generator resumed after the hold at ${at(upstream.record.resumedAt)}; finally ran at ${at(upstream.record.returnedAt)}`)
}

for (const withFinally of [false, true]) for (const delayMs of [0, 10, 100]) await variant(withFinally, delayMs, 400)
await variant(false, 10, 5000)
say('done')
