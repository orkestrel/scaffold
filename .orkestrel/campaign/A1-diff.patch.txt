+## Contract validation and errors
+
+A contract compiles at construction. Its schema projects to `parameters` through
+`schemaToParameters(createContract(shape).schema)` from `@orkestrel/contract`; an undefined
+projection leaves parameters absent. Supplying `contract` and `parameters` together throws a
+`ToolError` with code `SCHEMA`.
+
+The contract's `explain` method reports parse faults before the handler runs. It accepts coercible
+values, such as a numeric string for a number, and passes the original arguments unchanged.
+Handlers still narrow the fields they consume. A parse fault throws `ToolError` with code
+`ARGUMENTS`. The message names the first fault's path and reason, then its expected and received
+values when the fault carries them. Array paths join with `.`; string paths stay unchanged. A root
+array path is empty, so its message starts with `: `. The error's `context.faults` holds the full
+report. `variant` and `oneOf` faults carry neither expected nor received; a missing field carries
+expected alone. Contract construction errors from the dependency propagate unchanged.
+
+Use the guard to narrow an error at the direct execution boundary:
+
+```ts
+import type { ToolErrorCode } from '@orkestrel/tool'
+import { numberShape, objectShape } from '@orkestrel/contract'
+import { ToolError, createTool, isToolError } from '@orkestrel/tool'
+
+const tool = createTool({
+	name: 'amount',
+	contract: objectShape({ amount: numberShape() }),
+	execute: (args) => args.amount,
+})
+const context = { signal: new AbortController().signal }
+tool.execute({ amount: 3 }, context) // 3
+try {
+	tool.execute({ amount: 'invalid' }, context)
+} catch (error) {
+	if (!isToolError(error)) throw error
+	const code: ToolErrorCode = error.code
+	code // 'ARGUMENTS'

+conflict.code // 'SCHEMA'

+```

+than the error instance or fault report. `ToolError` extends `Error`, exposes its readonly `code`

import type { ToolErrorCode } from './types.js'

/**
 * Reports a schema conflict or argument validation failure with a machine-readable code.
 *
 * @example
 * ```ts
 * import { ToolError } from '@orkestrel/tool'
 *
 * const error = new ToolError('SCHEMA', 'Choose contract or parameters')
 * error.code // 'SCHEMA'
 * ```
 */
export class ToolError extends Error {
	readonly code: ToolErrorCode
	readonly context?: Readonly<Record<string, unknown>>

	constructor(code: ToolErrorCode, message: string, context?: Readonly<Record<string, unknown>>) {
		super(message)
		this.name = 'ToolError'
		this.code = code
		if (context !== undefined) this.context = context
	}
}
