#!/usr/bin/env bash
# A4c objective audit lane — GPT-6 Astra, read-only codex exec rooted at the mcp checkout. Cap 2400 s.
# Successor of A4b-run.sh: unit name, brief, journal; stages the U4f-only interdiff, the chain diff, the U4f
# report and its red/green logs, the P11 record and instrument, the A4b verdicts, the gate logs, the collision
# probe, and the G5 distillate beside the brief. Astra is the cross-engine lane (Opus wrote the chain incl. U4f).
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
J="$SCAF/tmp/codex/A4c-audit.jsonl"
L="$SCAF/tmp/codex/A4c-audit-last.md"
B="$MCP/tmp/codex/A4c-audit-brief.md"
cp "$SCAF/tmp/units/A4c-audit-brief.md" "$B"
cp "$SCAF/tmp/units/A4c-u4f-only.patch" "$MCP/tmp/codex/A4c-u4f-only.patch"
cp "$SCAF/tmp/units/A4c-diff.patch" "$MCP/tmp/codex/A4c-diff.patch"
for f in U4f-mcp-browser-fix-brief.md U4f-mcp-browser-report.md U4f-red.log.txt U4f-green.log.txt P11-a4b-probe.md P11-a4b-probe.test.ts.txt A4b-audit-analyst.md A4b-audit-checker.md U4f-mcp-gates-orchestrator.log.txt U4f-mcp-gates-test-full.log.txt collide3-mcp-after-u4f.txt G5-webmcp-distillate.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f"
done
echo "launch unit=A4c-audit model=gpt-6-astra sandbox=read-only cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/A4c-audit.launch.txt"
timeout 2400 codex exec --json -C "$MCP" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (every evidence file it names is staged beside it in tmp/codex/). You hold the OBJECTIVE lane (analyst) and the cross-engine lane: Opus 5 wrote the unit under audit. Perform the audit directly and spawn nothing. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/A4c-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/A4c-audit.launch.txt"
echo "unit=A4c-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
