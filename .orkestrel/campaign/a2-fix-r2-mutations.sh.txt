#!/usr/bin/env bash
# Successor of run-a2-fix-mutations.sh for the A2-fix-r2 commit: the same runner, writing its own log.
SHA="$1"
LIST="$2"
cd /c/Users/mikes/WebstormProjects/agent-audit || exit 9
OUT=/c/Users/mikes/WebstormProjects/scaffold/tmp/units/a2-fix-r2-probes.log.txt
: > "$OUT"
git checkout --detach "$SHA" >> "$OUT" 2>&1 || exit 8
git rev-parse HEAD >> "$OUT"
while IFS=$'\t' read -r name file expr test; do
	[ -z "$name" ] && continue
	echo "=== $name: $file :: $expr" >> "$OUT"
	before=$(md5sum "$file")
	sed -i "$expr" "$file"
	after=$(md5sum "$file")
	if [ "$before" = "$after" ]; then
		echo "MUTATION DID NOT APPLY" >> "$OUT"
	fi
	git diff --stat >> "$OUT"
	npx vitest run --config vite.config.ts --no-cache --reporter=dot --project src:core "$test" >> "$OUT" 2>&1
	echo "=== $name exit=$?" >> "$OUT"
	git checkout -- "$file"
done < "$LIST"
echo "=== control: restored tree, whole core project" >> "$OUT"
git status --porcelain >> "$OUT"
npx vitest run --config vite.config.ts --no-cache --reporter=dot --project src:core >> "$OUT" 2>&1
echo "=== control exit=$?" >> "$OUT"
grep -E "^===|MUTATION DID NOT APPLY|Tests |FAIL |AssertionError" "$OUT" | head -60
