#!/usr/bin/env bash
# Orchestrator tracked command: a target's release proof outside the upload window, generic successor of
# release-scaffold-chain.sh. Runs the package's own prepublishOnly script and logs each line, then a pack
# dry run. Usage: bash release-chain.sh <checkout> <label>. Logs: tmp/units/release-<checkout>-<label>*.txt
# in the scaffold checkout.
set -u
C="${1:?checkout}"
LABEL="${2:?label}"
S=/c/Users/mikes/WebstormProjects/scaffold
T="/c/Users/mikes/WebstormProjects/$C"
LOG="$S/tmp/units/release-$C-$LABEL.log.txt"
OUT="$S/tmp/units/release-$C-$LABEL-prepublish.out.txt"
cd "$T" || exit 9
{
	echo "# $C release chain ($LABEL) — $(date -u +%FT%TZ) — $(grep -m1 '"version"' package.json | tr -d ' \t,') — head $(git rev-parse --short HEAD)"
	echo "== npm run prepublishOnly"
	npm run prepublishOnly > "$OUT" 2>&1
	code=$?
	sed 's/\x1b\[[0-9;]*m//g' "$OUT" | grep -E "^> |Tests |Test Files|error|Error|FAIL|exit" | head -60
	echo "exit=$code"
	echo "== npm pack --dry-run"
	npm pack --dry-run > "$S/tmp/units/release-$C-$LABEL-pack.out.txt" 2>&1
	echo "exit=$?"
	grep -E "package size|unpacked size|total files|shasum" "$S/tmp/units/release-$C-$LABEL-pack.out.txt"
	echo "== status"
	git status --short | head -10
	echo "# end $(date -u +%FT%TZ)"
} > "$LOG" 2>&1
