#!/usr/bin/env bash
# Poll the registry until it serves mcp 0.0.31, or give up after about twenty minutes. Packuments lag an
# upload by minutes on this registry, so a 0.0.30 reading right after a successful publish is expected.
set -u
for attempt in $(seq 1 40); do
	version=$(npm view @orkestrel/mcp version 2>/dev/null)
	if [ "$version" = "0.0.31" ]; then
		echo "served 0.0.31 after ${attempt} reads"
		npm view @orkestrel/mcp dist.shasum
		exit 0
	fi
	sleep 30
done
echo "still serving $(npm view @orkestrel/mcp version 2>/dev/null) after forty reads"
exit 1
