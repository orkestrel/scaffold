function createParser() {
	let pending = ''
	return {
		parse(chunk) {
			pending += chunk
			const lines = pending.split('\n')
			pending = lines.pop() ?? ''
			return lines.filter((line) => line.trim().length > 0).map((line) => JSON.parse(line))
		},
		clear() { pending = '' },
	}
}

function createUpstream(deltas, holdMs) {
	const record = { started: 0, signals: [], abortedAt: undefined }
	async function* stream(_messages, signal) {
		record.started += 1
		record.signals.push(signal)
		signal.addEventListener('abort', () => { record.abortedAt = performance.now() }, { once: true })
		let content = ''
		for (const text of deltas) {
			if (signal.aborted) throw new Error('aborted before delta')
			content += text
			yield { channel: 'content', text }
			if (holdMs > 0) await new Promise((resolve) => setTimeout(resolve, holdMs))
		}
		return { content }
	}
	return {
		id: 'upstream', name: 'upstream', record,
		stream,
		async generate(messages, signal) { const g = stream(messages, signal); let s = await g.next(); while (!s.done) s = await g.next(); return s.value },
	}
}

const upstream = createUpstream(['relayed', ' answer'], 0)
const slow = createUpstream(['One,', ' two,', ' three'], 400)
const handler = createRelay({
	provider: upstream,
	authorize: (request) => request.headers.get('authorization') === `Bearer ${bearer}`,
})
const slowHandler = createRelay({ provider: slow, authorize: () => true })
const tight = createRelay({ provider: upstream, authorize: () => true, limit: 8 })
const dispatcher = createDispatcher({
	routes: [
		{ method: 'POST', path: '/relay', handler },
		{ method: 'POST', path: '/slow', handler: slowHandler },
		{ method: 'POST', path: '/tight', handler: tight },
	],
})
const server = createServer({ dispatcher, state: () => undefined, host: '127.0.0.1' })
const port = await server.start()
say(`started on 127.0.0.1:${port}; address=${JSON.stringify(server.address)}; DEFAULT_BODY_LIMIT=${DEFAULT_BODY_LIMIT}`)
process.on('SIGTERM', () => server.stop())
say(`SIGTERM listeners after the fence line: ${process.listenerCount('SIGTERM')}`)

const browser = createRelayProvider({
	url: `http://127.0.0.1:${port}/relay`,
	parser: createParser,
	headers: () => ({ authorization: `Bearer ${bearer}` }),
})
const messages = [{ id: '1', role: 'user', content: 'Say hello.' }]

// A. round trip over the socket

// C. 401 over the socket, upstream unentered
const wrong = createRelayProvider({ url: `http://127.0.0.1:${port}/relay`, parser: createParser, headers: () => ({ authorization: 'Bearer wrong' }) })

// D. 413 over the socket from the relay's own limit (8 bytes), not the server's

catch (error) { say(`D 413: code=${error.code} status=${error.status} message=${JSON.stringify(error.message)}; upstream.started=${upstream.record.started}`) }

// E. a browser-side cancel over the socket reaches the upstream signal
const slowBrowser = createRelayProvider({ url: `http://127.0.0.1:${port}/slow`, parser: createParser })
const abort = new AbortController()
const stream = slowBrowser.stream(messages, abort.signal)
const first = await stream.next()
const cancelledAt = performance.now()
abort.abort()
let settled
try { settled = await stream.next(); say(`E after abort: next -> done=${settled.done} value=${JSON.stringify(settled.value)}`) }
catch (error) { say(`E after abort: threw ${error.name} code=${error.code} partial=${JSON.stringify(error.partial)}`) }
for (let i = 0; i < 100 && slow.record.abortedAt === undefined; i += 1) await new Promise((resolve) => setTimeout(resolve, 10))

// F. stop() duration with a keep-alive client socket idle and the cancelled stream behind us

await server.stop()
say(`F stop() took ${(performance.now() - t0).toFixed(1)} ms; status=${server.status}; address=${JSON.stringify(server.address)}`)
