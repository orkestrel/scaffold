#!/usr/bin/env bash
# Orchestrator tracked command (U0f): rebuild and repack the ACCEPTED tool tree after U8c (the emitter),
# keeping the previous tarball; install tool alone into agent and tool+agent into the named consumers
# (mcp, ollama) through u0c-install-both.sh; refresh each consumer's guides/tool.md mirror by byte copy.
# Usage: bash u0f-repack-tool-after-u8c.sh [consumers...]   (default: agent ollama; pass mcp only when no unit writes there)
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
OUT="$SCAF/tmp/tarballs"
KEEP="$SCAF/tmp/tarballs-kept"
RECEIPT="$SCAF/tmp/units/u0f-receipt.md"
CONSUMERS="${*:-agent ollama}"
mkdir -p "$OUT" "$KEEP"
{
	echo "# U0f receipt — tool rebuilt and repacked after U8c"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "tool HEAD: $(git -C "$ROOT/tool" rev-parse --short HEAD)"
	echo "tool status: [$(git -C "$ROOT/tool" status --porcelain | tr '\n' ' ')]"
	echo "consumers: $CONSUMERS"
} > "$RECEIPT"
cd "$ROOT/tool" || exit 9
npm run build > "$SCAF/tmp/units/u0f-build.log.txt" 2>&1
echo "tool build exit: $?" >> "$RECEIPT"
PREV=$(ls -t "$OUT"/orkestrel-tool-*.tgz 2>/dev/null | head -1)
if [ -n "$PREV" ]; then cp "$PREV" "$KEEP/orkestrel-tool-u1d.tgz"; rm -f "$OUT"/orkestrel-tool-*.tgz; fi
npm pack --pack-destination "$OUT" > "$SCAF/tmp/units/u0f-pack.log.txt" 2>&1
echo "tool pack exit: $?" >> "$RECEIPT"
TARBALL=$(ls -t "$OUT"/orkestrel-tool-*.tgz | head -1)
echo "tarball: $TARBALL ($(stat -c %s "$TARBALL") bytes); previous kept as $KEEP/orkestrel-tool-u1d.tgz" >> "$RECEIPT"
echo "declaration mentions ToolManagerEventMap: $(tar -xOzf "$TARBALL" package/dist/src/core/index.d.ts | grep -c ToolManagerEventMap)" >> "$RECEIPT"
echo "runtime mentions emit(: $(tar -xOzf "$TARBALL" package/dist/src/core/index.js | grep -c 'emit(')" >> "$RECEIPT"
echo "--- install (tool alone into agent; tool+agent into the others)" >> "$RECEIPT"
bash "$SCAF/tmp/units/u0c-install-both.sh" $CONSUMERS >> "$SCAF/tmp/units/u0f-install.log.txt" 2>&1
for c in $CONSUMERS; do
	echo "$c: tool ToolManagerEventMap=$(grep -c ToolManagerEventMap "$ROOT/$c/node_modules/@orkestrel/tool/dist/src/core/index.d.ts" 2>/dev/null || echo absent) agent ToolContext=$(grep -c ToolContext "$ROOT/$c/node_modules/@orkestrel/agent/dist/src/core/index.d.ts" 2>/dev/null || echo absent)" >> "$RECEIPT"
	cp "$ROOT/tool/guides/tool.md" "$ROOT/$c/guides/tool.md"
	cmp "$ROOT/tool/guides/tool.md" "$ROOT/$c/guides/tool.md" && echo "$c: guides/tool.md mirror refreshed (cmp 0)" >> "$RECEIPT"
done
cat "$RECEIPT"
