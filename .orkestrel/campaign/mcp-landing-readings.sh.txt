#!/usr/bin/env bash
# The landing readings — Orchestrator tracked command, the authoritative sweep taken after the last writing
# unit exits and before the commit. Successor of a5c-evidence.sh with the landing's labels: the release-mode
# distribution proof, the gates, the projects the policy red short-circuits, and the teardown reading.
# Logs: scaffold tmp/units/mcp-landing-distribution.log.txt, gates-mcp-landing.log.txt, mcp-landing-tail.log.txt
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
M=/c/Users/mikes/WebstormProjects/mcp
cd "$M" || exit 9
{
	echo "# mcp landing distribution (release mode) — $(date -u +%FT%TZ) — head $(git rev-parse --short HEAD)"
	echo "# status: $(git status --short | tr '\n' ' ')"
	start=$(date +%s)
	npm run test:distribution -- --mode release > "$S/tmp/units/mcp-landing-distribution.out.txt" 2>&1
	code=$?
	end=$(date +%s)
	sed 's/\x1b\[[0-9;]*m//g' "$S/tmp/units/mcp-landing-distribution.out.txt" | grep -E 'Tests |Test Files|Duration|FAIL|Error|↓' | head -40
	echo "exit=$code wall=$((end - start))s"
	echo "# end $(date -u +%FT%TZ)"
} > "$S/tmp/units/mcp-landing-distribution.log.txt" 2>&1
cat "$S/tmp/units/mcp-landing-distribution.log.txt"
echo "== gates mcp landing"
bash "$S/tmp/units/gates3.sh" mcp landing
echo "== tail projects and teardown"
{
	echo "# mcp landing tail — $(date -u +%FT%TZ) — head $(git rev-parse --short HEAD)"
	echo "== teardown reading: distribution-* trees under the temp directory"
	ls -d /c/Users/mikes/AppData/Local/Temp/distribution-* 2>/dev/null | wc -l
	for project in config setup conformance integration guides; do
		echo "== npm run test:$project"
		npm run "test:$project" > "$S/tmp/units/mcp-landing-tail-$project.out.txt" 2>&1
		code=$?
		sed 's/\x1b\[[0-9;]*m//g' "$S/tmp/units/mcp-landing-tail-$project.out.txt" | grep -E 'Tests |Test Files|FAIL' | head -6
		echo "exit=$code"
	done
	echo "# end $(date -u +%FT%TZ)"
} > "$S/tmp/units/mcp-landing-tail.log.txt" 2>&1
cat "$S/tmp/units/mcp-landing-tail.log.txt"
