#!/usr/bin/env bash
# A4g objective audit lane — GPT-6 Astra, read-only codex exec rooted at the mcp checkout. Cap 2400 s.
# Successor of A4f-run.sh: unit name, brief, journal, evidence set (U4j report, A4f verdicts, P14, gates, probe, the interdiffs).
# report and its red/green logs, the P11 record and instrument, the A4b verdicts, the gate logs, the collision
# probe, and the G5 distillate beside the brief. Astra is the cross-engine lane (Opus wrote the chain incl. U4f).
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
J="$SCAF/tmp/codex/A4g-audit.jsonl"
L="$SCAF/tmp/codex/A4g-audit-last.md"
B="$MCP/tmp/codex/A4g-audit-brief.md"
cp "$SCAF/tmp/units/A4g-audit-brief.md" "$B"
cp "$SCAF/tmp/units/A4g-u4j-only.patch" "$MCP/tmp/codex/A4g-u4j-only.patch"
cp "$SCAF/tmp/units/A4f-u4i-only.patch" "$MCP/tmp/codex/A4f-u4i-only.patch"
cp "$SCAF/tmp/units/A4e-u4h-only.patch" "$MCP/tmp/codex/A4e-u4h-only.patch"
cp "$SCAF/tmp/units/A4d-u4g-only.patch" "$MCP/tmp/codex/A4d-u4g-only.patch"
cp "$SCAF/tmp/units/A4c-u4f-only.patch" "$MCP/tmp/codex/A4c-u4f-only.patch"
cp "$SCAF/tmp/units/A4g-diff.patch" "$MCP/tmp/codex/A4g-diff.patch"
for f in U4j-mcp-browser-fix-brief.md U4j-mcp-browser-report.md A4f-audit-analyst.md A4f-audit-reviewer.md A4f-audit-checker.md P14-a4f-probe.md U4j-mcp-gates-orchestrator.log.txt U4j-mcp-gates-test-full.log.txt collide3-mcp-after-u4j.txt; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f"
done
echo "launch unit=A4g-audit model=gpt-6-astra sandbox=read-only cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/A4g-audit.launch.txt"
timeout 2400 codex exec --json -C "$MCP" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (every evidence file it names is staged beside it in tmp/codex/). You hold the OBJECTIVE lane (analyst) and the cross-engine lane: Opus 5 wrote the unit under audit. Perform the audit directly and spawn nothing. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/A4g-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/A4g-audit.launch.txt"
echo "unit=A4g-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
