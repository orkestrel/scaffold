// Probe: cleanup on three adverse exits of AgentProvider.stream — an early `result` increment,
// a throwing `read`, and a consumer `return()` after the first delta. Observes reader cancellation
// (the source's cancel callback), parser clearing, and the deadline (recorded combined signal).
import { AgentProvider, isProviderAbortError, isProviderError } from 'file:///C:/Users/mikes/WebstormProjects/agent/dist/src/core/index.js'

const messages = [{ id: '1', role: 'user', content: 'hi' }]

function makeFixture(readImpl, log) {
	return class Fixture extends AgentProvider {
		name = 'fixture'
		frame() {
			return {
				parse: (chunk) => [chunk],
				clear: () => log.push('parser.clear'),
			}
		}
		body(request) {
			return { messages: request.messages }
		}
		read(record) {
			return readImpl(record)
		}
		finish() {
			log.push('finish')
			return []
		}
	}
}

function makeBody(chunks, log) {
	let index = 0
	return new ReadableStream({
		pull(controller) {
			if (index < chunks.length) {
				controller.enqueue(new TextEncoder().encode(chunks[index]))
				index += 1
				return
			}
			// Never close: the stream stays open so an early exit must cancel it.
			return new Promise(() => {})
		},
		cancel(reason) {
			log.push(`source.cancel(${reason === undefined ? 'undefined' : String(reason?.name ?? reason)})`)
		},
	})
}

function describe(error) {
	if (error === undefined) return 'no error'
	if (isProviderAbortError(error)) return `ProviderAbortError partial=${JSON.stringify(error.partial)}`
	if (isProviderError(error)) return `ProviderError code=${error.code} ${error.message}`
	return `OTHER ${error?.constructor?.name ?? typeof error}: ${error?.message ?? ''}`
}

async function settle(log, recorded) {
	await new Promise((resolve) => setTimeout(resolve, 80))
	return `log=[${log.join(', ')}] signalAbortedAfter80ms=${recorded.aborted}`
}

// Case F: a `result` increment arrives on the second record while the body stays open.
{
	const log = []
	let recorded
	const Fixture = makeFixture(
		(record) =>
			record === 'settle'
				? { content: '', thinking: '', tools: [], result: { content: 'AUTHORITATIVE', usage: { prompt: 1, completion: 2, total: 3 } } }
				: { content: record, thinking: '', tools: [] },
		log,
	)
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		strict: true,
		fetch: (_i, init) => {
			recorded = init.signal
			return Promise.resolve(new Response(makeBody(['first', 'settle', 'never-read'], log), { status: 200 }))
		},
	})
	const stream = provider.stream(messages, new AbortController().signal)
	const deltas = []
	let step = await stream.next()
	while (!step.done) {
		deltas.push(step.value.text)
		step = await stream.next()
	}
	console.log('F early result: deltas =', JSON.stringify(deltas), '; result =', JSON.stringify(step.value), ';', await settle(log, recorded))
}

// Case G: `read` throws a plain error on the second record.
{
	const log = []
	let recorded
	const Fixture = makeFixture((record) => {
		if (record === 'boom') throw new Error('hostile read')
		return { content: record, thinking: '', tools: [] }
	}, log)
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		fetch: (_i, init) => {
			recorded = init.signal
			return Promise.resolve(new Response(makeBody(['first', 'boom'], log), { status: 200 }))
		},
	})
	const stream = provider.stream(messages, new AbortController().signal)
	let error
	try {
		await stream.next()
		await stream.next()
	} catch (caught) {
		error = caught
	}
	console.log('G read throws:', describe(error), ';', await settle(log, recorded))
}

// Case H: the consumer calls return() after the first delta.
{
	const log = []
	let recorded
	const Fixture = makeFixture((record) => ({ content: record, thinking: '', tools: [] }), log)
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		fetch: (_i, init) => {
			recorded = init.signal
			return Promise.resolve(new Response(makeBody(['first', 'second'], log), { status: 200 }))
		},
	})
	const stream = provider.stream(messages, new AbortController().signal)
	const first = await stream.next()
	const returned = await stream.return({ content: 'returned' })
	console.log('H consumer return(): first =', JSON.stringify(first.value), '; return done =', returned.done, ';', await settle(log, recorded))
}

// Case I: strict with no result by end of input (body closes).
{
	const log = []
	let recorded
	const Fixture = makeFixture((record) => ({ content: record, thinking: '', tools: [] }), log)
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		strict: true,
		fetch: (_i, init) => {
			recorded = init.signal
			return Promise.resolve(new Response('only', { status: 200 }))
		},
	})
	let error
	try {
		const stream = provider.stream(messages, new AbortController().signal)
		let step = await stream.next()
		while (!step.done) step = await stream.next()
	} catch (caught) {
		error = caught
	}
	console.log('I strict without result:', describe(error), ';', await settle(log, recorded))
}
