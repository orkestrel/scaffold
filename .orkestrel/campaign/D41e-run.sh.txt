#!/usr/bin/env bash
# D4-1e implementation lane — GPT-6 Astra (sol route), workspace-write codex exec rooted at the scaffold checkout. Cap 2700 s.
# Successor of D41d-run.sh: unit name, brief, journal; stages the D4-1d brief, report, and partial diff beside the brief.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
U="D41e-scaffold-inventory-baseline"
J="$SCAF/tmp/codex/$U.jsonl"
L="$SCAF/tmp/codex/$U-last.md"
B="$SCAF/tmp/codex/$U-brief.md"
cp "$SCAF/tmp/units/$U-brief.md" "$B"
cp "$SCAF/tmp/units/D41d-scaffold-inventory-baseline-brief.md" "$SCAF/tmp/codex/D41d-scaffold-inventory-baseline-brief.md"
cp "$SCAF/.orkestrel/campaign/D41d-scaffold-inventory-baseline-report.md" "$SCAF/tmp/codex/D41d-scaffold-inventory-baseline-report.md"
cp "$SCAF/.orkestrel/campaign/D41d-partial-diff.patch.txt" "$SCAF/tmp/codex/D41d-partial-diff.patch.txt"
cp "$SCAF/.orkestrel/campaign/P6-surface-collisions.txt" "$SCAF/tmp/codex/P6-surface-collisions.txt"
echo "launch unit=$U model=gpt-6-astra sandbox=workspace-write cap=2700 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/$U.launch.txt"
timeout 2700 codex exec --json -C "$SCAF" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the unit brief at $B exactly; it amends the D4-1d brief staged beside it (read that one too, and the D4-1d report and partial diff). You hold the sol implementation route (GPT-6 Astra). Perform the assignment directly and spawn nothing; continue from the tree's partial state. Never touch .orkestrel/, .claude/, .agents/, dist/, package.json, tests/setupPolicy.ts, tests/policy.test.ts, src/bin/**, or any file the brief lists off-limits; host.json only for the seed. Your final message must be the Output shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/$U.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/$U.launch.txt"
echo "unit=$U exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
