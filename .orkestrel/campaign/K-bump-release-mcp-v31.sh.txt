#!/usr/bin/env bash
# Orchestrator tracked command, successor of bump-release-ollama-v17.sh: bump mcp to 0.0.31 (its published
# surface moved — the browser face and the server chain landed since 0.0.30), then run the release proof
# (release-chain.sh mcp v31: the package's own prepublishOnly, then a pack dry run).
# Run only after the scaffold 0.0.69 visit has cleared the standing policy red, because prepublishOnly runs
# `npm test`, whose chain stops at that project. Log: scaffold tmp/units/release-mcp-v31.log.txt
set -u
M=/c/Users/mikes/WebstormProjects/mcp
S=/c/Users/mikes/WebstormProjects/scaffold
cd "$M" || exit 9
echo "== self-version literals under src, tests, the guide, and the README"
grep -rn "0\.0\.30" src tests guides/mcp.md README.md 2>/dev/null | head
echo "== npm version 0.0.31 --no-git-tag-version"
npm version 0.0.31 --no-git-tag-version
echo "exit=$?"
git -C "$M" diff --stat
bash "$S/tmp/units/release-chain.sh" mcp v31
echo "== chain done"
cat "$S/tmp/units/release-mcp-v31.log.txt"
