# Orchestrator tracked command: stop the Ollama daemon this session started for the live proofs, by the recorded pid.
$dir = 'C:\Users\mikes\WebstormProjects\scaffold\tmp\ollama'
$pidFile = "$dir\serve.pid"
if (-not (Test-Path $pidFile)) {
	"no pid file at $pidFile; nothing stopped"
	exit 0
}
$id = [int](Get-Content $pidFile | Select-Object -First 1)
$proc = Get-Process -Id $id -ErrorAction SilentlyContinue
if ($null -eq $proc) {
	"pid $id is not running; nothing stopped"
} else {
	Stop-Process -Id $id
	"stopped ollama serve pid=$id"
}
