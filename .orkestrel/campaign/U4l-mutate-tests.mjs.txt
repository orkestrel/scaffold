// U4l controls: forces one expected value in a named test file so its assertion must fail.
// Usage: node u4l-mutate-tests.mjs <factories|validators|identity>
import { readFileSync, writeFileSync } from 'node:fs'

const root = 'C:/Users/mikes/WebstormProjects/mcp/tests/src/browser/'
const cases = {
	factories: {
		file: `${root}factories.test.ts`,
		before: "\t\texpect(bridge !== undefined).toBe('modelContext' in document)\n",
		after: "\t\texpect(bridge !== undefined).toBe(!('modelContext' in document))\n",
	},
	validators: {
		file: `${root}validators.test.ts`,
		before: "\t\texpect(isWebMCPDocument(document)).toBe('modelContext' in document)\n",
		after: "\t\texpect(isWebMCPDocument(document)).toBe(!('modelContext' in document))\n",
	},
	identity: {
		file: `${root}ModelContext.test.ts`,
		before: "\t\t\tfixture.registrations().find((entry) => entry.tool.name === 'kept'),\n\t\t\t'the registration the first publication made for the carried name',\n",
		after: "\t\t\tfixture.registrations().find((entry) => entry.tool.name === 'dropped'),\n\t\t\t'the registration the first publication made for the carried name',\n",
	},
}

const name = process.argv[2]
const chosen = cases[name]
if (chosen === undefined) throw new Error(`Unknown control '${name}'`)
const source = readFileSync(chosen.file, 'utf8')
const occurrences = source.split(chosen.before).length - 1
if (occurrences !== 1) throw new Error(`Expected one site for '${name}', matched ${occurrences}`)
writeFileSync(chosen.file, source.replace(chosen.before, chosen.after))
console.log(`forced the expected value for '${name}' in ${chosen.file}`)
