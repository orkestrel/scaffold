#!/usr/bin/env bash
# Unit O2 (tmp/units/o2-brief.md: OllamaProvider rebuilt on the AgentProvider base installed from the tarball packed at agent 5d288d7).
# Cap: 5400 s. Journal: scaffold/tmp/codex/o2.jsonl. Report: ollama/tmp/units/o2-report.md (and o2-last.md).
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
OLLAMA="C:/Users/mikes/WebstormProjects/ollama"
cd "$SCAF" || exit 9
mkdir -p "$OLLAMA/tmp/units"
cp "$SCAF/tmp/units/o2-brief.md" "$OLLAMA/tmp/units/o2-brief.md" || exit 8
cmp "$SCAF/tmp/units/o2-brief.md" "$OLLAMA/tmp/units/o2-brief.md" || exit 7
git -C "$ROOT/ollama" status --porcelain > "$SCAF/tmp/codex/o2.status-before.txt"
git -C "$ROOT/ollama" rev-parse --short HEAD >> "$SCAF/tmp/codex/o2.status-before.txt"
date -u +%FT%TZ > "$SCAF/tmp/codex/o2.launch.txt"
timeout 5400 codex exec --json -C "$OLLAMA" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$OLLAMA/tmp/units/o2-last.md" "Read tmp/units/o2-brief.md and execute it exactly. It is unit O2: OllamaProvider rebuilt on the installed AgentProvider base. Complete the whole unit. Your final message must be the report the brief specifies, and you also write that report to tmp/units/o2-report.md." < /dev/null > "$SCAF/tmp/codex/o2.jsonl" 2> "$SCAF/tmp/codex/o2.err"
echo "exit=$?" >> "$SCAF/tmp/codex/o2.launch.txt"
git -C "$ROOT/ollama" status --porcelain > "$SCAF/tmp/codex/o2.status-after.txt"
date -u +%FT%TZ >> "$SCAF/tmp/codex/o2.launch.txt"
