// Probe: does AgentProvider report a cancel as ProviderAbortError when the transport rejects with
// its own AbortError (not signal.reason)? Also: caller abort while awaiting body bytes, and
// deadline cleared after success. Runs against the built dist at commit cef565d.
import { AgentProvider, isProviderAbortError, isProviderError } from 'file:///C:/Users/mikes/WebstormProjects/agent/dist/src/core/index.js'

class Fixture extends AgentProvider {
	name = 'fixture'
	frame() {
		return { parse: (chunk) => [chunk], clear: () => {} }
	}
	body(request) {
		return { messages: request.messages }
	}
	read(record) {
		return { content: record, thinking: '', tools: [] }
	}
	finish() {
		return []
	}
}

const messages = [{ id: '1', role: 'user', content: 'hi' }]

async function drain(provider, signal) {
	const deltas = []
	const stream = provider.stream(messages, signal)
	try {
		let step = await stream.next()
		while (!step.done) {
			deltas.push(step.value)
			step = await stream.next()
		}
		return { deltas, result: step.value }
	} catch (error) {
		return { deltas, error }
	}
}

function describe(error) {
	if (error === undefined) return 'no error'
	if (isProviderAbortError(error)) return `ProviderAbortError partial=${JSON.stringify(error.partial)}`
	if (isProviderError(error)) return `ProviderError code=${error.code}`
	return `OTHER ${error?.constructor?.name ?? typeof error}: ${error?.name ?? ''} ${error?.message ?? ''}`
}

// Case A (control): a spec-conformant transport rejects with signal.reason on abort.
{
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		fetch: (_input, init) =>
			new Promise((_resolve, reject) => {
				init.signal.addEventListener('abort', () => reject(init.signal.reason), { once: true })
			}),
	})
	const outcome = await drain(provider, new AbortController().signal)
	console.log('A control (reject with signal.reason on deadline):', describe(outcome.error))
}

// Case B (attack): a transport rejects with its own AbortError DOMException on abort.
{
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 30,
		fetch: (_input, init) =>
			new Promise((_resolve, reject) => {
				init.signal.addEventListener(
					'abort',
					() => reject(new DOMException('The operation was aborted.', 'AbortError')),
					{ once: true },
				)
			}),
	})
	const outcome = await drain(provider, new AbortController().signal)
	console.log('B attack (reject with own AbortError on deadline):', describe(outcome.error))
}

// Case C: caller aborts while body bytes are awaited after one delta.
{
	let controllerRef
	const body = new ReadableStream({
		start(controller) {
			controllerRef = controller
			controller.enqueue(new TextEncoder().encode('first'))
		},
	})
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 5_000,
		fetch: () => Promise.resolve(new Response(body, { status: 200 })),
	})
	const abort = new AbortController()
	const stream = provider.stream(messages, abort.signal)
	const first = await stream.next()
	abort.abort()
	let error
	try {
		await stream.next()
	} catch (caught) {
		error = caught
	}
	console.log('C caller abort awaiting bytes: first delta =', JSON.stringify(first.value), '; then', describe(error))
	void controllerRef
}

// Case D: deadline cleared after success — the recorded combined signal must stay unaborted.
{
	let recorded
	const provider = new Fixture({
		url: 'http://x/',
		timeout: 20,
		fetch: (_input, init) => {
			recorded = init.signal
			return Promise.resolve(new Response('done', { status: 200 }))
		},
	})
	const outcome = await drain(provider, new AbortController().signal)
	await new Promise((resolve) => setTimeout(resolve, 60))
	console.log('D deadline cleared after success: result =', JSON.stringify(outcome.result), '; signal aborted after 60ms =', recorded.aborted)
}

// Case E: caller signal already aborted before the call.
{
	const provider = new Fixture({ url: 'http://x/', fetch: () => Promise.resolve(new Response('x')) })
	const controller = new AbortController()
	controller.abort()
	const outcome = await drain(provider, controller.signal)
	console.log('E pre-aborted:', describe(outcome.error))
}
