// Orchestrator probe P12 — reproduces the A4c analyst's unexecuted vectors (claims 1 and 2) in the
// real src:browser project. Run by explicit path; deleted after the reading is retained.
import { describe, expect, it } from 'vitest'
import { createPageServer } from '@src/browser'
import { isMCPError } from '@src/core'
import { createTool, createToolManager } from '@orkestrel/tool'
import { waitForDelay } from '@orkestrel/test'
import { createBridge, createDescribedTools } from '../../setupBrowser.js'

function outcome(promise: Promise<unknown>): Promise<string> {
	return Promise.race([
		promise.then(
			() => 'settled',
			(error: unknown) =>
				`rejected: ${isMCPError(error) ? error.code : error instanceof Error ? error.message : String(error)}`,
		),
		waitForDelay(1500).then(() => 'still pending after 1500 ms'),
	])
}

describe('P12 vectors', () => {
	it('claim 1: requests issued inside a connect listener after stop refuse rather than park', async () => {
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const page = createPageServer({ tools, client: { timeout: 1_000 } })
		let call: Promise<unknown> | undefined
		let task: Promise<unknown> | undefined
		let next: Promise<unknown> | undefined
		page.client.emitter.once('connect', () => {
			page.stop()
			call = page.client.call('echo', {})
			task = page.client.tasks.task('missing')
			next = page.client.listen(undefined, { signal: new AbortController().signal }).next()
		})
		const connected = await page.client.connect().then(
			() => 'resolved',
			(error: unknown) => `rejected: ${error instanceof Error ? error.message : String(error)}`,
		)
		if (call === undefined || task === undefined || next === undefined) {
			throw new Error('the connect listener did not run')
		}
		const readings = {
			connect: connected,
			call: await outcome(call),
			task: await outcome(task),
			next: await outcome(next),
		}
		console.log('P12 claim 1:', JSON.stringify(readings))
		expect(readings.call).toBe('rejected: -32600')
		expect(readings.task).toBe('rejected: -32600')
		expect(readings.next).toBe('rejected: -32600')
	})

	it('claim 2a: an addition on a manager a re-entrant publish replaced registers nothing', async () => {
		const { fixture, bridge } = createBridge()
		const first = createDescribedTools(5)
		const second = createToolManager()
		second.add(createTool({ name: 'lookup', description: 'Looks up', execute: () => 'second' }))
		// An earlier listener on the first manager republishes the second manager on any addition,
		// so the bridge's own listener on the first manager fires after its subscription was replaced.
		first.emitter.on('add', () => {
			void bridge.publish(second)
		})
		await bridge.publish(first)
		first.add(createTool({ name: 'subtract', description: 'Subtracts', execute: () => 1 }))
		await waitForDelay(200)
		const names = fixture.registrations().map((entry) => entry.tool.name)
		console.log('P12 claim 2a: registrations after re-entrant replacement =', JSON.stringify(names))
		expect(names).toEqual(['lookup'])
	})

	it('claim 2b: an addition made by an earlier clear listener survives the clear', async () => {
		const { fixture, bridge } = createBridge()
		const tools = createDescribedTools(5)
		tools.emitter.on('clear', () => {
			tools.add(createTool({ name: 'fresh', description: 'Fresh', execute: () => 1 }))
		})
		await bridge.publish(tools)
		tools.clear()
		await waitForDelay(200)
		const names = fixture.registrations().map((entry) => entry.tool.name)
		console.log(
			'P12 claim 2b: registrations after a clear whose earlier listener added fresh =',
			JSON.stringify(names),
		)
		expect(names).toEqual(['fresh'])
	})

	it('claim 2c: a remove then an add of the same name in one turn ends registered as the later tool', async () => {
		const { fixture, bridge } = createBridge()
		const tools = createDescribedTools(5)
		await bridge.publish(tools)
		tools.remove('add')
		tools.add(createTool({ name: 'add', description: 'Adds again', execute: () => 2 }))
		await waitForDelay(200)
		const registered = fixture.registrations()
		console.log(
			'P12 claim 2c:',
			JSON.stringify(registered.map((entry) => [entry.tool.name, entry.tool.description])),
		)
		expect(registered.map((entry) => entry.tool.name)).toEqual(['add'])
		expect(registered.map((entry) => entry.tool.description)).toEqual(['Adds again'])
	})

	it('claim 2d: a clear during a suspended publish ends empty', async () => {
		const { fixture, bridge } = createBridge()
		const tools = createDescribedTools(5)
		tools.add(createTool({ name: 'subtract', description: 'Subtracts', execute: () => 1 }))
		const resume = fixture.suspend()
		const pending = bridge.publish(tools)
		await waitForDelay(50)
		const held = fixture.holding()
		tools.clear()
		resume()
		await pending
		await waitForDelay(200)
		const names = fixture.registrations().map((entry) => entry.tool.name)
		console.log('P12 claim 2d: held', held, '| registrations after the clear =', JSON.stringify(names))
		expect(names).toEqual([])
	})
})
