// B2 receipt server: serves the built @orkestrel/ollama core entry and its @orkestrel closure to a
// Chromium page through an import map, and mounts createRelay at POST /relay over a server-side
// OllamaProvider pointed at the local daemon, behind a custom bearer. The page drives OllamaProvider
// directly against the daemon and through the relay.
import { createServer } from 'node:http'
import { readFile } from 'node:fs/promises'
import { Readable } from 'node:stream'
import { pathToFileURL } from 'node:url'

const OLLAMA = 'C:/Users/mikes/WebstormProjects/ollama'
const DAEMON = 'http://localhost:11434'
const MODEL = 'qwen3.5:2b-q4_K_M'
const PAGE = new URL('./b2.html', import.meta.url)
const PORT = 8766
const TOKEN = 'fixture-token'
const ollama = await import(pathToFileURL(`${OLLAMA}/dist/src/core/index.js`).href)
const agent = await import(pathToFileURL(`${OLLAMA}/node_modules/@orkestrel/agent/dist/src/core/index.js`).href)

const entryPath = (name) => name === 'ollama' ? `${OLLAMA}/dist/src/core/index.js` : `${OLLAMA}/node_modules/@orkestrel/${name}/dist/src/core/index.js`

async function closure() {
	const seen = new Map()
	const queue = ['ollama']
	while (queue.length > 0) {
		const name = queue.shift()
		if (seen.has(name)) continue
		const text = await readFile(entryPath(name), 'utf8')
		const imports = [...text.matchAll(/from "@orkestrel\/([a-z]+)"/g)].map((match) => match[1])
		seen.set(name, imports)
		queue.push(...imports)
	}
	return seen
}

const graph = await closure()
console.log('closure: ' + [...graph.keys()].join(' '))
const importmap = JSON.stringify({ imports: Object.fromEntries([...graph.keys()].map((name) => [`@orkestrel/${name}`, `/modules/${name}/index.js`])) })

const provider = ollama.createOllama({ model: MODEL, url: DAEMON, options: { num_predict: 24, temperature: 0 } })
const handler = agent.createRelay({
	provider,
	authorize: (request) => {
		const ok = request.headers.get('authorization') === 'Bearer ' + TOKEN
		console.log('relay authorize: ' + (ok ? 'accepted' : 'refused'))
		return ok
	},
})

function headersOf(req) {
	const entries = []
	for (const [key, value] of Object.entries(req.headers)) if (typeof value === 'string') entries.push([key, value])
	return entries
}

async function serveFile(res, path, type) {
	try {
		const text = await readFile(path, 'utf8')
		res.writeHead(200, { 'content-type': type, 'cache-control': 'no-store' })
		res.end(text)
	} catch (error) {
		res.writeHead(404, { 'content-type': 'text/plain' })
		res.end(String(error))
	}
}

const server = createServer(async (req, res) => {
	const url = new URL(req.url ?? '/', `http://localhost:${PORT}`)
	if (req.method === 'GET' && url.pathname === '/') {
		const page = (await readFile(PAGE, 'utf8')).replaceAll('IMPORT_MAP', importmap).replaceAll('MODEL_NAME', MODEL).replaceAll('DAEMON_URL', DAEMON)
		res.writeHead(200, { 'content-type': 'text/html; charset=utf-8', 'cache-control': 'no-store' })
		res.end(page)
		return
	}
	const module = /^\/modules\/([a-z]+)\/index\.js$/.exec(url.pathname)
	if (req.method === 'GET' && module !== null) {
		await serveFile(res, entryPath(module[1]), 'text/javascript; charset=utf-8')
		return
	}
	if (req.method === 'POST' && url.pathname === '/relay') {
		const controller = new AbortController()
		res.on('close', () => {
			if (!res.writableFinished) {
				console.log('relay: client connection closed before the response finished; aborting inbound')
				controller.abort(new Error('client disconnected'))
			}
		})
		const request = new Request(url, { method: 'POST', headers: headersOf(req), body: Readable.toWeb(req), duplex: 'half', signal: controller.signal })
		const response = await handler(request)
		console.log('relay response: ' + response.status)
		res.writeHead(response.status, Object.fromEntries(response.headers))
		if (response.body === null) {
			res.end()
			return
		}
		Readable.fromWeb(response.body).pipe(res)
		return
	}
	res.writeHead(404, { 'content-type': 'text/plain' })
	res.end('not found')
})

server.listen(PORT, '127.0.0.1', () => {
	console.log(`b2 server listening on http://127.0.0.1:${PORT}/`)
})
