// Orchestrator probe P10: read the mirrored supervisor guide with scaffold's installed @orkestrel/guide
// and report its Surface population, as the hosted-guide gate will.
import { readFileSync } from 'node:fs'
import { createGuide } from '@orkestrel/guide'

const path = process.argv[2]
const text = readFileSync(path, 'utf8')
const guide = createGuide(text)
const surface = guide.surface()
const names = surface.map((symbol) => symbol.name)
console.log(`guide: ${path}`)
console.log(`surface entries: ${surface.length}`)
console.log(`distinct names: ${new Set(names).size}`)
console.log(`first: ${names.slice(0, 8).join(', ')}`)
console.log(`methods groups: ${guide.methods().length}`)
