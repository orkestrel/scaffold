#!/usr/bin/env bash
# D4-1c implementation lane — GPT-6 Astra (sol route), workspace-write codex exec rooted at the scaffold checkout. Cap 2400 s.
# Successor of D43-run.sh: unit name, brief, journal; stages the D4-1b and D4-3 reports, the U14 report, and the P6 measurement beside the brief.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
U="D41c-scaffold-growth-refusal"
J="$SCAF/tmp/codex/$U.jsonl"
L="$SCAF/tmp/codex/$U-last.md"
B="$SCAF/tmp/codex/$U-brief.md"
cp "$SCAF/tmp/units/$U-brief.md" "$B"
cp "$SCAF/.orkestrel/campaign/D41b-scaffold-host-guides-report.md" "$SCAF/tmp/codex/D41b-scaffold-host-guides-report.md"
cp "$SCAF/.orkestrel/campaign/D43-scaffold-surface-rule-report.md" "$SCAF/tmp/codex/D43-scaffold-surface-rule-report.md"
cp "$SCAF/.orkestrel/campaign/U14-scaffold-setup-rename-report.md" "$SCAF/tmp/codex/U14-scaffold-setup-rename-report.md"
cp "$SCAF/.orkestrel/campaign/P6-surface-collisions.txt" "$SCAF/tmp/codex/P6-surface-collisions.txt"
echo "launch unit=$U model=gpt-6-astra sandbox=workspace-write cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/$U.launch.txt"
timeout 2400 codex exec --json -C "$SCAF" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the unit brief at $B exactly (the D4-1b, D4-3, and U14 reports and the P6 measurement are staged beside it). You hold the sol implementation route (GPT-6 Astra). Perform the assignment directly and spawn nothing. Never touch .orkestrel/, .claude/, .agents/, host.json, tests/setupPolicy.ts, tests/policy.test.ts, src/bin/**, or any file the brief lists off-limits. Your final message must be the Output shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/$U.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/$U.launch.txt"
echo "unit=$U exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
