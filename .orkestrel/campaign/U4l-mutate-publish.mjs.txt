// U4l carrier 1 control: flattens `#publish`'s try/finally to the reconcile-then-prune order
// (the U4j order), so the pin's prune assertion has an order that must break it.
import { readFileSync, writeFileSync } from 'node:fs'

const target = 'C:/Users/mikes/WebstormProjects/mcp/src/browser/ModelContext.ts'
const before = [
	'\t\tif (this.#destroyed) return\n',
	'\t\ttry {\n',
	'\t\t\tfor (const projection of projections) await this.#reconcile(tools, projection, options)\n',
	'\t\t} finally {\n',
	'\t\t\tthis.#prune(projections)\n',
	'\t\t}\n',
	'\t}\n',
].join('')
const after = [
	'\t\tif (this.#destroyed) return\n',
	'\t\tfor (const projection of projections) await this.#reconcile(tools, projection, options)\n',
	'\t\tthis.#prune(projections)\n',
	'\t}\n',
].join('')

const source = readFileSync(target, 'utf8')
const occurrences = source.split(before).length - 1
if (occurrences !== 1) throw new Error(`Expected one #publish block, matched ${occurrences}`)
writeFileSync(target, source.replace(before, after))
console.log('mutated #publish to the reconcile-then-prune order')
