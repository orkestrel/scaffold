// Orchestrator tracked instrument (K-scaffold-prep): re-establish the committed surface inventory
// after the hosted agent and ollama guide mirrors were refreshed from their pushed main. The seed
// inventory was measured against a stale agent mirror, so the refreshed floor carries collisions
// that predate the rule but were never recorded; `stageHost` refuses growth against a present
// inventory and establishes only against an absent one, so this removes host.json, establishes
// into a temporary host, writes the manifest back, and prints exactly what the record admitted or
// dropped. `npm run build` (build:inventory) canonicalises the file afterwards.
import { mkdtempSync, readFileSync, rmSync, unlinkSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { pathToFileURL } from 'node:url'

const checkout = 'C:/Users/mikes/WebstormProjects/scaffold'
const inventory = join(checkout, 'host.json')
const server = await import(pathToFileURL(join(checkout, 'dist/src/server/index.js')).href)
const before = JSON.parse(readFileSync(inventory, 'utf8'))
const temporary = mkdtempSync(join(tmpdir(), 'orkestrel-reestablish-'))
unlinkSync(inventory)
try {
	server.stageHost(checkout, temporary, { establish: true, report: (message) => console.error(message) })
	const manifest = JSON.parse(readFileSync(join(temporary, 'manifest.json'), 'utf8'))
	writeFileSync(inventory, `${JSON.stringify(manifest, null, '\t')}\n`)
	const was = new Map(before.surface.map((entry) => [entry.name, entry.owners]))
	const now = new Map(manifest.surface.map((entry) => [entry.name, entry.owners]))
	for (const [name, owners] of now) {
		const record = was.get(name)
		if (record === undefined || !owners.every((owner) => record.includes(owner))) {
			console.log(`admitted: ${name} (${owners.join(', ')})${record === undefined ? '' : ` was (${record.join(', ')})`}`)
		}
	}
	for (const [name, owners] of was) {
		if (!now.has(name)) console.log(`dropped: ${name} (${owners.join(', ')})`)
	}
	console.log(`surface: ${before.surface.length} -> ${manifest.surface.length}; digest ${before.digest.slice(0, 16)} -> ${manifest.digest.slice(0, 16)}`)
} catch (error) {
	writeFileSync(inventory, `${JSON.stringify(before, null, '\t')}\n`)
	console.error(`restored the prior inventory after a failure: ${error instanceof Error ? error.message : String(error)}`)
	process.exitCode = 1
} finally {
	rmSync(temporary, { recursive: true, force: true })
}
