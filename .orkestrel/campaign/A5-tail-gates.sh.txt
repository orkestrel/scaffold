#!/usr/bin/env bash
# A5 tail gates — Orchestrator tracked command (the A5 checker's claim-10 vector): the mcp test projects that
# `npm run test` never reached after the policy project's standing red short-circuited the chain — config, setup,
# conformance, integration — each run directly; plus the distribution teardown reading (no distribution-* scratch
# tree under the temp directory). Log: scaffold tmp/units/A5-tail-gates.log.txt
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
M=/c/Users/mikes/WebstormProjects/mcp
LOG="$S/tmp/units/A5-tail-gates.log.txt"
cd "$M" || exit 9
{
echo "# A5 tail gates — $(date -u +%FT%TZ) — mcp head $(git rev-parse --short HEAD)"
echo "== teardown reading: distribution-* trees under the temp directory"
ls -d /c/Users/mikes/AppData/Local/Temp/distribution-* 2>/dev/null | wc -l
for project in config setup conformance integration; do
	echo "== npm run test:$project"
	npm run "test:$project" > "$S/tmp/units/A5-tail-gates-$project.out.txt" 2>&1
	code=$?
	sed 's/\x1b\[[0-9;]*m//g' "$S/tmp/units/A5-tail-gates-$project.out.txt" | grep -E 'Tests |Test Files|FAIL' | head -6
	echo "exit=$code"
done
echo "# end $(date -u +%FT%TZ)"
} > "$LOG" 2>&1
cat "$LOG"
