#!/usr/bin/env bash
# Orchestrator instrument for unit A5 (settles audit claim 7): the guides project's relay cases, with their 1000 ms
# stop bound, read under host contention — three sequential runs of `npm run test:guides` while `npm run build`
# runs beside each in the same checkout — plus the same three runs idle. Run after A5-fix lands (the tree clean).
set -u
ROOT=/c/Users/mikes/WebstormProjects
AGENT="$ROOT/agent"
LOG="$ROOT/scaffold/tmp/units/a5-contended.log.txt"
cd "$AGENT" || exit 9
{
	echo "# A5 contended reading — guides project beside a build, then idle"
	echo "date: $(date -u +%FT%TZ)"
	echo "HEAD: $(git rev-parse --short HEAD); status: [$(git status --porcelain | tr '\n' ' ')]"
	for i in 1 2 3; do
		npm run build > /dev/null 2>&1 &
		BUILD=$!
		START=$(date +%s%N)
		npm run test:guides > "$ROOT/scaffold/tmp/units/a5-contended-run-$i.txt" 2>&1
		EXIT=$?
		END=$(date +%s%N)
		wait $BUILD
		echo "contended run $i: exit=$EXIT wall=$(( (END - START) / 1000000 )) ms; $(grep -E 'Tests ' "$ROOT/scaffold/tmp/units/a5-contended-run-$i.txt" | sed 's/\x1b\[[0-9;]*m//g' | tr -s ' ')"
	done
	for i in 1 2 3; do
		START=$(date +%s%N)
		npm run test:guides > "$ROOT/scaffold/tmp/units/a5-idle-run-$i.txt" 2>&1
		EXIT=$?
		END=$(date +%s%N)
		echo "idle run $i: exit=$EXIT wall=$(( (END - START) / 1000000 )) ms; $(grep -E 'Tests ' "$ROOT/scaffold/tmp/units/a5-idle-run-$i.txt" | sed 's/\x1b\[[0-9;]*m//g' | tr -s ' ')"
	done
	echo "status after: [$(git status --porcelain | tr '\n' ' ')]"
} > "$LOG" 2>&1
cat "$LOG"
