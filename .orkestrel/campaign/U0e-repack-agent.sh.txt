#!/usr/bin/env bash
# Orchestrator tracked command (U0e): rebuild and repack the ACCEPTED agent tree, keeping the previous tarball,
# then install tool AND agent together into mcp and ollama through u0c-install-both.sh (never the agent alone —
# a lone agent install re-resolves tool from the registry, the U0b clobber).
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
OUT="$SCAF/tmp/tarballs"
KEEP="$SCAF/tmp/tarballs-kept"
RECEIPT="$SCAF/tmp/units/u0e-receipt.md"
mkdir -p "$OUT" "$KEEP"
{
	echo "# U0e receipt — agent rebuilt and repacked after U2d"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "agent HEAD: $(git -C "$ROOT/agent" rev-parse --short HEAD)"
	echo "agent status: [$(git -C "$ROOT/agent" status --porcelain | tr '\n' ' ')]"
} > "$RECEIPT"
cd "$ROOT/agent" || exit 9
npm run build > "$SCAF/tmp/units/u0e-build.log.txt" 2>&1
echo "agent build exit: $?" >> "$RECEIPT"
PREV=$(ls -t "$OUT"/orkestrel-agent-*.tgz 2>/dev/null | head -1)
if [ -n "$PREV" ]; then cp "$PREV" "$KEEP/orkestrel-agent-u2b.tgz"; rm -f "$OUT"/orkestrel-agent-*.tgz; fi
npm pack --pack-destination "$OUT" > "$SCAF/tmp/units/u0e-pack.log.txt" 2>&1
echo "agent pack exit: $?" >> "$RECEIPT"
TARBALL=$(ls -t "$OUT"/orkestrel-agent-*.tgz | head -1)
echo "tarball: $TARBALL ($(stat -c %s "$TARBALL") bytes); previous kept as $KEEP/orkestrel-agent-u2b.tgz" >> "$RECEIPT"
echo "runtime mentions the deny-listener guard order (authorize before append): $(tar -xOzf "$TARBALL" package/dist/src/core/index.js | grep -c 'ToolContext\|signal' )" >> "$RECEIPT"
echo "--- install tool+agent together into mcp and ollama" >> "$RECEIPT"
bash "$SCAF/tmp/units/u0c-install-both.sh" mcp ollama >> "$SCAF/tmp/units/u0e-install.log.txt" 2>&1
for c in mcp ollama; do
	echo "$c: tool ToolErrorContext=$(grep -c ToolErrorContext "$ROOT/$c/node_modules/@orkestrel/tool/dist/src/core/index.d.ts") agent ToolContext=$(grep -c ToolContext "$ROOT/$c/node_modules/@orkestrel/agent/dist/src/core/index.d.ts")" >> "$RECEIPT"
done
cat "$RECEIPT"
