// Orchestrator probe P13 — reproduces the A4d analyst's unexecuted vectors (claims 2 and 4) in the
// real src:browser project. Run by explicit path; deleted after the reading is retained.
import { describe, expect, it } from 'vitest'
import { createPageServer } from '@src/browser'
import { isMCPError } from '@src/core'
import { createTool, createToolManager } from '@orkestrel/tool'
import { waitForDelay } from '@orkestrel/test'
import { createBridge, createDescribedTools } from '../../setupBrowser.js'

function outcome(promise: Promise<unknown>): Promise<string> {
	const started = performance.now()
	return Promise.race([
		promise.then(
			() => `settled after ${Math.round(performance.now() - started)} ms`,
			(error: unknown) =>
				`rejected after ${Math.round(performance.now() - started)} ms: ${isMCPError(error) ? error.code : error instanceof Error ? error.message : String(error)}`,
		),
		waitForDelay(3000).then(() => 'still pending after 3000 ms'),
	])
}

describe('P13 vectors', () => {
	it('claim 2: requests issued inside a connect listener after disconnect settle before the deadline', async () => {
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const page = createPageServer({ tools, client: { timeout: 2_000 } })
		let call: Promise<unknown> | undefined
		let task: Promise<unknown> | undefined
		let next: Promise<unknown> | undefined
		let disconnected: Promise<unknown> | undefined
		page.client.emitter.once('connect', () => {
			disconnected = page.client.disconnect()
			call = page.client.call('echo', {})
			task = page.client.tasks.task('missing')
			next = page.client.listen(undefined, { signal: new AbortController().signal }).next()
		})
		const connected = await page.client.connect().then(
			() => 'resolved',
			(error: unknown) => `rejected: ${error instanceof Error ? error.message : String(error)}`,
		)
		if (call === undefined || task === undefined || next === undefined || disconnected === undefined) {
			throw new Error('the connect listener did not run')
		}
		const readings = {
			connect: connected,
			disconnect: await outcome(disconnected),
			call: await outcome(call),
			task: await outcome(task),
			next: await outcome(next),
		}
		console.log('P13 claim 2:', JSON.stringify(readings))
		page.stop()
		expect(readings.call).not.toBe('still pending after 3000 ms')
		expect(readings.task).not.toBe('still pending after 3000 ms')
		expect(readings.next).not.toBe('still pending after 3000 ms')
		expect(readings.call).not.toMatch(/timed out/)
		expect(readings.task).not.toMatch(/timed out/)
	})

	it('claim 4: a same-name addition made by an earlier clear listener survives the clear', async () => {
		const { fixture, bridge } = createBridge()
		const tools = createDescribedTools(5)
		tools.emitter.on('clear', () => {
			tools.add(createTool({ name: 'add', description: 'Adds again', execute: () => 2 }))
		})
		await bridge.publish(tools)
		tools.clear()
		await waitForDelay(200)
		const registered = fixture.registrations()
		console.log(
			'P13 claim 4:',
			JSON.stringify(registered.map((entry) => [entry.tool.name, entry.tool.description])),
		)
		expect(registered.map((entry) => entry.tool.name)).toEqual(['add'])
		expect(registered.map((entry) => entry.tool.description)).toEqual(['Adds again'])
	})
})
