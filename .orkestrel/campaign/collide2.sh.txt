#!/usr/bin/env bash
# Orchestrator probe (successor of collide.sh; the A4 and A8 lanes read the first form as a bare count):
# name collisions between one checkout's own `export function|const|class` declarations in src/** and
# tests/** and the exports of the installed @orkestrel/test (core, browser, server) and @orkestrel/contract.
# Prints the pattern, the paths swept, every hit, and an explicit "collisions: none" line when clean.
# Read-only. Usage: bash collide2.sh <checkout>
set -u
C="${1:?checkout}"
ROOT=/c/Users/mikes/WebstormProjects
NM="$ROOT/$C/node_modules/@orkestrel"
LIST=$(mktemp)
for f in "$NM"/test/dist/src/core/index.d.ts "$NM"/test/dist/src/browser/index.d.ts "$NM"/test/dist/src/server/index.d.ts "$NM"/contract/dist/src/core/index.d.ts; do
	[ -f "$f" ] || continue
	grep -o 'export declare \(function\|const\|class\) [A-Za-z_][A-Za-z0-9_]*' "$f" | awk '{print $4}'
done | sort -u > "$LIST"
echo "probe: export-name collisions in $C"
echo "date: $(date -u +%FT%TZ)"
echo "installed: test $(grep -o '"version": *"[^"]*"' "$NM/test/package.json" | head -1), contract $(grep -o '"version": *"[^"]*"' "$NM/contract/package.json" | head -1); export names read: $(wc -l < "$LIST") (declaration files: test core, test browser, test server, contract core)"
echo "pattern: ^export (async )?(function|const|class) NAME\\b for each installed NAME"
echo "paths: $ROOT/$C/src/** and $ROOT/$C/tests/** (node_modules excluded)"
files=$(find "$ROOT/$C/src" "$ROOT/$C/tests" -type f -name '*.ts' 2>/dev/null | wc -l)
echo "files swept: $files"
hits=0
while read -r name; do
	[ -n "$name" ] || continue
	while read -r line; do
		[ -n "$line" ] || continue
		hits=$((hits + 1))
		echo "  collision: $name at ${line#$ROOT/$C/}"
	done < <(grep -rn -E "^export (async )?(function|const|class) $name\b" "$ROOT/$C/src" "$ROOT/$C/tests" 2>/dev/null | grep -v node_modules)
done < "$LIST"
if [ "$hits" -eq 0 ]; then echo "collisions: none"; else echo "collisions: $hits"; fi
rm -f "$LIST"
