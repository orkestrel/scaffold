#!/usr/bin/env bash
# P19 — Orchestrator probe: reproduce U4l's four red readings on the host with the unit's retained
# instruments (u4l-mutate-publish.mjs, u4l-mutate-tests.mjs), restoring each byte copy and checking
# its SHA-256, then one green run over the three owned test files. Log: p19-run.log.txt beside this file.
set -u
SP="/c/Users/mikes/AppData/Local/Temp/claude/C--Users-mikes-WebstormProjects-scaffold/bab586ba-1c34-4a7e-a280-b3a7e9f303d2/scratchpad"
M=/c/Users/mikes/WebstormProjects/mcp
T=$M/tests/src/browser
LOG="$SP/p19-run.log.txt"
PIN='prunes a failed publication and keeps the names it carries'
FAC='builds a bridge exactly where this page exposes the registry'
VAL='detects the registry exactly where this page exposes the property'

control() {
	# $1 label, $2 file, $3 mutation command, $4 test file (relative), $5 test name
	local label="$1" file="$2" mutate="$3" spec="$4" name="$5"
	cp "$file" "$SP/p19-$label.bak"
	echo "== $label: before $(sha256sum "$file" | cut -c1-16)"
	$mutate || { echo "mutation failed"; cp "$SP/p19-$label.bak" "$file"; return; }
	echo "   mutated $(sha256sum "$file" | cut -c1-16)"
	echo "   npm run test:src:browser -- $spec -t \"$name\""
	npm run test:src:browser -- "$spec" -t "$name" > "$SP/p19-$label.out.txt" 2>&1
	echo "   exit=$?"
	sed 's/\x1b\[[0-9;]*m//g' "$SP/p19-$label.out.txt" | grep -E "Tests |AssertionError" | head -3 | sed 's/^/   /'
	cp "$SP/p19-$label.bak" "$file"
	echo "   restored $(sha256sum "$file" | cut -c1-16)"
}

{
	echo "# P19 — $(date -u +%FT%TZ) — U4l controls replayed on the host"
	cd "$M" || exit 9
	control publish "$M/src/browser/ModelContext.ts" "node $SP/u4l-mutate-publish.mjs" tests/src/browser/ModelContext.test.ts "$PIN"
	control identity "$T/ModelContext.test.ts" "node $SP/u4l-mutate-tests.mjs identity" tests/src/browser/ModelContext.test.ts "$PIN"
	control factories "$T/factories.test.ts" "node $SP/u4l-mutate-tests.mjs factories" tests/src/browser/factories.test.ts "$FAC"
	control validators "$T/validators.test.ts" "node $SP/u4l-mutate-tests.mjs validators" tests/src/browser/validators.test.ts "$VAL"
	echo "== green: npm run test:src:browser -- the three owned files, restored"
	npm run test:src:browser -- tests/src/browser/ModelContext.test.ts tests/src/browser/factories.test.ts tests/src/browser/validators.test.ts > "$SP/p19-green.out.txt" 2>&1
	echo "   exit=$?"
	sed 's/\x1b\[[0-9;]*m//g' "$SP/p19-green.out.txt" | grep -E "Tests |Test Files" | head -2 | sed 's/^/   /'
	echo "   tree: $(git -C "$M" status --short --untracked-files=no | wc -l) tracked paths"
	echo "# end $(date -u +%FT%TZ)"
} > "$LOG" 2>&1
