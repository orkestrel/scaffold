#!/usr/bin/env bash
# D4-6 implementation lane — GPT-6 Astra (sol route), workspace-write codex exec rooted at the scaffold checkout. Cap 2700 s.
# Stages the AD4 verdicts and the D4-3 / D4-1e reports beside the brief.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
U="D46-scaffold-shape-fixes"
J="$SCAF/tmp/codex/$U.jsonl"
L="$SCAF/tmp/codex/$U-last.md"
B="$SCAF/tmp/codex/$U-brief.md"
cp "$SCAF/tmp/units/$U-brief.md" "$B"
for f in AD4-audit-analyst.md AD4-audit-reviewer.md AD4-audit-checker.md D45-scaffold-audit-fixes-report.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$SCAF/tmp/codex/$f"
done
echo "launch unit=$U model=gpt-6-astra sandbox=workspace-write cap=2700 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/$U.launch.txt"
timeout 2700 codex exec --json -C "$SCAF" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the unit brief at $B exactly (the AD4 verdicts and the D4-5 report are staged beside it). You hold the sol implementation route (GPT-6 Astra). Perform the assignment directly and spawn nothing. Never touch .orkestrel/, .claude/, .agents/, dist/, host.json, package-lock.json, src/server/helpers.ts, src/server/types.ts, tests/setupPolicy.ts, package.json, or any file the brief lists off-limits. Your final message must be the Output shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/$U.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/$U.launch.txt"
echo "unit=$U exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
