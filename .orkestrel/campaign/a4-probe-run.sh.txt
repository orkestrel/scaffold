#!/usr/bin/env bash
# Shakedown of run-a4-pack.sh (first use of its flags): the same chain against the audit worktree at 0fa4090,
# packing into a separate directory and writing probe-named logs. The real A4 run follows A2-fix.
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
SRC="$ROOT/agent-audit"
OUT="$SCAF/tmp/tarballs-probe"
mkdir -p "$OUT"
RECEIPT="$SCAF/tmp/units/a4-probe-receipt.md"
{
	echo "# A4 probe receipt — agent tarball from the audit worktree into ollama"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "source HEAD: $(git -C "$SRC" rev-parse --short HEAD)"
	echo "source status: [$(git -C "$SRC" status --porcelain | tr '\n' ' ')]"
	echo "ollama HEAD: $(git -C "$ROOT/ollama" rev-parse --short HEAD)"
	echo "ollama declared range before: $(grep '"@orkestrel/agent"' "$ROOT/ollama/package.json")"
	echo "ollama installed before: $(grep '"version"' "$ROOT/ollama/node_modules/@orkestrel/agent/package.json")"
} > "$RECEIPT"
cd "$SRC" || exit 9
npm run build > "$SCAF/tmp/units/a4-probe-build.log.txt" 2>&1
echo "source build exit: $?" >> "$RECEIPT"
npm pack --pack-destination "$OUT" > "$SCAF/tmp/units/a4-probe-pack.log.txt" 2>&1
echo "source pack exit: $?" >> "$RECEIPT"
TARBALL=$(ls -t "$OUT"/orkestrel-agent-*.tgz | head -1)
echo "tarball: $TARBALL ($(stat -c %s "$TARBALL") bytes)" >> "$RECEIPT"
tar -tzf "$TARBALL" | grep -c "dist/src/core" >> "$RECEIPT"
cd "$ROOT/ollama" || exit 9
npm install --no-save --ignore-scripts "$TARBALL" > "$SCAF/tmp/units/a4-probe-install.log.txt" 2>&1
echo "ollama install exit: $?" >> "$RECEIPT"
echo "ollama declared range after: $(grep '"@orkestrel/agent"' "$ROOT/ollama/package.json")" >> "$RECEIPT"
echo "ollama installed after: $(grep '"version"' "$ROOT/ollama/node_modules/@orkestrel/agent/package.json")" >> "$RECEIPT"
echo "installed declaration carries AgentProvider: $(grep -c 'AgentProvider' "$ROOT/ollama/node_modules/@orkestrel/agent/dist/src/core/index.d.ts")" >> "$RECEIPT"
echo "installed declaration carries RelayProvider: $(grep -c 'RelayProvider' "$ROOT/ollama/node_modules/@orkestrel/agent/dist/src/core/index.d.ts")" >> "$RECEIPT"
echo "ollama package.json/lock dirty after install: [$(git -C "$ROOT/ollama" status --porcelain -- package.json package-lock.json | tr '\n' ' ')]" >> "$RECEIPT"
cat "$RECEIPT"
