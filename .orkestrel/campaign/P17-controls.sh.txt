#!/usr/bin/env bash
# Orchestrator probe P17 — replays three writer-only live mutation controls (U13g ×2, U13e ×1) on byte
# copies of tests/setupServer.ts in the ollama checkout, restoring after each; the page suite runs
# in the real Edge against the live daemon.
set -u
S="C:/Users/mikes/AppData/Local/Temp/claude/C--Users-mikes-WebstormProjects-scaffold/bab586ba-1c34-4a7e-a280-b3a7e9f303d2/scratchpad"
L="/c/Users/mikes/WebstormProjects/scaffold/tmp/units"
cd /c/Users/mikes/WebstormProjects/ollama || exit 9
F=tests/setupServer.ts
cp "$F" "$S/setupServer.ts.p17-backup"
echo "before: $(sha256sum "$F" | cut -c1-16)"
run_control() {
	local label="$1" mutation="$2" title="$3" anchor="${4:-}"
	echo "== $label: $mutation"
	if [ -n "$anchor" ]; then node "$S/p17-mutate.mjs" "$F" "$mutation" "$anchor"; else node "$S/p17-mutate.mjs" "$F" "$mutation"; fi
	grep -c 'P17-' "$F"
	npx vitest run --config vite.config.ts --no-cache --reporter=verbose --project service tests/service/page.test.ts -t "$title" > "$L/P17-$label.log.txt" 2>&1
	echo "$label exit=$?"
	sed 's/\x1b\[[0-9;]*m//g' "$L/P17-$label.log.txt" | grep -E '✓|×|Tests  |AssertionError|Error:' | cut -c1-200
	cp "$S/setupServer.ts.p17-backup" "$F"
	echo "restored: $(sha256sum "$F" | cut -c1-16)"
}
run_control a-losing-release drop-losing-release "releases the acquisition its deadline outran instead of waiting on it"
run_control b-post-release-read drop-post-release-read "refuses an attempt whose release crossed its allowance"
run_control c-turn-off-by-one turn-off-by-one "executes a page tool through an agent over a live Ollama relay" "						tool: (call, result) => void tools.push({ turn: turns.at(-1), call, result }),"
grep -c 'P17-' "$F"
git status --short | grep -c probe
