#!/usr/bin/env bash
# Orchestrator tracked command (U0c): install the staged tool AND agent tarballs into each consumer in ONE
# npm install, so the resolver keeps both (a separate agent install re-resolves tool from the registry — the
# U0b clobber). Uses the newest tarballs in tmp/tarballs; never links; records a receipt. Re-runnable.
# Usage: bash u0c-install-both.sh [consumers...]   (default: mcp ollama; pass "agent" to refresh tool there too)
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
OUT="$SCAF/tmp/tarballs"
RECEIPT="$SCAF/tmp/units/u0c-receipt.md"
TOOL=$(ls -t "$OUT"/orkestrel-tool-*.tgz 2>/dev/null | head -1)
AGENT=$(ls -t "$OUT"/orkestrel-agent-*.tgz 2>/dev/null | head -1)
CONSUMERS="${*:-mcp ollama}"
{
	echo "# U0c receipt — tool and agent tarballs installed together"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "tool tarball: $TOOL ($(stat -c %s "$TOOL") bytes)"
	echo "agent tarball: $AGENT ($(stat -c %s "$AGENT") bytes)"
	echo "consumers: $CONSUMERS"
} > "$RECEIPT"
for c in $CONSUMERS; do
	cd "$ROOT/$c" || { echo "## $c: missing" >> "$RECEIPT"; continue; }
	{
		echo
		echo "## $c"
		echo "tool ToolContext before: $(grep -c ToolContext node_modules/@orkestrel/tool/dist/src/core/index.d.ts 2>/dev/null || echo absent)"
		echo "agent ToolContext before: $(grep -c ToolContext node_modules/@orkestrel/agent/dist/src/core/index.d.ts 2>/dev/null || echo absent)"
	} >> "$RECEIPT"
	if [ "$c" = "agent" ]; then
		npm install --no-save --ignore-scripts --no-audit --no-fund "$TOOL" > "$SCAF/tmp/units/u0c-install-$c.log.txt" 2>&1
	else
		npm install --no-save --ignore-scripts --no-audit --no-fund "$TOOL" "$AGENT" > "$SCAF/tmp/units/u0c-install-$c.log.txt" 2>&1
	fi
	{
		echo "install exit: $?"
		echo "tool ToolContext after: $(grep -c ToolContext node_modules/@orkestrel/tool/dist/src/core/index.d.ts 2>/dev/null || echo absent)"
		echo "agent ToolContext after: $(grep -c ToolContext node_modules/@orkestrel/agent/dist/src/core/index.d.ts 2>/dev/null || echo absent)"
		echo "manifest/lock dirty: [$(git status --porcelain -- package.json package-lock.json | tr '\n' ' ')]"
	} >> "$RECEIPT"
done
cat "$RECEIPT"
