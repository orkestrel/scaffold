#!/usr/bin/env bash
# Unit A4-4 (Orchestrator tracked command): rebuild and repack the agent tree at the A5 commit, install the tarball
# into ollama (never link), record the replaced range, and prove the declaration moved. Successor of a4-3-run.sh.txt:
# A5 changed one TSDoc line in src/core/factories.ts (the relay fence's stop comment), so dist and the tarball move.
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
OUT="$SCAF/tmp/tarballs"
mkdir -p "$OUT"
RECEIPT="$SCAF/tmp/units/a4-4-receipt.md"
PHRASE="refuse new connections, drain, then close"
{
	echo "# A4-4 receipt — agent tarball into ollama after A5"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "agent HEAD: $(git -C "$ROOT/agent" rev-parse --short HEAD)"
	echo "agent status: [$(git -C "$ROOT/agent" status --porcelain | tr '\n' ' ')]"
	echo "ollama HEAD: $(git -C "$ROOT/ollama" rev-parse --short HEAD)"
	echo "ollama declared range before: $(grep '"@orkestrel/agent"' "$ROOT/ollama/package.json")"
	echo "ollama installed before: $(grep '"version"' "$ROOT/ollama/node_modules/@orkestrel/agent/package.json")"
	echo "installed declaration carries the corrected stop comment before: $(grep -c "$PHRASE" "$ROOT/ollama/node_modules/@orkestrel/agent/dist/src/core/index.d.ts")"
} > "$RECEIPT"
cd "$ROOT/agent" || exit 9
npm run build > "$SCAF/tmp/units/a4-4-build.log.txt" 2>&1
echo "agent build exit: $?" >> "$RECEIPT"
echo "built declaration carries the corrected stop comment: $(grep -c "$PHRASE" "$ROOT/agent/dist/src/core/index.d.ts")" >> "$RECEIPT"
KEPT="$SCAF/tmp/tarballs-probe/orkestrel-agent-a4-3.tgz"
mkdir -p "$SCAF/tmp/tarballs-probe"
cp "$(ls -t "$OUT"/orkestrel-agent-*.tgz | head -1)" "$KEPT"
rm -f "$OUT"/orkestrel-agent-*.tgz
npm pack --pack-destination "$OUT" > "$SCAF/tmp/units/a4-4-pack.log.txt" 2>&1
echo "agent pack exit: $?" >> "$RECEIPT"
TARBALL=$(ls -t "$OUT"/orkestrel-agent-*.tgz | head -1)
echo "tarball: $TARBALL ($(stat -c %s "$TARBALL") bytes); previous kept as $KEPT ($(stat -c %s "$KEPT") bytes)" >> "$RECEIPT"
echo "runtime entry identical to the A4-3 tarball (dist/src/core/index.js): $(cmp -s <(tar -xOzf "$KEPT" package/dist/src/core/index.js) <(tar -xOzf "$TARBALL" package/dist/src/core/index.js) && echo yes || echo no)" >> "$RECEIPT"
echo "declaration differs from the A4-3 tarball (dist/src/core/index.d.ts): $(cmp -s <(tar -xOzf "$KEPT" package/dist/src/core/index.d.ts) <(tar -xOzf "$TARBALL" package/dist/src/core/index.d.ts) && echo no || echo yes)" >> "$RECEIPT"
cd "$ROOT/ollama" || exit 9
npm install --no-save --ignore-scripts --no-audit --no-fund "$TARBALL" > "$SCAF/tmp/units/a4-4-install.log.txt" 2>&1
echo "ollama install exit: $?" >> "$RECEIPT"
echo "ollama declared range after: $(grep '"@orkestrel/agent"' "$ROOT/ollama/package.json")" >> "$RECEIPT"
echo "ollama installed after: $(grep '"version"' "$ROOT/ollama/node_modules/@orkestrel/agent/package.json")" >> "$RECEIPT"
echo "installed declaration carries the corrected stop comment after: $(grep -c "$PHRASE" "$ROOT/ollama/node_modules/@orkestrel/agent/dist/src/core/index.d.ts")" >> "$RECEIPT"
echo "ollama package.json/lock dirty after install: [$(git -C "$ROOT/ollama" status --porcelain -- package.json package-lock.json | tr '\n' ' ')]" >> "$RECEIPT"
cat "$RECEIPT"
