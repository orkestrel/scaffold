// Probe: pipe write pending on backpressure (consumer never pulls the second chunk), then the
// signal aborts and the consumer cancels its reader without reading. Does the source get cancelled?
function makeSource(chunks, log) {
	let index = 0
	return new ReadableStream({
		pull(controller) {
			if (index < chunks.length) {
				controller.enqueue(new TextEncoder().encode(chunks[index]))
				index += 1
				return
			}
			return new Promise(() => {})
		},
		cancel(reason) {
			log.push(`source.cancel(${reason?.name ?? reason})`)
		},
	})
}
const settle = (ms = 50) => new Promise((resolve) => setTimeout(resolve, ms))

// E: abort, then cancel the reader with no intervening read.
{
	const log = []
	const controller = new AbortController()
	const piped = makeSource(['one', 'two'], log).pipeThrough(new TransformStream(), { signal: controller.signal })
	const reader = piped.getReader()
	await reader.read()
	await settle()
	controller.abort()
	await settle()
	log.push(`after abort, before cancel: [${log.join('|')}]`)
	try {
		await reader.cancel()
		log.push('cancel resolved')
	} catch (error) {
		log.push(`cancel rejected(${error?.name ?? error})`)
	}
	await settle()
	console.log('E abort then reader.cancel, no read:', log.join(', '))
}

// F: no abort at all; cancel the reader with a pipe write pending on backpressure.
{
	const log = []
	const piped = makeSource(['one', 'two'], log).pipeThrough(new TransformStream(), { signal: new AbortController().signal })
	const reader = piped.getReader()
	await reader.read()
	await settle()
	await reader.cancel()
	await settle()
	console.log('F reader.cancel with a pending write, no abort:', log.join(', '))
}

// G: same as E but the consumer is a for-await loop that throws after the first chunk (the engine's shape).
{
	const log = []
	const controller = new AbortController()
	const piped = makeSource(['one', 'two'], log).pipeThrough(new TransformStream(), { signal: controller.signal })
	async function* chunks(body) {
		const reader = body.getReader()
		try {
			for (;;) {
				const step = await reader.read()
				if (step.done) break
				yield step.value
			}
		} finally {
			try {
				await reader.cancel()
			} catch {
				log.push('inner cancel rejected')
			} finally {
				reader.releaseLock()
			}
		}
	}
	try {
		for await (const chunk of chunks(piped)) {
			void chunk
			await settle()
			controller.abort()
			throw new Error('consumer throws after abort')
		}
	} catch (error) {
		log.push(`loop threw(${error?.message})`)
	}
	await settle()
	console.log('G for-await consumer throws after abort:', log.join(', '))
}
