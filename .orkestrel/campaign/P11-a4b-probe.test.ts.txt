// Orchestrator probe P11 — reproduces the A4b analyst's unexecuted vectors (claims 1 and 6) in the
// real src:browser project. Run by explicit path; deleted after the reading is retained.
import { describe, expect, it } from 'vitest'
import { createModelContext, createPageServer } from '@src/browser'
import { createTool, createToolManager } from '@orkestrel/tool'
import { waitForDelay } from '@orkestrel/test'
import { installModelContext } from '../../fixtures/modelContext.js'

function outcome(promise: Promise<unknown>): Promise<string> {
	return Promise.race([
		promise.then(() => 'settled', (error: unknown) => `rejected: ${error instanceof Error ? error.message : String(error)}`),
		waitForDelay(1500).then(() => 'still pending after 1500 ms'),
	])
}

describe('P11 vectors', () => {
	it('claim 1: task requests after stop settle rather than park', async () => {
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'echoes', execute: () => 'ok' }))
		const page = createPageServer({ tools })
		await page.client.connect()
		page.stop()
		const call = await outcome(page.client.call('echo', {}))
		const task = await outcome(page.client.tasks.task('missing'))
		console.log('P11 claim 1: call after stop =', call, '| tasks.task after stop =', task)
		expect(task).not.toBe('still pending after 1500 ms')
	})

	it('claim 6: destroy from a change listener during replacement leaves no registration', async () => {
		const host = document.implementation.createHTMLDocument()
		const fixture = installModelContext(host)
		const bridge = createModelContext({ document: host })
		if (bridge === undefined) throw new Error('bridge absent')
		const tools = createToolManager()
		tools.add(createTool({ name: 'echo', description: 'A', execute: () => 'ok' }))
		await bridge.publish(tools)
		tools.add(createTool({ name: 'echo', description: 'B', execute: () => 'ok' }))
		bridge.emitter.once('change', () => {
			bridge.destroy()
		})
		await bridge.publish(tools)
		const names = fixture.registrations().map((registration) => registration.tool.name)
		console.log('P11 claim 6: registrations alive after destroy-during-replacement =', names)
		expect(names).toEqual([])
	})
})
