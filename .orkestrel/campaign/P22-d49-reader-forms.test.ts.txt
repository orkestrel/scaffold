// P22 — Orchestrator probe: which `export` spellings reach which refusal of readPolicyDeclarations
// (the declaration refusal at the accepted-list fallthrough, the unreadable-name refusal, the
// statement refusal), and the red reading of the overload pin with the accepted-list line removed
// is taken separately by mutation. Run through the setup project; remove after the reading.
import { it } from 'vitest'
import { readPolicyDeclarations } from './setupPolicy.js'

const FORMS: ReadonlyArray<readonly [string, string]> = [
	['export import', "export import Legacy = require('node:path')\n"],
	['export assignment', 'export = 1\n'],
	['namespace export', 'export as namespace Legacy\n'],
	['declare module string', "export declare module 'legacy' {}\n"],
	['declare global', 'export declare global {}\n'],
	['declare function', 'export declare function buildResult(input: string): string\n'],
	['abstract class', 'export abstract class Legacy {}\n'],
	['declare const enum', 'export declare const enum Legacy { Ready }\n'],
	['overload + namespace merge', 'export function legacy(): void\nexport function legacy() {}\nexport namespace legacy { export const value = 1 }\n'],
]

it('P22 reader forms', () => {
	for (const [label, text] of FORMS) {
		try {
			const read = readPolicyDeclarations('tests/setupServer.ts', text)
			console.log(`P22 ${label}: READ ${JSON.stringify(read)}`)
		} catch (error) {
			console.log(`P22 ${label}: THROW ${error instanceof Error ? error.message : String(error)}`)
		}
	}
})
