// Reads the A5b seam workflow's journal and prints each seam's verdict, its per-claim rulings without the
// evidence prose, and its findings outside the claims in full. Usage: node read-seams.mjs <journal.jsonl> [full]
import { readFileSync } from 'node:fs'

const [file, mode] = process.argv.slice(2)
const seams = []
for (const line of readFileSync(file, 'utf8').split('\n')) {
	if (line.trim() === '') continue
	let row
	try {
		row = JSON.parse(line)
	} catch {
		continue
	}
	if (row.type !== 'result') continue
	const value = row.result ?? row.value ?? row.output
	const parsed = typeof value === 'string' ? tryParse(value) : value
	if (parsed && parsed.seam) seams.push(parsed)
}

function tryParse(text) {
	try {
		return JSON.parse(text)
	} catch {
		return undefined
	}
}

seams.sort((a, b) => (a.seam < b.seam ? -1 : 1))
for (const seam of seams) {
	console.log(`\n===== ${seam.seam} — VERDICT: ${seam.verdict} =====`)
	for (const claim of seam.claims ?? []) {
		console.log(`  ${claim.number}: ${claim.ruling}${mode === 'full' ? ` — ${claim.evidence}` : ''}`)
	}
	for (const item of seam.outside ?? []) {
		console.log(`  outside [${item.tag}] ${item.finding}`)
	}
}
console.log(`\nseams read: ${seams.map((s) => s.seam).join(', ')}`)
