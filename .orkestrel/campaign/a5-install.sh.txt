#!/usr/bin/env bash
# Unit A5-install (Orchestrator tracked command, on the user's explicit request of 2026-09-14): declare
# @orkestrel/router and @orkestrel/server as development dependencies of agent at the catalog versions
# (router 0.0.14, server 0.0.19 — both already installed as transitive development dependencies), then
# check whether the reify reverted the G1 head start (the @orkestrel/guide tarball installed --no-save)
# and, if it did, reinstall that tarball. Receipt: scaffold/tmp/units/a5-install-receipt.md.
set -u
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
AGENT="$ROOT/agent"
RECEIPT="$SCAF/tmp/units/a5-install-receipt.md"
TARBALL="$SCAF/tmp/tarballs/orkestrel-guide-0.0.18.tgz"
{
	echo "# A5-install receipt — router and server declared as agent development dependencies"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "agent HEAD: $(git -C "$AGENT" rev-parse --short HEAD)"
	echo "agent status before: [$(git -C "$AGENT" status --porcelain | tr '\n' ' ')]"
	echo "router installed before: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/router/package.json")"
	echo "server installed before: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/server/package.json")"
	echo "guide installed before: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/guide/package.json")"
	echo "installed guide grammar admits abstract before: $(grep -c 'abstract' "$AGENT/node_modules/@orkestrel/guide/dist/src/core/index.js")"
	echo "guide tarball present: $(ls -la "$TARBALL" 2>&1)"
} > "$RECEIPT"
cd "$AGENT" || exit 9
npm install --save-dev --ignore-scripts --no-audit --no-fund "@orkestrel/router@^0.0.14" "@orkestrel/server@^0.0.19" > "$SCAF/tmp/units/a5-install.log.txt" 2>&1
echo "npm install exit: $?" >> "$RECEIPT"
echo "router installed after: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/router/package.json")" >> "$RECEIPT"
echo "server installed after: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/server/package.json")" >> "$RECEIPT"
echo "guide installed after: $(grep -m1 '"version"' "$AGENT/node_modules/@orkestrel/guide/package.json")" >> "$RECEIPT"
GRAMMAR=$(grep -c 'abstract' "$AGENT/node_modules/@orkestrel/guide/dist/src/core/index.js")
echo "installed guide grammar admits abstract after install: $GRAMMAR" >> "$RECEIPT"
if [ "$GRAMMAR" = "0" ]; then
	echo "the reify reverted the G1 head start; reinstalling the guide tarball --no-save" >> "$RECEIPT"
	npm install --no-save --ignore-scripts --no-audit --no-fund "$TARBALL" > "$SCAF/tmp/units/a5-guide-reinstall.log.txt" 2>&1
	echo "guide tarball reinstall exit: $?" >> "$RECEIPT"
	echo "installed guide grammar admits abstract after reinstall: $(grep -c 'abstract' "$AGENT/node_modules/@orkestrel/guide/dist/src/core/index.js")" >> "$RECEIPT"
fi
echo "agent status after: [$(git -C "$AGENT" status --porcelain | tr '\n' ' ')]" >> "$RECEIPT"
echo >> "$RECEIPT"
echo '```diff' >> "$RECEIPT"
git -C "$AGENT" diff -- package.json >> "$RECEIPT"
echo '```' >> "$RECEIPT"
echo >> "$RECEIPT"
echo "lockfile diff stat: $(git -C "$AGENT" diff --stat -- package-lock.json | tail -1)" >> "$RECEIPT"
echo "lockfile root devDependencies now name router/server: $(grep -c -E '"@orkestrel/(router|server)": "\^0\.0\.(14|19)"' "$AGENT/package-lock.json")" >> "$RECEIPT"
cat "$RECEIPT"
