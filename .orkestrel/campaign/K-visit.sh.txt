#!/usr/bin/env bash
# Orchestrator tracked command: the release visit of one target against the published scaffold
# (.agents/skills/orkestrel-publish/references/wave.md § Visit a repository), steps 1–8 minus the dist
# comparison (step 9 is a separate reading). Usage: bash visit.sh <checkout> <label>
# 1 re-pin every @orkestrel range to the registry caret (npm pkg set) and install; 2 the preparation commit
# (manifest + lockfile, explicit pathspec); 3 scaffold overwrite; 4 scaffold audit (exit 0 proves the sweep);
# 5 the full install; 6 the self-pin sweep is reported, not applied (grep for the prior scaffold range);
# 7 the mutating format; 8 the gates. Logs: tmp/units/visit-<checkout>-<label>*.txt in the scaffold checkout.
set -u
C="${1:?checkout}"
LABEL="${2:?label}"
SCAF=/c/Users/mikes/WebstormProjects/scaffold
T="/c/Users/mikes/WebstormProjects/$C"
LOG="$SCAF/tmp/units/visit-$C-$LABEL.log.txt"
cd "$T" || exit 9
{
	echo "# visit $C ($LABEL) — $(date -u +%FT%TZ) — head $(git rev-parse --short HEAD), dirty $(git status --short | wc -l)"
	echo "== 1 re-pin every @orkestrel range to the registry caret"
	for dep in $(node -p "const p=require('./package.json');Object.keys({...(p.dependencies||{}),...(p.devDependencies||{}),...(p.peerDependencies||{})}).filter((n)=>n.startsWith('@orkestrel/')).join(' ')"); do
		v=$(npm view "$dep" version 2>/dev/null | tail -1)
		[ -z "$v" ] && { echo "   $dep: registry read failed"; continue; }
		section=$(node -p "const p=require('./package.json');p.dependencies&&p.dependencies['$dep']?'dependencies':(p.devDependencies&&p.devDependencies['$dep']?'devDependencies':'peerDependencies')")
		before=$(node -p "require('./package.json')['$section']['$dep']")
		npm pkg set "$section.$dep=^$v" > /dev/null
		echo "   $dep: $section $before -> ^$v"
	done
	echo "== 1b npm install"
	npm install > "$SCAF/tmp/units/visit-$C-$LABEL-install1.out.txt" 2>&1
	echo "exit=$?"
	echo "== 2 preparation commit"
	git add -- package.json package-lock.json
	git -c user.name="$(git config user.name)" commit -q -m "chore: re-pin the fleet ranges to the registry before the scaffold visit" -m "Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>"
	echo "commit=$(git rev-parse --short HEAD) exit=$?"
	echo "== 3 scaffold overwrite"
	node node_modules/@orkestrel/scaffold/dist/bin/main.js overwrite > "$SCAF/tmp/units/visit-$C-$LABEL-overwrite.out.txt" 2>&1
	echo "exit=$?"
	tail -5 "$SCAF/tmp/units/visit-$C-$LABEL-overwrite.out.txt"
	echo "== 4 scaffold audit"
	node node_modules/@orkestrel/scaffold/dist/bin/main.js audit > "$SCAF/tmp/units/visit-$C-$LABEL-audit.out.txt" 2>&1
	echo "exit=$?"
	tail -4 "$SCAF/tmp/units/visit-$C-$LABEL-audit.out.txt"
	echo "== 5 full install"
	npm install > "$SCAF/tmp/units/visit-$C-$LABEL-install2.out.txt" 2>&1
	echo "exit=$?"
	echo "== 6 self-pin sweep (report only)"
	grep -rn "0\.0\.67" tests src 2>/dev/null | head -5
	echo "== 7 format"
	npm run format > "$SCAF/tmp/units/visit-$C-$LABEL-format.out.txt" 2>&1
	echo "exit=$?"
	echo "== status after the visit"
	git status --short | head -40
	echo "== 8 gates"
	for s in format:check lint:check check build test; do
		echo "== npm run $s"
		npm run "$s" > "$SCAF/tmp/units/visit-$C-$LABEL-$s.out.txt" 2>&1
		code=$?
		tail -12 "$SCAF/tmp/units/visit-$C-$LABEL-$s.out.txt" | sed 's/\x1b\[[0-9;]*m//g' | grep -E "Tests |Test Files|error|Error|exit|✓|×" | head -8
		echo "exit=$code"
	done
	echo "# end $(date -u +%FT%TZ)"
} > "$LOG" 2>&1
