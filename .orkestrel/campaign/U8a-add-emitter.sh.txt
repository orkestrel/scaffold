#!/usr/bin/env bash
# Orchestrator tracked command (U8a): declare @orkestrel/emitter ^0.0.10 as a runtime dependency of
# @orkestrel/tool (user authorization 2026-09-15, ruling 1) and install it on the host, so the bench
# unit U8 (which cannot reach the network) finds the manifest, the lockfile, and node_modules ready.
set -u
T=/c/Users/mikes/WebstormProjects/tool
SCAF=/c/Users/mikes/WebstormProjects/scaffold
RECEIPT="$SCAF/tmp/units/u8a-receipt.md"
cd "$T" || exit 9
{
	echo "# U8a receipt — emitter declared and installed in tool"
	echo
	echo "date: $(date -u +%FT%TZ)"
	echo "tool HEAD: $(git rev-parse --short HEAD)"
	echo "status before: [$(git status --porcelain | tr '\n' ' ')]"
} > "$RECEIPT"
sed -i 's|^\t\t"@orkestrel/contract": "\^0.0.17"$|\t\t"@orkestrel/contract": "^0.0.17",\n\t\t"@orkestrel/emitter": "^0.0.10"|' package.json
echo "dependencies after edit:" >> "$RECEIPT"
sed -n '/"dependencies"/,/}/p' package.json >> "$RECEIPT"
npm install --no-audit --no-fund > "$SCAF/tmp/units/u8a-install.log.txt" 2>&1
echo "npm install exit: $?" >> "$RECEIPT"
echo "installed: $(npm ls @orkestrel/emitter --depth=0 2>/dev/null | grep emitter)" >> "$RECEIPT"
echo "lockfile diff: $(git diff --stat package-lock.json | tail -1)" >> "$RECEIPT"
echo "status after: [$(git status --porcelain | tr '\n' ' ')]" >> "$RECEIPT"
cat "$RECEIPT"
