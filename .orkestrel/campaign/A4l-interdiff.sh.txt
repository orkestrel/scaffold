#!/usr/bin/env bash
# A4l interdiff — Orchestrator instrument. Recreates a scratch worktree of mcp at 7959f08 (the earlier one
# was removed at the browser-face landing), builds K from A4k-whole.patch (U4e … U4e-d) and L from the live
# tree (U4e … U4e-e, captured as A4l-whole.patch), and diffs K..L into A4l-u4e-e-only.patch. Removes the
# worktree afterwards.
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
M=/c/Users/mikes/WebstormProjects/mcp
W="/c/Users/mikes/AppData/Local/Temp/claude/C--Users-mikes-WebstormProjects-scaffold/bab586ba-1c34-4a7e-a280-b3a7e9f303d2/scratchpad/u4e-base"
WHOLE="$S/tmp/units/A4l-whole.patch"
PREV="$S/tmp/units/A4k-whole.patch"
ONLY="$S/tmp/units/A4l-u4e-e-only.patch"
git -C "$M" diff HEAD > "$WHOLE"
echo "whole chain: $(grep -c '^diff --git' "$WHOLE") files, $(wc -c < "$WHOLE") bytes"
rm -rf "$W"
git -C "$M" worktree prune
git -C "$M" worktree add -q --detach "$W" 7959f08 || { echo "worktree add failed"; exit 2; }
git -C "$W" apply "$PREV" || { echo "apply of A4k-whole.patch failed"; exit 3; }
git -C "$W" add -A
git -C "$W" -c user.name=interdiff -c user.email=interdiff@local commit -qm "state after U4e-d (scratch)"
K=$(git -C "$W" rev-parse HEAD)
git -C "$W" reset -q --hard 7959f08
git -C "$W" clean -fdq
git -C "$W" apply "$WHOLE" || { echo "apply of A4l-whole.patch failed"; exit 4; }
git -C "$W" add -A
git -C "$W" -c user.name=interdiff -c user.email=interdiff@local commit -qm "state after U4e-e (scratch)"
L=$(git -C "$W" rev-parse HEAD)
git -C "$W" diff "$K" "$L" > "$ONLY"
echo "U4e-e-only: $(grep -c '^diff --git' "$ONLY") files, $(wc -c < "$ONLY") bytes; K=$K L=$L"
git -C "$W" diff --stat "$K" "$L"
git -C "$M" worktree remove --force "$W"
git -C "$M" worktree prune
echo "worktrees: $(git -C "$M" worktree list | wc -l)"
