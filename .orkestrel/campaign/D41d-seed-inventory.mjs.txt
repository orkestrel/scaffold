import assert from 'node:assert/strict'
import { existsSync, readFileSync, writeFileSync } from 'node:fs'
import { registerHooks } from 'node:module'
import { fileURLToPath } from 'node:url'
import { format } from 'oxfmt'

// Load the source graph without rebuilding the reserved distribution directory.
const root = new URL('../../', import.meta.url)
const hooks = registerHooks({
	resolve(specifier, context, nextResolve) {
		if (specifier === '@src/core') {
			return nextResolve(new URL('src/core/index.ts', root).href, context)
		}
		if (specifier.startsWith('.') && specifier.endsWith('.js') && context.parentURL?.startsWith(new URL('src/', root).href)) {
			const source = new URL(specifier.slice(0, -3) + '.ts', context.parentURL)
			if (existsSync(source)) return nextResolve(source.href, context)
		}
		return nextResolve(specifier, context)
	},
})

try {
	const { computeManifestDigest, readSurfaceBaseline, readSurfaceCollisions } = await import('../../src/server/helpers.ts')
	const path = new URL('host.json', root)
	const inventory = JSON.parse(readFileSync(path, 'utf8'))
	const entries = JSON.stringify(inventory.entries)
	const roots = JSON.stringify(inventory.roots)
	const surface = [...readSurfaceCollisions(fileURLToPath(new URL('guides', root)))].map(([name, owners]) => ({ name, owners }))
	const digest = computeManifestDigest(inventory.entries, inventory.roots, surface)
	const formatted = await format('host.json', `${JSON.stringify({ entries: inventory.entries, roots: inventory.roots, surface, digest }, null, '\t')}\n`, JSON.parse(readFileSync(new URL('.oxfmtrc.json', root), 'utf8')))
	assert.deepEqual(formatted.errors, [])
	writeFileSync(path, formatted.code)
	assert.deepEqual(readSurfaceBaseline(fileURLToPath(path)), surface)
	const seeded = JSON.parse(readFileSync(path, 'utf8'))
	assert.equal(JSON.stringify(seeded.entries), entries)
	assert.equal(JSON.stringify(seeded.roots), roots)
	console.log(JSON.stringify({ collisions: surface.length, digest, verified: true }))
} finally {
	hooks.deregister()
}
