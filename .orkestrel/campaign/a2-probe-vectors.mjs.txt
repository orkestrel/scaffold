const encode = (text) => new TextEncoder().encode(text)
function body(chunks, close = true, log = []) {
	let index = 0
	return new ReadableStream({
		pull(controller) {
			if (index < chunks.length) {
				controller.enqueue(chunks[index])
				index += 1
				return
			}
			if (close) controller.close()
			else return new Promise(() => {})
		},
		cancel() {
			log.push('cancel')
		},
	})
}
function lines() {
	let buffer = ''
	return {
		parse(chunk) {
			buffer += chunk
			const parts = buffer.split('\n')
			buffer = parts.pop() ?? ''
			const out = []
			for (const part of parts) if (part.length > 0) { try { out.push(JSON.parse(part)) } catch {} }
			return out
		},
		clear() { buffer = '' },
	}
}
const describe = (error) => error === undefined ? 'no error' : isProviderAbortError(error) ? `ProviderAbortError partial=${JSON.stringify(error.partial)}` : isProviderError(error) ? `ProviderError ${error.code} ${error.status ?? ''} ${error.message}` : `OTHER ${error?.constructor?.name}: ${error?.message}`

// V5: inbound request signal aborts while the response queue holds an unread frame; nothing else happens.
{
	let returned = false
	let cleaned = false
	const provider = {
		id: 'p', name: 'p', async generate() { return { content: '' } },
		async *stream(_m, signal) {
			try {
				yield { channel: 'content', text: 'a' }
				yield { channel: 'content', text: 'b' }
				await new Promise((_r, reject) => signal.addEventListener('abort', () => reject(signal.reason), { once: true }))
			} finally { cleaned = true }
		},
	}
	const wrapped = { ...provider, stream(...args) { const it = provider.stream(...args); const original = it.return.bind(it); it.return = (v) => { returned = true; return original(v) }; return it } }
	const handler = createRelay({ provider: wrapped, authorize: () => true })
	const abort = new AbortController()
	const request = new Request('http://relay.test/', { method: 'POST', body: '{"messages":[]}', signal: abort.signal })
	const response = await handler(request)
	const reader = response.body.getReader()
	await reader.read()
	await new Promise((r) => setTimeout(r, 30))
	abort.abort()
	await new Promise((r) => setTimeout(r, 60))
	console.log('V5 inbound abort with queued frame, no consumer action: iterator returned =', returned, '; generator cleaned =', cleaned)
}
