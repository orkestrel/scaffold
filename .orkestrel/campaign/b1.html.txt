<pre id="out"></pre>
<script type="module">
import { createRelayProvider, isProviderError, isProviderAbortError } from '@orkestrel/agent'

const out = document.getElementById('out')
const log = (line) => {
	out.textContent += line + '\n'
	console.log(line)
}

function lines() {
	let buffer = ''
	return {
		parse(chunk) {
			buffer += chunk
			const parts = buffer.split('\n')
			buffer = parts.pop() ?? ''
			const records = []
			for (const part of parts) if (part.length > 0) records.push(JSON.parse(part))
			return records
		},
		clear() {
			buffer = ''
		},
	}
}

const provider = (token) => createRelayProvider({
	url: location.origin + '/relay',
	parser: lines,
	headers: () => ({ authorization: 'Bearer ' + token }),
})

log('user agent: ' + navigator.userAgent)

try {
	const result = await provider('fixture-token').generate([{ id: 'm1', role: 'user', content: 'hello from chromium' }], new AbortController().signal)
	log('case 1 round trip: ' + JSON.stringify(result))
} catch (error) {
	log('case 1 FAILED: ' + String(error))
}

try {
	const controller = new AbortController()
	const stream = provider('fixture-token').stream([{ id: 'm2', role: 'user', content: 'hang' }], controller.signal)
	const first = await stream.next()
	log('case 2 first delta: ' + JSON.stringify(first.value))
	const second = await stream.next()
	log('case 2 second delta: ' + JSON.stringify(second.value))
	controller.abort(new Error('browser cancel'))
	await stream.next()
	log('case 2 FAILED: no throw after the cancel')
} catch (error) {
	log('case 2 browser cancel: ' + (isProviderAbortError(error) ? 'ProviderAbortError partial=' + JSON.stringify(error.partial) : 'OTHER ' + String(error)))
}

try {
	await provider('wrong-token').generate([{ id: 'm3', role: 'user', content: 'hi' }], new AbortController().signal)
	log('case 3 FAILED: no throw on a refused token')
} catch (error) {
	log('case 3 refused token: ' + (isProviderError(error) ? 'ProviderError ' + error.code + ' ' + error.status + ' "' + error.message + '"' : 'OTHER ' + String(error)))
}

log('done')
</script>
