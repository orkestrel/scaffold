#!/usr/bin/env bash
# A5 objective audit lane — GPT-6 Astra, read-only codex exec rooted at the mcp checkout. Cap 2400 s.
# Successor of A4m-run.sh: unit name, brief, journal, evidence set (the U5 brief chain and the U5c report, the
# U5c delta, the Orchestrator's own distribution run, the gates after U5c, the plan and the D1b design record).
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
J="$SCAF/tmp/codex/A5-audit.jsonl"
L="$SCAF/tmp/codex/A5-audit-last.md"
B="$MCP/tmp/codex/A5-audit-brief.md"
cp "$SCAF/tmp/units/A5-audit-brief.md" "$B"
cp "$SCAF/tmp/units/A5-diff.patch" "$MCP/tmp/codex/A5-diff.patch"
for f in U5-mcp-distribution-brief.md U5b-mcp-distribution-brief.md U5c-mcp-distribution-brief.md U5c-mcp-distribution-report.md A5-distribution-orchestrator.log.txt U5c-mcp-gates-orchestrator.log.txt U5c-mcp-gates-test-full.log.txt plan.md D1b-design-astra.md U0f-receipt.md U0e-repack-agent-receipt.md P2-c3-probe.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f"
done
echo "launch unit=A5-audit model=gpt-6-astra sandbox=read-only cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/A5-audit.launch.txt"
timeout 2400 codex exec --json -C "$MCP" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (every evidence file it names is staged beside it in tmp/codex/). You hold the OBJECTIVE lane (analyst) and the cross-engine lane: Claude Opus 5 wrote the unit under audit. Perform the audit directly and spawn nothing. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/A5-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/A5-audit.launch.txt"
echo "unit=A5-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
