#!/usr/bin/env bash
# Mutation probe M6 on A1-fix-2: in the audit worktree at the given commit, strip the `cause` from the
# abort error and confirm the pinning test reddens; then restore the file and confirm the control is green.
SHA="$1"
cd /c/Users/mikes/WebstormProjects/agent-audit || exit 9
OUT=/c/Users/mikes/WebstormProjects/scaffold/tmp/units/a1-fix-2-probe-m6.log.txt
: > "$OUT"
git checkout --detach "$SHA" >> "$OUT" 2>&1 || exit 8
git rev-parse HEAD >> "$OUT"
echo "=== mutation: cause stripped" >> "$OUT"
sed -i 's/error === combined.reason ? undefined : { cause: error },/undefined,/' src/core/AgentProvider.ts
echo "cause occurrences after mutation: $(grep -c 'cause: error' src/core/AgentProvider.ts)" >> "$OUT"
git diff --stat >> "$OUT"
npx vitest run --config vite.config.ts --no-cache --reporter=dot --project src:core tests/src/core/AgentProvider.test.ts >> "$OUT" 2>&1
echo "=== mutation exit=$?" >> "$OUT"
echo "=== control: file restored" >> "$OUT"
git checkout -- src/core/AgentProvider.ts
git status --porcelain >> "$OUT"
npx vitest run --config vite.config.ts --no-cache --reporter=dot --project src:core tests/src/core/AgentProvider.test.ts >> "$OUT" 2>&1
echo "=== control exit=$?" >> "$OUT"
grep -E "^===|cause occurrences|Tests |AssertionError|×" "$OUT" | head -30
