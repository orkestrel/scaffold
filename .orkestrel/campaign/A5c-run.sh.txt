#!/usr/bin/env bash
# A5c objective audit lane — GPT-6 Astra, read-only codex exec rooted at the mcp checkout. Cap 2400 s.
# Successor of A5b-run.sh: the unit name, brief, journal, and evidence set (the three U5 briefs and reports,
# the A5c delta, the Orchestrator's distribution run, the gates and tail gates after U5e, the probes, and the
# A5b verdict with its lane files).
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
J="$SCAF/tmp/codex/A5c-audit.jsonl"
L="$SCAF/tmp/codex/A5c-audit-last.md"
B="$MCP/tmp/codex/A5c-audit-brief.md"
cp "$SCAF/tmp/units/A5c-audit-brief.md" "$B"
for f in U5c-mcp-distribution-brief.md U5d-mcp-distribution-brief.md U5e-mcp-receipt-residue-brief.md U5c-mcp-distribution-report.md U5d-mcp-distribution-report.md U5e-mcp-receipt-residue-report.md A5c-diff.patch.txt A5c-distribution-orchestrator.log.txt U5e-mcp-gates-orchestrator.log.txt U5e-mcp-gates-test-full.log.txt A5c-tail-gates.log.txt P25-u5d-receipt-probe.md P28-u5e-teardown-probe.md K-dist-compare-mcp-v31.txt A5b-audit-verdict.md A5b-audit-analyst.md A5b-audit-reviewer.md A5b-audit-checker.md A5b-seam-verdicts.md plan.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f" 2>/dev/null || echo "staged: missing $f"
done
echo "launch unit=A5c-audit model=gpt-6-astra sandbox=read-only cap=2400 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/A5c-audit.launch.txt"
timeout 2400 codex exec --json -C "$MCP" --sandbox read-only --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the audit brief at $B exactly (every evidence file it names is staged beside it in tmp/codex/). You hold the OBJECTIVE lane (analyst) and the CROSS-ENGINE lane: Claude Opus 5 wrote every unit under audit, so yours is the only lane whose engine did not write them. Perform the audit directly and spawn nothing; the sandbox runs no Chromium and no install, so name each unexecuted vector UNRESOLVED with its exact command and read the Orchestrator's own runs instead. Your final message must be the verdict shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/A5c-audit.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/A5c-audit.launch.txt"
echo "unit=A5c-audit exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
