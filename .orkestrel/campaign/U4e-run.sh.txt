#!/usr/bin/env bash
# U4e implementation lane — GPT-6 Astra (sol route), workspace-write codex exec rooted at the mcp checkout. Cap 3000 s
# (observed high marks: D46 and the A-lanes ran well under 2700 s; this unit adds a core producer, its tests, and the
# executed guide proof). Stages the campaign plan and the U3d pair (the Output shape the brief names) beside the brief.
# Run only after the U4 chain landed and CHECKPOINT_LINE in the brief names that commit.
set -u
SCAF="C:/Users/mikes/WebstormProjects/scaffold"
MCP="C:/Users/mikes/WebstormProjects/mcp"
U="U4e-mcp-list-changed"
J="$SCAF/tmp/codex/$U.jsonl"
L="$SCAF/tmp/codex/$U-last.md"
B="$MCP/tmp/codex/$U-brief.md"
cp "$SCAF/tmp/units/$U-brief.md" "$B"
for f in plan.md U3d-mcp-finish-brief.md U3d-mcp-finish-report.md; do
	cp "$SCAF/.orkestrel/campaign/$f" "$MCP/tmp/codex/$f"
done
if grep -q CHECKPOINT_LINE "$B"; then echo "refused: CHECKPOINT_LINE is unfilled in $B"; exit 8; fi
echo "launch unit=$U model=gpt-6-astra sandbox=workspace-write cap=3000 start=$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$SCAF/tmp/codex/$U.launch.txt"
timeout 3000 codex exec --json -C "$MCP" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$L" "Read and execute the unit brief at $B exactly (the campaign plan and the U3d brief and report are staged beside it in tmp/codex/). You hold the sol implementation route (GPT-6 Astra). Perform the assignment directly and spawn nothing. Never touch package.json, package-lock.json, dist/, src/browser/, tests/src/browser/, guides/tool.md, the scaffold repair set, or any file the brief lists off-limits. Your final message must be the Output shape the brief specifies." < /dev/null > "$J" 2> "$SCAF/tmp/codex/$U.err"
EXIT=$?
echo "exit=$EXIT end=$(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$SCAF/tmp/codex/$U.launch.txt"
echo "unit=$U exit=$EXIT"
grep -o '"thread_id":"[^"]*"' "$J" | head -1
wc -c < "$L" 2>/dev/null
