#!/usr/bin/env bash
# Orchestrator probe (successor of collide2.sh; the A13 reviewer read that the pattern missed `export let`,
# `export type`/`interface`, and re-export forms): name collisions between one checkout's own exported
# declarations in src/** and tests/** and the exports of the installed @orkestrel/test (core, browser, server)
# and @orkestrel/contract. Covers function, const, let, class, interface, type, enum, and `export { a, b as c }`
# re-export lists. Prints the pattern, the paths swept, every hit, and an explicit "collisions: none" line.
# Read-only. Usage: bash collide3.sh <checkout>
set -u
C="${1:?checkout}"
ROOT=/c/Users/mikes/WebstormProjects
NM="$ROOT/$C/node_modules/@orkestrel"
LIST=$(mktemp)
for f in "$NM"/test/dist/src/core/index.d.ts "$NM"/test/dist/src/browser/index.d.ts "$NM"/test/dist/src/server/index.d.ts "$NM"/contract/dist/src/core/index.d.ts; do
	[ -f "$f" ] || continue
	grep -oE 'export declare (async )?(function|const|let|class|interface|type|enum) [A-Za-z_][A-Za-z0-9_]*' "$f" | awk '{print $NF}'
done | sort -u > "$LIST"
echo "probe: export-name collisions in $C (declarations and re-export lists)"
echo "date: $(date -u +%FT%TZ)"
echo "installed: test $(grep -o '"version": *"[^"]*"' "$NM/test/package.json" | head -1), contract $(grep -o '"version": *"[^"]*"' "$NM/contract/package.json" | head -1); export names read: $(wc -l < "$LIST")"
echo "pattern A: ^export (async )?(function|const|let|class|interface|type|enum) NAME\\b"
echo "pattern B: NAME inside an export { ... } list (as its exported spelling)"
echo "paths: $ROOT/$C/src/** and $ROOT/$C/tests/** (node_modules excluded)"
files=$(find "$ROOT/$C/src" "$ROOT/$C/tests" -type f -name '*.ts' 2>/dev/null | wc -l)
echo "files swept: $files"
hits=0
while read -r name; do
	[ -n "$name" ] || continue
	while read -r line; do
		[ -n "$line" ] || continue
		hits=$((hits + 1))
		echo "  collision: $name at ${line#$ROOT/$C/}"
	done < <(grep -rn -E "^export (async )?(function|const|let|class|interface|type|enum) $name\b" "$ROOT/$C/src" "$ROOT/$C/tests" 2>/dev/null | grep -v node_modules)
	while read -r line; do
		[ -n "$line" ] || continue
		hits=$((hits + 1))
		echo "  collision (re-export list): $name at ${line#$ROOT/$C/}"
	done < <(grep -rn -E "^export \{[^}]*\b(as )?$name\b[^}]*\}" "$ROOT/$C/src" "$ROOT/$C/tests" 2>/dev/null | grep -v node_modules)
done < "$LIST"
if [ "$hits" -eq 0 ]; then echo "collisions: none"; else echo "collisions: $hits"; fi
rm -f "$LIST"
