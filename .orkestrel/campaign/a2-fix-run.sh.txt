#!/usr/bin/env bash
# Unit A2-fix (tmp/units/a2-fix-brief.md: findings F10–F19 of audit round A2-R1 on the relay).
# Cap: 5400 s. Journal: scaffold/tmp/codex/a2-fix.jsonl. Report: agent/tmp/units/a2-fix-report.md (and a2-fix-last.md).
ROOT=/c/Users/mikes/WebstormProjects
SCAF="$ROOT/scaffold"
AGENT="C:/Users/mikes/WebstormProjects/agent"
cd "$SCAF" || exit 9
cp "$SCAF/tmp/units/a2-fix-brief.md" "$AGENT/tmp/units/a2-fix-brief.md" || exit 8
cmp "$SCAF/tmp/units/a2-fix-brief.md" "$AGENT/tmp/units/a2-fix-brief.md" || exit 7
git -C "$ROOT/agent" status --porcelain > "$SCAF/tmp/codex/a2-fix.status-before.txt"
git -C "$ROOT/agent" rev-parse --short HEAD >> "$SCAF/tmp/codex/a2-fix.status-before.txt"
date -u +%FT%TZ > "$SCAF/tmp/codex/a2-fix.launch.txt"
timeout 5400 codex exec --json -C "$AGENT" --sandbox workspace-write --model gpt-6-astra -c 'model_reasoning_effort="high"' --output-last-message "$AGENT/tmp/units/a2-fix-last.md" "Read tmp/units/a2-fix-brief.md and execute it exactly. It is unit A2-fix, the fix round after audit A2-R1 on the relay. Complete the whole unit. Your final message must be the report the brief specifies, and you also write that report to tmp/units/a2-fix-report.md." < /dev/null > "$SCAF/tmp/codex/a2-fix.jsonl" 2> "$SCAF/tmp/codex/a2-fix.err"
echo "exit=$?" >> "$SCAF/tmp/codex/a2-fix.launch.txt"
git -C "$ROOT/agent" status --porcelain > "$SCAF/tmp/codex/a2-fix.status-after.txt"
date -u +%FT%TZ >> "$SCAF/tmp/codex/a2-fix.launch.txt"
