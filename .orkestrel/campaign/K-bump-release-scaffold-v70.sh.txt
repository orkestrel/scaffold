#!/usr/bin/env bash
# Orchestrator tracked command: bump scaffold to 0.0.70 (its vendored `dist/host` surface moved — the hosted
# mcp and ollama mirrors were refreshed from their pushed mains), refresh the three generated-manifest fixtures
# that carry this package's own range, then run the release proof.
# The fixture step is not optional: those `.txt` files are snapshots of a manifest this package generates, so
# they carry its own caret and redden on the self-pin sweep at every bump. The 0.0.69 release learned this.
# Log: scaffold tmp/units/release-scaffold-v70.log.txt
set -u
S=/c/Users/mikes/WebstormProjects/scaffold
cd "$S" || exit 9
echo "== npm version 0.0.70 --no-git-tag-version"
npm version 0.0.70 --no-git-tag-version
echo "exit=$?"
echo "== refresh the generated-manifest fixtures that carry this package's own range"
npx vitest run --config vite.config.ts --no-cache --reporter=dot --project src:core tests/src/core/compilers.test.ts -u 2>&1 | sed 's/\x1b\[[0-9;]*m//g' | grep -E 'Snapshots|Tests |Test Files|FAIL' | head
echo "== what the bump and the refresh moved"
git diff --stat -- package.json package-lock.json tests/src/core/fixtures
echo "== any 0.0.69 literal left in the fixtures"
grep -c '0\.0\.69' tests/src/core/fixtures/*.txt 2>/dev/null | grep -v ':0' || echo "none"
bash "$S/tmp/units/release-scaffold-chain.sh" v70
echo "== chain done"
cat "$S/tmp/units/release-scaffold-v70.log.txt"
