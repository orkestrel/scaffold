# Orchestrator tracked command: remove a node_modules junction (the link only, never its target), then report.
param([string]$Link)
$item = Get-Item -LiteralPath $Link -ErrorAction SilentlyContinue
if ($null -eq $item) {
	"absent: $Link"
} elseif ($item.LinkType -eq 'Junction') {
	$item.Delete()
	"junction removed: $Link"
} else {
	"not a junction, left alone: $Link ($($item.LinkType))"
}
