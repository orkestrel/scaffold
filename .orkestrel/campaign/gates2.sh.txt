#!/usr/bin/env bash
# Orchestrator tracked command: authoritative non-mutating gates in one checkout, in AGENTS.md order.
# Successor of gates.sh: adds `npm run build` between `check` and `test` (the A8 reviewer read the
# missing build block). Usage: bash gates2.sh <checkout> <label>
set -u
C="${1:?checkout}"
LABEL="${2:?label}"
SCAF=/c/Users/mikes/WebstormProjects/scaffold
LOG="$SCAF/tmp/units/gates-$LABEL.log.txt"
cd "/c/Users/mikes/WebstormProjects/$C" || exit 9
{
	echo "# $C gates ($LABEL) — $(date -u +%FT%TZ) — format:check, lint:check, check, build, test"
	for s in format:check lint:check check build test; do
		echo "== npm run $s"
		npm run "$s" > "$SCAF/tmp/units/gates-$LABEL-$s.out.txt" 2>&1
		code=$?
		tail -40 "$SCAF/tmp/units/gates-$LABEL-$s.out.txt" | sed 's/\x1b\[[0-9;]*m//g'
		echo "exit=$code"
	done
} > "$LOG" 2>&1
grep -n '^== \|^exit=' "$LOG"
grep -E "Tests +[0-9]+ passed" "$SCAF/tmp/units/gates-$LABEL-test.out.txt" | sed 's/\x1b\[[0-9;]*m//g'
