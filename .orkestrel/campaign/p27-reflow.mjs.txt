// P27 — Orchestrator probe, read-only: find every hunk in a unified diff whose removed and added
// lines are the same text differently wrapped. That is formatter residue a formatter which preserves
// author line breaks inside an object literal cannot see. Usage: node p27-reflow.mjs <patch file>
import { readFileSync } from 'node:fs'

const text = readFileSync(process.argv[2], 'utf8')
const lines = text.split('\n')

function normalize(parts) {
	return parts
		.join(' ')
		.replace(/\s+/gu, '')
		.replace(/,(?=[)\]}])/gu, '')
		.trim()
}

let header = ''
let removed = []
let added = []
const hits = []

function settle() {
	if (removed.length === 0 || added.length === 0) return
	if (normalize(removed) === normalize(added)) {
		hits.push({ header, removed: removed.length, added: added.length, text: removed.join(' ').trim() })
	}
	removed = []
	added = []
}

for (const line of lines) {
	if (line.startsWith('@@')) {
		settle()
		header = line
		continue
	}
	if (line.startsWith('-') && !line.startsWith('---')) {
		if (added.length > 0) settle()
		removed.push(line.slice(1))
		continue
	}
	if (line.startsWith('+') && !line.startsWith('+++')) {
		added.push(line.slice(1))
		continue
	}
	settle()
}
settle()

console.log(`reflow-only hunks: ${hits.length}`)
for (const hit of hits) console.log(`  ${hit.header}\n    ${hit.text.slice(0, 120)}`)
