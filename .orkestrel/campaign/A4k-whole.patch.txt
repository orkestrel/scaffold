+	const tools = createToolManager()

+	const controller = new AbortController()

+		createJSONRPCRequest({
+			method: 'subscriptions/listen',

+			},
+		}),

+	)

+	return {
+		tools,
+		stream,
+		controller,
+		close() {

+	}
