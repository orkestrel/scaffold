import { readSkillExports } from '../../tests/setupPolicy.js'
import { expect, it } from 'vitest'

it('dumps the installed test entries', () => {
	const core = readSkillExports(process.cwd(), '@orkestrel/test')
	const browser = readSkillExports(process.cwd(), '@orkestrel/test/browser')
	console.log('CORE OUTCOME', core.outcome, core.names.length)
	console.log('CORE', core.names.join(' '))
	console.log('BROWSER OUTCOME', browser.outcome, browser.names.length)
	console.log('BROWSER', browser.names.join(' '))
	expect(core.outcome).toBe('read')
	expect(browser.outcome).toBe('read')
})
