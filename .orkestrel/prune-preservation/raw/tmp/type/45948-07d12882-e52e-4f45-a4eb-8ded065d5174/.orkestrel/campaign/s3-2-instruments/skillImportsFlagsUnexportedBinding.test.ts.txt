import { readFileSync } from 'node:fs'
import { describe, expect, it } from 'vitest'
import { inspectSkillImports } from '../../../tests/setupPolicy.ts'

describe('inspectSkillImports over a planted unresolved binding', () => {
	it('reports one violation naming the unexported buildRefusal2 binding', () => {
		const path = 'tmp/probe/s3-2/planted-statechart.md'
		const content = readFileSync(path, 'utf8')
		const violations = inspectSkillImports(process.cwd(), path, content)
		expect(violations).toHaveLength(1)
		expect(violations[0]?.message).toContain('buildRefusal2')
	})
})
