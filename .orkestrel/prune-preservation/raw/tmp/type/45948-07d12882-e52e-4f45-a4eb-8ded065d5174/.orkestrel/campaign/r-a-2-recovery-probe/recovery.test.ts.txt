import { describe, expect, it } from 'vitest'
import { page } from 'vitest/browser'
import { readName, readPage, readRefusal, resolveAccessible } from '@orkestrel/test/browser'
import { CONTACT_PATH, COPY, PRODUCTS_PATH, hashHref } from '@app/browser'
import { clearSurface, openSurface } from '../../../tests/app/browser/setup.js'
import { CASES, VIEWPORTS } from './setup.js'

describe('recovered empty catalog Contact resolution', () => {
	it('resolves Contact to its route when empty catalog notices are painted', async () => {
		try {
			for (const viewport of VIEWPORTS) {
				for (const reached of CASES) {
					await page.viewport(viewport.width, viewport.height)
					await openSurface({ catalog: reached.catalog })
					window.location.hash = hashHref(reached.path)
					await expect
						.poll(() => readPage(), { timeout: 4_000 })
						.toContain(reached.notice)
					expect(readRefusal(COPY.contact)).toBeUndefined()
					const contact = resolveAccessible('link', COPY.contact)
					const destination = contact.getAttribute('href')
					expect(destination).toBe(hashHref(CONTACT_PATH))
					console.info(
						`RA2_RECOVERY viewport=${String(viewport.width)}x${String(viewport.height)} route=${reached.path} name=${readName(contact)} destination=${destination ?? 'undefined'}`,
					)
					clearSurface()
				}
			}
		} finally {
			clearSurface()
			await page.viewport(1280, 800)
		}
	})

	it('refuses an added duplicate Contact control and resolves after its removal', async () => {
		let duplicate: HTMLAnchorElement | undefined
		try {
			await page.viewport(1280, 800)
			await openSurface({ catalog: { products: [] } })
			window.location.hash = hashHref(PRODUCTS_PATH)
			await expect
				.poll(() => readPage(), { timeout: 4_000 })
				.toContain('No products in this catalog yet.')
			expect(readRefusal(COPY.contact)).toBeUndefined()
			duplicate = document.createElement('a')
			duplicate.href = hashHref(CONTACT_PATH)
			duplicate.textContent = COPY.contact
			document.body.append(duplicate)
			expect(readRefusal(COPY.contact)).toBe(
				`Interactive target "${COPY.contact}" is ambiguous across 2 elements`,
			)
			duplicate.remove()
			duplicate = undefined
			expect(readRefusal(COPY.contact)).toBeUndefined()
		} finally {
			duplicate?.remove()
			clearSurface()
			await page.viewport(1280, 800)
		}
	})
})
