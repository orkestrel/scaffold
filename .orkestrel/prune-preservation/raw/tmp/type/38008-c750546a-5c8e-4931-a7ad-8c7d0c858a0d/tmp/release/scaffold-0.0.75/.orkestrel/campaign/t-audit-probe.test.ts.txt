// Orchestrator reproduction of the T1+T2 audit's objective-lane vectors that predict a defect:
// claims 3, 4, 5, 6, 15, 16, 17. A probe: it settles the findings and is deleted afterwards.
// Placed at tests/src/browser/probe.test.ts for the run and removed before any writer is briefed.
import type { StateScenario } from '@src/core'
import { createSignal, STATECHART_ATTRIBUTES, STATECHART_STATUSES } from '@src/core'
import {
	build,
	createHarness,
	createStorage,
	mount,
	readRefusal,
	waitForAnimations,
	waitForState,
} from '@src/browser'
import { afterEach, describe, expect, it } from 'vitest'

const mounted: Element[] = []
afterEach(() => {
	for (const element of mounted.splice(0)) element.remove()
})

describe('T audit vectors', () => {
	it('claim 17: a builder returning undefined', async () => {
		const phases: string[] = []
		const scenarios: ReadonlyArray<StateScenario<'closed', 'toggle', undefined>> = [
			{
				transition: { name: 'row', from: 'closed', event: 'toggle', to: 'closed' },
				arrange() {
					phases.push('arrange')
				},
				act() {
					phases.push('act')
				},
				assert() {
					phases.push('assert')
					throw new Error('the assertion that must fail')
				},
			},
		]
		const harness = createHarness({ scenarios, build: () => undefined, state: () => 'closed' })
		try {
			await harness.execute()
			console.log('claim 17:', JSON.stringify({ phases, status: harness.status, passed: harness.passed, failed: harness.failed, failures: harness.failures }))
		} finally {
			harness.destroy()
		}
	})

	it('claim 16: a throwing state reader', async () => {
		const scenarios: ReadonlyArray<StateScenario<'closed', 'toggle', { flag: boolean }>> = [
			{
				transition: { name: 'row', from: 'closed', event: 'toggle', to: 'closed' },
				arrange() {},
				act() {},
				assert() {},
			},
		]
		const harness = createHarness({
			scenarios,
			build: () => ({ flag: true }),
			state: () => {
				throw new Error('reader broke')
			},
		})
		try {
			let rejection: unknown
			try {
				await harness.execute()
			} catch (cause) {
				rejection = cause
			}
			console.log('claim 16:', JSON.stringify({ rejection: rejection instanceof Error ? rejection.message : String(rejection), status: harness.root.getAttribute(STATECHART_ATTRIBUTES.status), statuses: STATECHART_STATUSES }))
		} finally {
			harness.destroy()
		}
	})

	it('claim 15: rows sharing a name', async () => {
		const scenarios: ReadonlyArray<StateScenario<'closed', 'toggle', { n: number }>> = [
			{ transition: { name: 'same', from: 'closed', event: 'toggle', to: 'closed' }, arrange() {}, act() {}, assert() {} },
			{ transition: { name: 'same', from: 'closed', event: 'toggle', to: 'closed' }, arrange() {}, act() {}, assert() { throw new Error('second fails') } },
		]
		const harness = createHarness({ scenarios, build: () => ({ n: 1 }), state: () => 'closed' })
		try {
			await harness.execute()
			console.log('claim 15:', JSON.stringify({ failures: harness.failures, results: [...harness.root.querySelectorAll(`[${STATECHART_ATTRIBUTES.result}]`)].map((row) => row.getAttribute(STATECHART_ATTRIBUTES.result)) }))
		} finally {
			harness.destroy()
		}
	})

	it('claim 4: listeners left on a shared signal after settled waits', async () => {
		const signal = createSignal()
		const box = mount(build('div', { attributes: { style: 'width: 20px; height: 20px; transition: width 30ms' } }))
		mounted.push(box)
		const before = signal.count
		for (let index = 0; index < 5; index += 1) {
			await waitForAnimations(box, { signal: signal.signal, budget: 200 })
		}
		console.log('claim 4:', JSON.stringify({ before, after: signal.count }))
	})

	it('claim 5: a hostile target throwing undefined', () => {
		const button = mount(build('button', { text: 'Hostile' }))
		mounted.push(button)
		Object.defineProperty(button, 'tabIndex', {
			get() {
				throw undefined
			},
		})
		let threw = false
		let reading: string | undefined = 'unset'
		try {
			reading = readRefusal('Hostile')
		} catch {
			threw = true
		}
		console.log('claim 5:', JSON.stringify({ threw, reading }))
	})

	it('claim 6: an unsafe integer quota and an overwrite under quota', () => {
		const unsafe = createStorage({ quota: Number.MAX_SAFE_INTEGER + 1 })
		unsafe.setItem('a', '1')
		const limited = createStorage({ quota: 1 })
		limited.setItem('k', '1')
		let overwrite = 'accepted'
		try {
			limited.setItem('k', '2')
		} catch (cause) {
			overwrite = cause instanceof DOMException ? cause.name : String(cause)
		}
		const seed = { a: '1' }
		const seeded = createStorage({ values: seed })
		seed.a = 'mutated'
		console.log('claim 6:', JSON.stringify({ unsafeAccepted: true, overwrite, seededAfterMutation: seeded.getItem('a') }))
	})

	it('claim 3: an abort reason shaped like the timeout voice', async () => {
		const button = mount(build('button', { text: 'Pin', attributes: { 'aria-pressed': 'false' } }))
		mounted.push(button)
		const controller = new AbortController()
		const reason = new Error('Condition ""Pin" to announce "pressed=true"" did not hold forged')
		controller.abort(reason)
		let rejection: unknown
		try {
			await waitForState('Pin', 'pressed', { signal: controller.signal, budget: 100 })
		} catch (cause) {
			rejection = cause
		}
		console.log('claim 3:', JSON.stringify({ identity: rejection === reason, message: rejection instanceof Error ? rejection.message : String(rejection) }))
		expect(rejection).toBe(reason)
	})
})
