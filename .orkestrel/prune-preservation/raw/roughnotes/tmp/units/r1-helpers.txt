/** The number a non-string plugin name carries in these proofs. */
const NUMBERED_NAME = 42

/** Builds a named entry, so a proof compares entries by reference rather than by shape. */
function createNamedEntry(name: string): Plugin {
	return { name }
}

/**
 * Builds an entry whose `name` is not a string. The plugin type declares `name` as a string, so
 * the number is written after construction, which is how an untyped plugin reaches the merge.
 */
function createNumberedEntry(): Plugin {
	const entry: Plugin = { name: 'conformance-numbered' }
	Reflect.set(entry, 'name', NUMBERED_NAME)
	return entry
}

/** Reads the entries a configuration carries, and fails loudly when it carries none. */
function readPlugins(config: UserConfig): readonly PluginOption[] {
	const plugins = config.plugins
	if (plugins === undefined) throw new Error('The configuration carries no plugins')
	return plugins
}

/** Keys entries by the `name` each carries, which is the selection this merge departs from. */
function keyByName(plugins: readonly Plugin[]): readonly Plugin[] {
	const keyed = new Map<unknown, Plugin>()
	for (const plugin of plugins) keyed.set(readField(plugin, 'name'), plugin)
	return [...keyed.values()]
}

/**
 * A value carrying `command` without `mode`. No Vitest invocation record carries `command` alone,
 * so this is an override and merges.
 */
const COMMAND_ONLY = Object.freeze({ command: 'serve', base: '/conformance-command-only/' })
