// Orchestrator reproduction probe, second file: claim 1's remaining transitions — (c) a same-view
// navigation while the menu is open, and (d) widening past the breakpoint while the menu is open.
// Collected by the app:browser project for one run and deleted afterwards.
import { describe, expect, it } from 'vitest'
import { page } from 'vitest/browser'
import { readStates, resolveAccessible, waitForState } from '@orkestrel/test/browser'
import { COPY } from '@app/browser'
import { clearSurface, openSite, openSurface } from './setup.js'

async function settle(): Promise<void> {
	await new Promise((resolve) => setTimeout(resolve, 700))
}

describe('audit probe 2', () => {
	it('claim 1c: same-view navigation while the menu is open', async () => {
		await page.viewport(390, 844)
		await openSurface()
		window.location.hash = '#/products/roughnotes-pro'
		await settle()
		const trigger = resolveAccessible('button', COPY.menu)
		const menu = document.getElementById('site-menu')
		const events: string[] = []
		for (const name of ['hide.bs.offcanvas', 'hidden.bs.offcanvas']) {
			menu?.addEventListener(name, () => events.push(name))
		}
		await openSite()
		await settle()
		console.info(`PROBE c1c open on detail: ${JSON.stringify(readStates(trigger))} hash=${window.location.hash}`)
		window.location.hash = '#/products/advantage-plus'
		await settle()
		console.info(`PROBE c1c after same-view hash change: ${JSON.stringify(readStates(trigger))} events=${events.join(',') || 'none'} menu classes=${menu?.className ?? ''} hash=${window.location.hash}`)
		window.location.hash = '#/about'
		await settle()
		console.info(`PROBE c1c after cross-view hash change: ${JSON.stringify(readStates(trigger))} events=${events.join(',') || 'none'} menu classes=${menu?.className ?? ''}`)
		clearSurface()
		expect(true).toBe(true)
	})

	it('claim 1d: widening past the breakpoint while the menu is open', async () => {
		await page.viewport(390, 844)
		await openSurface()
		const trigger = resolveAccessible('button', COPY.menu)
		const menu = document.getElementById('site-menu')
		const events: string[] = []
		for (const name of ['hide.bs.offcanvas', 'hidden.bs.offcanvas']) {
			menu?.addEventListener(name, () => events.push(name))
		}
		await openSite()
		await settle()
		console.info(`PROBE c1d open at 390: ${JSON.stringify(readStates(trigger))}`)
		await page.viewport(1280, 800)
		await settle()
		console.info(`PROBE c1d after widening: ${JSON.stringify(readStates(trigger))} attribute=${trigger.getAttribute('aria-expanded') ?? 'none'} trigger visible=${String(trigger.checkVisibility())} events=${events.join(',') || 'none'} menu classes=${menu?.className ?? ''}`)
		let waited = 'absent within budget'
		await waitForState(COPY.menu, 'expanded', { absent: true, budget: 2000 }).catch((cause: unknown) => {
			waited = cause instanceof Error ? cause.message.slice(0, 200) : String(cause)
		})
		console.info(`PROBE c1d waitForState(expanded, absent) at 1280: ${waited}`)
		await page.viewport(390, 844)
		await settle()
		console.info(`PROBE c1d back at 390: ${JSON.stringify(readStates(trigger))} menu classes=${menu?.className ?? ''}`)
		clearSurface()
		expect(true).toBe(true)
	})
})
