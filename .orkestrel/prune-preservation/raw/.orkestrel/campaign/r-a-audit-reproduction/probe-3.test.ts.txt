// Orchestrator reproduction probe, third file: claim 2's binding-removed reading — with the
// `:aria-expanded` binding removed from App.vue by the mutation script, open the menu through the
// setup module's `openSite` (which settles on Bootstrap's own class), record `shown.bs.offcanvas`,
// and read the trigger. Collected by the app:browser project for one run and deleted afterwards.
import { describe, expect, it } from 'vitest'
import { page } from 'vitest/browser'
import { readStates, resolveAccessible } from '@orkestrel/test/browser'
import { COPY } from '@app/browser'
import { clearSurface, openSite, openSurface } from './setup.js'

describe('audit probe 3', () => {
	it('claim 2: the trigger after shown with the binding removed', async () => {
		await page.viewport(390, 844)
		await openSurface()
		const trigger = resolveAccessible('button', COPY.menu)
		const menu = document.getElementById('site-menu')
		let shown = 0
		menu?.addEventListener('shown.bs.offcanvas', () => {
			shown += 1
		})
		console.info(`PROBE c2 before: states=${JSON.stringify(readStates(trigger))} attribute=${trigger.getAttribute('aria-expanded') ?? 'null'}`)
		await openSite()
		await new Promise((resolve) => setTimeout(resolve, 700))
		console.info(`PROBE c2 after: shown events=${String(shown)} states=${JSON.stringify(readStates(trigger))} attribute=${trigger.getAttribute('aria-expanded') ?? 'null'} menu classes=${menu?.className ?? ''}`)
		clearSurface()
		expect(true).toBe(true)
	})
})
